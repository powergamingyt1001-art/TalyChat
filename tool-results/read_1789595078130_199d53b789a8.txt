    72→    72→function bubbleStyleClass(style: string, isMine: boolean): string {

    74→    74→    case 'sharp':
    75→    75→      return 'bubble-sharp rounded-none'
    76→    76→    case 'tail':
    77→    77→      return isMine ? 'bubble-tail-sent' : 'bubble-tail-received'
    78→    78→    case 'none':
    79→    79→      return 'bubble-none'
    80→    80→    default:
    81→    81→      return '' // default bubble shape
    82→    82→  }

   543→   543→interface MessageActionMenuProps {

   559→   559→
   560→   560→export function MessageActionMenu({
   561→   561→  open,
   562→   562→  message,
   563→   563→  isMine,
   564→   564→  anchorRect,
   565→   565→  onClose,
   566→   566→  onReply,
   567→   567→  onCopy,
   568→   568→  onForward,
   569→   569→  onEdit,
   570→   570→  onDelete,
   571→   571→  onPin,
   572→   572→  onReact,
   573→   573→  onSchedule,
   574→   574→}: MessageActionMenuProps) {
   575→   575→  // Backdrop + a popover positioned above the anchor.
   576→   576→  const [showExpandedReactions, setShowExpandedReactions] = React.useState(false)
   577→   577→  if (!open || !message) return null
   578→   578→
   579→   579→  // Compute position (default: above anchor, centered). Fall back to top center.
   580→   580→  const style: React.CSSProperties = (() => {
   581→   581→    if (anchorRect) {
   582→   582→      const top = Math.max(8, anchorRect.top - 64)
   583→   583→      const left = Math.min(
   584→   584→        Math.max(8, anchorRect.left + anchorRect.width / 2 - 150),
   585→   585→        typeof window !== 'undefined' ? window.innerWidth - 308 : 1000
   586→   586→      )
   587→   587→      return { top, left }
   588→   588→    }
   589→   589→    return { top: 80, left: 8 }
   590→   590→  })()
   591→   591→
   592→   592→  return (
   593→   593→    <div
   594→   594→      className="fixed inset-0 z-50"
   595→   595→      onClick={onClose}
   596→   596→      onContextMenu={(e) => {
   597→   597→        e.preventDefault()
   598→   598→        onClose()
   599→   599→      }}
   600→   600→    >
   601→   601→      <div
   602→   602→        className="absolute"
   603→   603→        style={style}
   604→   604→        onClick={(e) => e.stopPropagation()}
   605→   605→      >
   606→   606→        {/* Reaction emoji row */}
   607→   607→        <div className="mb-1 flex items-center gap-1 rounded-full border bg-popover px-2 py-1 shadow-md">
   608→   608→          {QUICK_REACTIONS.map((emoji) => (
   609→   609→            <button
   610→   610→              key={emoji}
   611→   611→              type="button"
   612→   612→              onClick={() => {
   613→   613→                onReact(emoji)
   614→   614→                onClose()

   617→   617→              className="flex h-9 w-9 min-h-[44px] min-w-[44px] items-center justify-center rounded-full text-xl transition-transform hover:scale-125"
   618→   618→            >
   619→   619→              {emoji}
   620→   620→            </button>
   621→   621→          ))}
   622→   622→          {/* Expand button for more reactions */}
   623→   623→          <button
   624→   624→            type="button"
   625→   625→            onClick={(e) => {
   626→   626→              e.stopPropagation()
   627→   627→              setShowExpandedReactions((v) => !v)
   628→   628→            }}
   629→   629→            aria-label="More reactions"
   630→   630→            aria-expanded={showExpandedReactions}
   631→   631→            className="flex h-9 w-9 min-h-[44px] min-w-[44px] items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-accent"
   632→   632→          >
   633→   633→            <SmilePlus className="h-5 w-5" />
   634→   634→          </button>
   635→   635→        </div>
   636→   636→
   637→   637→        {/* Expanded reactions grid */}
   638→   638→        {showExpandedReactions && (
   639→   639→          <div className="mb-1 grid w-[280px] max-w-[calc(100vw-16px)] grid-cols-8 gap-1 rounded-md border bg-popover p-2 shadow-md">
   640→   640→            {EXTENDED_REACTIONS.map((emoji) => (
   641→   641→              <button
   642→   642→                key={emoji}
   643→   643→                type="button"
   644→   644→                onClick={() => {
   645→   645→                  onReact(emoji)
   646→   646→                  onClose()
   647→   647→                }}
   648→   648→                aria-label={`React with ${emoji}`}
   649→   649→                className="flex h-8 w-8 items-center justify-center rounded-full text-lg transition-transform hover:scale-125 hover:bg-accent"
   650→   650→              >
   651→   651→                {emoji}
   652→   652→              </button>
   653→   653→            ))}
   654→   654→          </div>
   655→   655→        )}
   656→   656→
   657→   657→        {/* Action items */}
   658→   658→        <div className="flex w-[300px] max-w-[calc(100vw-16px)] flex-col gap-0.5 rounded-md border bg-popover p-1 shadow-md">
   659→   659→          <ActionMenuItem icon={<Reply className="h-4 w-4" />} label="Reply" onClick={() => { onReply(); onClose() }} />
   660→   660→          <ActionMenuItem icon={<Copy className="h-4 w-4" />} label="Copy" onClick={() => { onCopy(); onClose() }} />
   661→   661→          <ActionMenuItem icon={<Forward className="h-4 w-4" />} label="Forward" onClick={() => { onForward(); onClose() }} />
   662→   662→          {onSchedule && (

   664→   664→              icon={<Clock className="h-4 w-4" />}
   665→   665→              label="Schedule"
   666→   666→              onClick={() => { onSchedule(); onClose() }}
   667→   667→            />
   668→   668→          )}
   669→   669→          {isMine && message.type === 'text' && !message.deletedAt && (
   670→   670→            <ActionMenuItem icon={<Pencil className="h-4 w-4" />} label="Edit" onClick={() => { onEdit(); onClose() }} />
   671→   671→          )}
   672→   672→          {isMine && !message.deletedAt && (

   674→   674→              icon={<Trash2 className="h-4 w-4" />}
   675→   675→              label="Delete"
   676→   676→              destructive
   677→   677→              onClick={() => { onDelete(); onClose() }}

   680→   680→          <ActionMenuItem icon={<Pin className="h-4 w-4" />} label={message.pinnedAt ? 'Unpin' : 'Pin'} onClick={() => { onPin(); onClose() }} />
   681→   681→        </div>
   682→   682→      </div>
   683→   683→    </div>
   684→   684→  )
