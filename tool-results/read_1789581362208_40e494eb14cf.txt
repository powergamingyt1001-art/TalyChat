   173→      toast({ title: 'Failed to load stats', description: e?.message, variant: 'destructive' })

   175→      setLoading(false)
   176→    }
   177→  }, [toast])
   178→
   179→  React.useEffect(() => {
   180→    load(period)
   181→  }, [period, load])

   189→          <p className="text-sm text-muted-foreground/80">TalyChat admin overview &amp; analytics</p>
   190→        </div>
   191→        <div className="segmented-control" role="group" aria-label="Period selector">
   192→          {PERIODS.map((p) => {
   193→            const active = period === p.id
   194→            return (
   195→              <button
   196→                key={p.id}
   197→                onClick={() => setPeriod(p.id)}
   198→                aria-pressed={active}
   199→                className={cn(active ? 'active' : '')}

   201→                {p.label}
   202→              </button>
   203→            )
   204→          })}
   205→        </div>
   206→      </div>
   207→
   208→      {/* Overview metrics — V3 grouped section with light grey container + section-header */}
   209→      <section
   210→        className="animate-fade-in-up rounded-xl bg-muted/30 p-4"
   211→        style={{ animationDelay: '60ms' }}
   212→      >
   213→        <div className="section-header mb-4">
   214→          Overview metrics
   215→        </div>
   216→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   217→          {METRICS.map((m, i) => {
   218→            if (loading || !stats) {
   219→              return (
   220→                <Skeleton
   221→                  key={m.label}
   222→                  className={cn(
   223→                    'h-[110px] rounded-xl',
   224→                    m.featured && 'sm:col-span-2 sm:h-[120px]'
   225→                  )}
   226→                />
   227→              )
   228→            }
   229→            return (
   230→              <KpiCard
   231→                key={m.label}
   232→                metric={m}
   233→                value={m.value(stats)}
   234→                hint={m.hint ? m.hint(stats) : undefined}
   235→                onClick={() => setDrilldown(m)}
   236→                featured={!!m.featured}
   237→                style={{ animationDelay: `${60 + i * 30}ms` }}

   239→            )
   240→          })}

   244→      {/* V2 — User Status cards (4 cards) */}

   445→        <CardContent>
   446→          <div className="h-[300px] w-full">
   447→            {loading || !stats ? (
   448→              <Skeleton className="h-full w-full" />
   449→            ) : (
   450→              <ResponsiveContainer width="100%" height="100%">

   562→// Drilldown dialog — opens when any metric card is clicked.
   563→// Renders a mini chart for that metric + small stats summary.
   564→// ============================================================
   565→function DrilldownDialog({
   566→  metric,
   567→  stats,
   568→  period,
   569→  onClose,
   570→}: {
   571→  metric: MetricDef | UserStatusDef | SubsDef | null
   572→  stats: AdminStats | null
   573→  period: Period
   574→  onClose: () => void
   575→}) {
   576→  if (!metric) return null
   577→
   578→  // Resolve common fields
   579→  const Icon = metric.icon
   580→  const label = metric.label
   581→  const accent = metric.accent
   582→  const rawValue = metric.value(stats as AdminStats)
   583→  const valueStr = typeof rawValue === 'string' ? rawValue : rawValue.toLocaleString('en-IN')
   584→
   585→  const chart = (metric as MetricDef).chart ?? 'none'
   586→  const periodLabel =
   587→    period === '7days' ? '7 Days' : period === '1month' ? '1 Month' : 'All Time'
   588→
   589→  // V3 — small stats summary (current + period + breakdown for pie metrics)
   590→  const summary: Array<{ label: string; value: string }> = [
   591→    { label: 'Current value', value: stats ? valueStr : '—' },
   592→    { label: 'Period', value: periodLabel },
   593→  ]
   594→  if (chart === 'piePremium' && stats) {
   595→    summary.push(
   596→      { label: 'Premium', value: stats.premium.toLocaleString('en-IN') },
   597→      { label: 'Expired', value: stats.expiredPlans.toLocaleString('en-IN') },
   598→      {
   599→        label: 'Free',
   600→        value: Math.max(0, stats.totalUsers - stats.premium).toLocaleString('en-IN'),
   601→      },
   602→    )
   603→  } else if (chart === 'pieActiveInactive' && stats) {
   604→    summary.push(
   605→      { label: 'Active', value: (stats.activeVsInactive?.active ?? stats.activeInactive.active).toLocaleString('en-IN') },
   606→      { label: 'Inactive', value: (stats.activeVsInactive?.inactive ?? stats.activeInactive.inactive).toLocaleString('en-IN') },
   607→    )
   608→  } else if (chart === 'pieSubs' && stats) {
   609→    summary.push(
   610→      { label: 'Paid', value: (stats.subscriptions?.paid ?? 0).toLocaleString('en-IN') },
   611→      { label: 'Free', value: (stats.subscriptions?.free ?? 0).toLocaleString('en-IN') },
   612→      { label: 'Expired', value: (stats.subscriptions?.expired ?? 0).toLocaleString('en-IN') },
   613→    )

   615→

   618→      <DialogContent className="sm:max-w-2xl">
   619→        <DialogHeader>

   621→            <span
   622→              className="flex h-8 w-8 items-center justify-center rounded-lg"
   623→              style={{ background: `color-mix(in oklch, ${ACCENT_VAR[accent]} 12%, transparent)`, color: ACCENT_VAR[accent] }}
   624→            >
   625→              <Icon className="h-4 w-4" />
   626→            </span>
   627→            {label}
   628→          </DialogTitle>
   629→          <DialogDescription>
   630→            Tap-to-see detail · <span className="font-medium">{periodLabel}</span>
   631→          </DialogDescription>
   632→        </DialogHeader>
   633→
   634→        <div className="h-[320px] w-full">
   635→          {!stats ? (
   636→            <Skeleton className="h-full w-full" />
   637→          ) : chart === 'none' ? (
   638→            <div className="flex h-full flex-col items-center justify-center gap-2 text-muted-foreground/80">
   639→              <Icon className="h-12 w-12 opacity-30" />
   640→              <p className="text-sm">No time-series breakdown for this metric.</p>
   641→              <p className="text-xs">
   642→                Snapshot value: <span className="font-semibold text-foreground">{valueStr}</span>

   644→            </div>
   645→          ) : chart === 'salesData' ? (
   646→            <ResponsiveContainer width="100%" height="100%">
