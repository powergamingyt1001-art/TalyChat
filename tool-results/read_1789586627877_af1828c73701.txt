   108→   108→// ---------------------------------------------------------------------------
   109→   109→// Safe customizer hook (works with or without the CustomizerProvider).
   110→   110→// ---------------------------------------------------------------------------
   111→   111→
   112→   112→function useCustomizerSafe(preferences: any) {
   113→   113→  try {
   114→   114→    const ctx = useCustomizer()
   115→   115→    return ctx
   116→   116→  } catch {
   117→   117→    // CustomizerProvider not available — derive from preferences prop.
   118→   118→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   119→   119→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   120→   120→    return {
   121→   121→      preferences,
   122→   122→      wallpaper,
   123→   123→      fontFamily,
   124→   124→      messageStyle: preferences?.messageStyle || 'bubble',
   125→   125→      fontSize: preferences?.fontSize || 14,
   126→   126→      setPreference: async () => {},
   127→   127→    }
   128→   128→  }
   129→   129→}
