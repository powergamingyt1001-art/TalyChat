   176→export interface DiscoverScreenProps {
   177→  // V12 — Called when a user taps "Open" on a joined group. Wired up in
   178→  // TalyApp to open the group's chat conversation.
   179→  onOpenChat?: (c: {
   180→    conversationId: string
   181→    name: string
   182→    avatar?: string
   183→    isGroup: boolean
   184→  }) => void
   185→  // V12 — Called when the empty-state "Create {Category} Group" button is
   186→  // tapped. Wired up to navigate the user to the Groups tab so they can
   187→  // start a community in the chosen category.
   188→  onCreateGroup?: (category?: string) => void
   189→}
   190→
   191→export function DiscoverScreen({ onOpenChat, onCreateGroup }: DiscoverScreenProps = {}) {
   192→  const { toast } = useToast()
   193→  const [trending, setTrending] = useState<GroupItem[]>([])
   194→  const [popular, setPopular] = useState<GroupItem[]>([])
   195→  const [newGroups, setNewGroups] = useState<GroupItem[]>([])
   196→  const [sponsored, setSponsored] = useState<any[]>([])
   197→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   198→  const [categoryResults, setCategoryResults] = useState<GroupItem[]>([])
   199→  const [loading, setLoading] = useState(true)
   200→  const [catLoading, setCatLoading] = useState(false)
   201→  const [joining, setJoining] = useState<string | null>(null)
   202→  // V12 — Track the group currently being "left" (used to disable the Leave
   203→  // button + show a spinner during the leave API call).
   204→  const [leaving, setLeaving] = useState<string | null>(null)
   205→  // joinedIds = public joined; requestedIds = private groups with sent requests
   206→  const [joinedIds, setJoinedIds] = useState<Set<string>>(new Set())
   207→  const [requestedIds, setRequestedIds] = useState<Set<string>>(new Set())
   208→  const [preview, setPreview] = useState<GroupItem | null>(null)
   209→
   210→  useEffect(() => {
   211→    let cancelled = false
   212→    ;(async () => {
   213→      setLoading(true)
   214→      try {
   215→        const [t, p, n] = await Promise.all([
   216→          apiFetch('/api/discover?sort=trending').catch(() => null),
   217→          apiFetch('/api/discover?sort=popular').catch(() => null),
   218→          apiFetch('/api/discover?sort=new').catch(() => null),
   219→        ])
   220→        if (cancelled) return
   221→        // Defensive: API returns { groups: [...], sponsored: [...] }.
   222→        // Fall back to bare arrays if the shape is ever flat.
   223→        const tGroups: GroupItem[] = (Array.isArray(t)
   224→          ? t
   225→          : t?.groups || t?.items || []) as GroupItem[]
   226→        const pGroups: GroupItem[] = (Array.isArray(p)
   227→          ? p
   228→          : p?.groups || p?.items || []) as GroupItem[]
   229→        const nGroups: GroupItem[] = (Array.isArray(n)
   230→          ? n
   231→          : n?.groups || n?.items || []) as GroupItem[]
   232→        setTrending(tGroups)
   233→        setPopular(pGroups)
   234→        setNewGroups(nGroups)
   235→        // Sponsored ads: prefer the trending response's sponsored list,
   236→        // fall back to popular's, then to an empty array.
   237→        const sponsoredList: any[] = Array.isArray(t?.sponsored)
   238→          ? t.sponsored
   239→          : Array.isArray(p?.sponsored)
   240→            ? p.sponsored
   241→            : Array.isArray(n?.sponsored)
   242→              ? n.sponsored
   243→              : []
   244→        setSponsored(sponsoredList)
   245→        const joined = new Set<string>([
   246→          ...tGroups.filter((g) => g.isJoined).map((g) => g.id),
   247→          ...pGroups.filter((g) => g.isJoined).map((g) => g.id),
   248→          ...nGroups.filter((g) => g.isJoined).map((g) => g.id),
   249→        ])
   250→        setJoinedIds(joined)
   251→      } finally {
   252→        if (!cancelled) setLoading(false)
   253→      }
   254→    })()
   255→    return () => {
   256→      cancelled = true
   257→    }
   258→  }, [])
   259→
   260→  // Load the user's sent group join requests once on mount so we can mark
   261→  // private groups as "requested".
   262→  useEffect(() => {
   263→    ;(async () => {
   264→      try {
   265→        const res: any = await apiFetch('/api/groups/requests/sent')
   266→        const list: any[] = Array.isArray(res) ? res : (res?.requests || res?.items || [])
   267→        const pending = new Set<string>(
   268→          list
   269→            .filter((r: any) => r.status === 'pending' && r.group?.id)
   270→            .map((r: any) => r.group.id),
   271→        )
   272→        setRequestedIds(pending)
   273→      } catch {
   274→        // ignore — feature best-effort
   275→      }
   276→    })()
   277→  }, [])
   278→
   279→  const fetchCategory = async (cat: string) => {
   280→    setCatLoading(true)
   281→    try {
   282→      const res: any = await apiFetch(
   283→        `/api/discover?category=${encodeURIComponent(cat)}&sort=trending`,
   284→      )
   285→      // Defensive: API returns { groups: [...] }, fall back to bare array.
   286→      const list: GroupItem[] = (Array.isArray(res)
   287→        ? res
   288→        : res?.groups || res?.items || []) as GroupItem[]
   289→      setCategoryResults(list)
   290→    } catch (e: any) {
   291→      toast({ title: e?.message || 'Failed to load category', variant: 'destructive' })
   292→    } finally {
   293→      setCatLoading(false)
   294→    }
   295→  }
   296→
   297→  const handleCategoryClick = (cat: string) => {
   298→    if (activeCategory === cat) {
   299→      setActiveCategory(null)
   300→      setCategoryResults([])
   301→    } else {
   302→      setActiveCategory(cat)
   303→      fetchCategory(cat)
   304→    }
   305→  }
   306→
   307→  // Returns the join state for a group, factoring in joined + requested sets.
   308→  const joinStateFor = (g: GroupItem): JoinState => {
   309→    if (joinedIds.has(g.id)) return 'joined'
   310→    if (!g.isPublic && requestedIds.has(g.id)) return 'requested'
   311→    return 'idle'
   312→  }
   313→
   314→  const handleJoin = async (g: GroupItem) => {
   315→    setJoining(g.id)
   316→    try {
   317→      const res: any = await apiFetch('/api/groups/join', {
   318→        method: 'POST',
   319→        body: JSON.stringify({ groupId: g.id }),
   320→      })
   321→      if (res?.alreadyMember) {
   322→        toast({ title: 'You are already a member' })
   323→        setJoinedIds((prev) => new Set(prev).add(g.id))
   324→      } else if (res?.requested) {
   325→        // Private group → request was sent
   326→        toast({
   327→          title: 'Join request sent! Wait for admin approval.',
   328→        })
   329→        setRequestedIds((prev) => new Set(prev).add(g.id))
   330→      } else {
   331→        // Public group → direct join
   332→        toast({ title: `Joined ${g.name} 🎉` })
   333→        setJoinedIds((prev) => new Set(prev).add(g.id))
   334→      }
   335→    } catch (e: any) {
   336→      toast({ title: e?.message || 'Failed to join', variant: 'destructive' })
   337→    } finally {
   338→      setJoining(null)
   339→    }
   340→  }
   341→
   342→  // V12 — Open the chat for a joined group. Calls onOpenChat (wired from
   343→  // TalyApp) when a conversationId is linked to the group. Falls back to
   344→  // a friendly toast if the callback is missing (defensive — should never
   345→  // happen in production wiring, but avoids a no-op click).
   346→  const handleOpen = (g: GroupItem) => {
   347→    if (!g.conversationId) {
   348→      toast({
   349→        title: 'No chat linked to this group yet',
   350→        description: 'Please try again in a moment.',
   351→        variant: 'destructive',
   352→      })
   353→      return
   354→    }
   355→    if (onOpenChat) {
   356→      onOpenChat({
   357→        conversationId: g.conversationId,
   358→        name: g.name,
   359→        avatar: g.logo || undefined,
   360→        isGroup: true,
   361→      })
   362→    } else {
   363→      toast({ title: 'Open coming soon' })
   364→    }
   365→  }
   366→
   367→  // V12 — Leave a joined group. Calls POST /api/groups/[id]/leave, then
   368→  // removes the group id from joinedIds so the button reverts to Join.
   369→  const handleLeave = async (g: GroupItem) => {
   370→    setLeaving(g.id)
   371→    try {
   372→      await apiFetch(`/api/groups/${g.id}/leave`, { method: 'POST' })
   373→      toast({ title: `Left ${g.name}` })
   374→      setJoinedIds((prev) => {
   375→        const next = new Set(prev)
   376→        next.delete(g.id)
   377→        return next
   378→      })
   379→    } catch (e: any) {
   380→      toast({ title: e?.message || 'Failed to leave group', variant: 'destructive' })
   381→    } finally {
   382→      setLeaving(null)
   383→    }
   384→  }
   385→
   386→  // V12 — Reset the active category filter (used by the "Browse Trending
   387→  // instead" link in the empty state and by the active-filter banner's
   388→  // clear (✕) button).
   389→  const clearCategory = () => {
   390→    setActiveCategory(null)
   391→    setCategoryResults([])
   392→  }
   393→
   394→  // V12 — Handle the "Create {Category} Group" CTA in the no-results
   395→  // empty state. If the parent wired onCreateGroup, delegate to it
   396→  // (TalyApp navigates to the Groups tab). Otherwise show a friendly
   397→  // toast so the click is never a no-op.
   398→  const handleCreateGroup = () => {
   399→    if (onCreateGroup) {
   400→      onCreateGroup(activeCategory || undefined)
   401→    } else {
   402→      toast({
   403→        title: 'Open the Groups tab',
   404→        description: activeCategory
   405→          ? `Tap the + button there to start a ${activeCategory} community.`
   406→          : 'Tap the + button there to start a new community.',
   407→      })
   408→    }
   409→  }
   410→
   411→  const totalGroups = trending.length + popular.length + newGroups.length
   412→
   413→  return (
   414→    <div className="mx-auto max-w-2xl px-4 pb-20 pt-6 lg:pb-6 lg:pt-8">
   415→      {/* Header */}
   416→      <motion.div
   417→        initial={{ opacity: 0, y: -8 }}
   418→        animate={{ opacity: 1, y: 0 }}
   419→        transition={{ duration: 0.3 }}
   420→      >
   421→        <h1 className="section-header">Discover</h1>
   422→        <p className="mt-1 text-sm text-muted-foreground">
   423→          Find communities that match your interests
   424→        </p>
   425→      </motion.div>
   426→
   427→      {/* Category chips — solid emerald active state + horizontal scroll snap */}
   428→      <div
   429→        className="no-scrollbar scroll-pan-y -mx-4 mt-3 w-full overflow-x-auto px-4"
   430→        style={{ scrollSnapType: 'x proximity' }}
   431→      >
   432→        <div className="flex min-w-0 gap-2 pb-1">
   433→          {CATEGORIES.map((cat) => {
   434→            const active = activeCategory === cat
   435→            return (
   436→              <button
   437→                key={cat}
   438→                type="button"
   439→                onClick={() => handleCategoryClick(cat)}
   440→                className={`category-chip min-h-[36px] ${active ? 'active' : ''}`}
   441→                style={{ scrollSnapAlign: 'start' }}
   442→              >
   443→                {cat}
   444→              </button>
   445→            )
   446→          })}
   447→        </div>
   448→      </div>
   449→
   450→      {/* V12 — Active filter banner: shows below the chips row when a
   451→          category filter is selected, with a clear (✕) button to reset. */}
   452→      {activeCategory && (
   453→        <div className="mt-3 flex items-center justify-between gap-2 rounded-lg bg-emerald-50 px-3 py-2 text-xs font-medium text-emerald-700 dark:bg-emerald-950/20 dark:text-emerald-300">
   454→          <span className="truncate">
   455→            Showing results for{' '}
   456→            <span className="font-bold">&lsquo;{activeCategory}&rsquo;</span>
   457→          </span>
   458→          <button
   459→            type="button"
   460→            onClick={clearCategory}
   461→            aria-label="Clear category filter"
   462→            className="inline-flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-emerald-700 transition-colors hover:bg-emerald-500/15 dark:text-emerald-300"
   463→          >
   464→            <X className="h-3.5 w-3.5" />
   465→          </button>
   466→        </div>
   467→      )}
   468→
   469→      {/* Empty state — no groups loaded */}
   470→      {!loading && totalGroups === 0 && !activeCategory && (
   471→        <div className="dotted-bg mt-6 flex flex-col items-center justify-center rounded-2xl border border-border bg-card p-10 text-center">
   472→          <Compass className="h-12 w-12 text-muted-foreground/60" />
   473→          <p className="mt-3 text-sm font-semibold text-foreground">
   474→            No communities yet
   475→          </p>
   476→          <p className="mt-1 text-xs text-muted-foreground">
   477→            Check back soon — new communities are added every day.
   478→          </p>
   479→        </div>
   480→      )}
   481→
   482→      {/* Category results */}
   483→      {activeCategory && (
   484→        <section className="mt-4 animate-fade-in-up">
   485→          <h2 className="section-header">
   486→            {activeCategory} communities
   487→          </h2>
   488→          <div className="mt-3">
   489→            {catLoading ? (
   490→              <InlineLoadingRow />
   491→            ) : categoryResults.length === 0 ? (
   492→              // V12 — Friendly no-results empty state: magnifying glass,
   493→              // "No groups found in {Category}", "Be the first to create
   494→              // one!" + a Create button (emerald action-btn) and a "Browse
   495→              // Trending instead" text link.
   496→              <div className="flex flex-col items-center justify-center rounded-2xl border border-dashed border-border bg-card p-8 text-center">
   497→                <Search className="h-12 w-12 text-muted-foreground/40" />
   498→                <p className="mt-3 text-sm font-semibold text-foreground">
   499→                  No groups found in {activeCategory}
   500→                </p>
   501→                <p className="mt-1 text-xs text-muted-foreground">
   502→                  Be the first to create one!
   503→                </p>
   504→                <button
   505→                  type="button"
   506→                  onClick={handleCreateGroup}
   507→                  className="action-btn mt-4 min-h-[44px]"
   508→                >
   509→                  <Plus className="h-4 w-4" /> Create {activeCategory} Group
   510→                </button>
   511→                <button
   512→                  type="button"
   513→                  onClick={clearCategory}
   514→                  className="mt-3 inline-flex min-h-[40px] items-center gap-1 text-xs font-medium text-muted-foreground transition-colors hover:text-primary"
   515→                >
   516→                  Browse Trending instead <ChevronRight className="h-3 w-3" />
   517→                </button>
   518→              </div>
   519→            ) : (
   520→              <ul className="space-y-2">
   521→                {categoryResults.map((g) => (
   522→                  <li key={g.id}>
   523→                    <GroupRow
   524→                      g={g}
   525→                      state={joinStateFor(g)}
   526→                      joining={joining === g.id}
   527→                      leaving={leaving === g.id}
   528→                      onJoin={handleJoin}
   529→                      onOpen={handleOpen}
   530→                      onLeave={handleLeave}
   531→                      onPreview={() => setPreview(g)}
   532→                    />
   533→                  </li>
   534→                ))}
   535→              </ul>
   536→            )}
   537→          </div>
   538→        </section>
   539→      )}
   540→
   541→      {/* Default sections */}
   542→      {!activeCategory && (
   543→        <>
   544→          <Section
   545→            title="Trending"
   546→            icon="🔥"
   547→            loading={loading}
   548→            groups={trending}
   549→            joinStateFor={joinStateFor}
   550→            joining={joining}
   551→            leaving={leaving}
   552→            onJoin={handleJoin}
   553→            onOpen={handleOpen}
   554→            onLeave={handleLeave}
   555→            onPreview={(g) => setPreview(g)}
   556→            delay={0.05}
   557→          />
   558→          <Section
   559→            title="Popular"
   560→            icon="⭐"
   561→            loading={loading}
   562→            groups={popular}
   563→            joinStateFor={joinStateFor}
   564→            joining={joining}
   565→            leaving={leaving}
   566→            onJoin={handleJoin}
   567→            onOpen={handleOpen}
   568→            onLeave={handleLeave}
   569→            onPreview={(g) => setPreview(g)}
   570→            delay={0.1}
   571→          />
   572→          <Section
   573→            title="New"
   574→            icon="🆕"
   575→            loading={loading}
   576→            groups={newGroups}
   577→            joinStateFor={joinStateFor}
   578→            joining={joining}
   579→            leaving={leaving}
   580→            onJoin={handleJoin}

   902→// ============================================================
   903→// MiniStackedAvatars — 2-3 small overlapping letter-avatars with +N.
   904→// Used inside group cards/rows to humanize the member count display.
   905→// ============================================================
   906→function MiniStackedAvatars({
   907→  logo,
   908→  name,
   909→  memberCount,
   910→  small = false,
   911→}: {
   912→  logo?: string | null
   913→  name: string
   914→  memberCount: number
   915→  small?: boolean
   916→}) {
   917→  const baseName = name || 'G'
   918→  const initials = (baseName.replace(/\s+/g, ' ').trim().split(' ').filter(Boolean))
   919→  const derived: string[] = []
   920→  for (let i = 0; i < initials.length && derived.length < 3; i++) {
   921→    derived.push(initials[i][0]?.toUpperCase() || 'U')
   922→  }
   923→  while (derived.length < 3) {
   924→    derived.push(baseName[0]?.toUpperCase() || 'U')
   925→  }
   926→  const visible = derived.slice(0, 3)
   927→  const extra = Math.max(0, memberCount - 3)
   928→  const sizeClass = small ? 'h-4 w-4' : 'h-5 w-5'
   929→  const textSize = small ? 'text-[7px]' : 'text-[8px]'
   930→  const offset = small ? 8 : 10
   931→  const slotClass = (i: number) => {
   932→    switch (i % 3) {
   933→      case 0:
   934→        return 'bg-emerald-500/15 text-emerald-600'
   935→      case 1:
   936→        return 'bg-amber-500/15 text-amber-600'
   937→      default:
   938→        return 'bg-sky-500/15 text-sky-600'
   939→    }
   940→  }
   941→  return (
   942→    <span className="relative inline-flex items-center" aria-hidden>
   943→      <span className="relative inline-flex" style={{ minWidth: `${offset * 2 + 8}px` }}>
   944→        {visible.map((letter, i) => {
   945→          const z = visible.length - i
   946→          return (
   947→            <span
   948→              key={i}
   949→              className={`absolute flex ${sizeClass} items-center justify-center rounded-full ${textSize} font-bold ring-1 ring-card ${slotClass(i)}`}
   950→              style={{ left: `${i * offset}px`, zIndex: z }}
   951→            >
   952→              {i === 0 && logo ? (
   953→                <img
   954→                  src={logo}
   955→                  alt=""
   956→                  className="h-full w-full rounded-full object-cover"
   957→                />
   958→              ) : (
   959→                letter
   960→              )}
   961→            </span>
   962→          )
   963→        })}
   964→        {extra > 0 && (
   965→          <span
   966→            className={`absolute flex ${sizeClass} items-center justify-center rounded-full bg-muted ${textSize} font-bold text-muted-foreground ring-1 ring-card`}
   967→            style={{ left: `${visible.length * offset}px`, zIndex: visible.length + 1 }}
   968→          >
   969→            +{extra > 99 ? '99' : extra}
   970→          </span>
   971→        )}
   972→      </span>
   973→    </span>
   974→  )
   975→}
