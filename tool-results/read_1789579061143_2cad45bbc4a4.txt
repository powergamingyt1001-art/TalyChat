   101→// ---------------------------------------------------------------------------
   102→// Safe customizer hook (works with or without the CustomizerProvider).
   103→// ---------------------------------------------------------------------------
   104→
   105→function useCustomizerSafe(preferences: any) {
   106→  try {
   107→    const ctx = useCustomizer()
   108→    return ctx
   109→  } catch {
   110→    // CustomizerProvider not available — derive from preferences prop.
   111→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   112→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   113→    return {
   114→      preferences,
   115→      wallpaper,
   116→      fontFamily,
   117→      messageStyle: preferences?.messageStyle || 'bubble',
   118→      fontSize: preferences?.fontSize || 14,
   119→      setPreference: async () => {},
   120→    }
   121→  }
   122→}

   588→      }

   590→      setText('')
   591→      // Reset textarea height

   594→      }
   595→    } catch (e: any) {
   596→      toast({ title: e.message || 'Failed to send', variant: 'destructive' })
   597→    } finally {
   598→      setSending(false)
   599→    }
   600→  }
