   106→// ---------------------------------------------------------------------------
   107→// Safe customizer hook (works with or without the CustomizerProvider).
   108→// ---------------------------------------------------------------------------
   109→
   110→function useCustomizerSafe(preferences: any) {
   111→  try {
   112→    const ctx = useCustomizer()
   113→    return ctx
   114→  } catch {
   115→    // CustomizerProvider not available — derive from preferences prop.
   116→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   117→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   118→    return {
   119→      preferences,
   120→      wallpaper,
   121→      fontFamily,
   122→      messageStyle: preferences?.messageStyle || 'bubble',
   123→      fontSize: preferences?.fontSize || 14,
   124→      setPreference: async () => {},
   125→    }
   126→  }
   127→}
