   153→export function ProfileScreen() {
   154→  const { user, updateUser, logout, setAuth } = useAuth()
   155→  const { toast } = useToast()
   156→  const { play: soundManager } = useSound()
   157→  const [loading, setLoading] = useState(true)
   158→  const [profile, setProfile] = useState<any>(null)
   159→  const [editOpen, setEditOpen] = useState(false)
   160→  const [settingsOpen, setSettingsOpen] = useState(false)
   161→  const [buyPlan, setBuyPlan] = useState<null | { id: string; label: string; price: number; months: number }>(null)
   162→  const [redeemCode, setRedeemCode] = useState('')
   163→  const [redeemLoading, setRedeemLoading] = useState(false)
   164→  const [referral, setReferral] = useState<ReferralData | null>(null)
   165→
   166→  // V2 — admin login via Redeem section.
   167→  // When the user types "admin.in" (case-insensitive) as the redeem code
   168→  // and taps Redeem, we open an admin-login password dialog instead of
   169→  // calling the normal /api/redeem endpoint. On success we setAuth() and
   170→  // the app re-renders as the admin panel (see src/app/page.tsx).
   171→  const [adminLoginOpen, setAdminLoginOpen] = useState(false)
   172→  const [adminPassword, setAdminPassword] = useState('')
   173→  const [adminLoginLoading, setAdminLoginLoading] = useState(false)
   174→
   175→  // Block list state
   176→  const [blockedOpen, setBlockedOpen] = useState(false)
   177→  const [blockedCount, setBlockedCount] = useState<number>(0)
   178→
   179→  // Behavior + watch-ad state
   180→  const [behavior, setBehavior] = useState<BehaviorData | null>(null)
   181→  const [watchingAd, setWatchingAd] = useState<AdData | null>(null)
   182→  const [watching, setWatching] = useState(false)
   183→  const [watchCountdown, setWatchCountdown] = useState(5)
   184→
   185→  // Team sheet state
   186→  const [teamSheetOpen, setTeamSheetOpen] = useState(false)
   187→
   188→  const loadProfile = useCallback(async () => {
   189→    setLoading(true)
   190→    try {
   191→      const res: any = await apiFetch('/api/users/me')
   192→      // Defensive: API returns { user: {...}, preferences: {...} }.
   193→      // Fall back to bare object if shape changes.
   194→      const u = res?.user || (res?.id ? res : null)
   195→      setProfile(u)
   196→      if (u) updateUser(u)
   197→    } catch (e: any) {
   198→      toast({ title: e.message, variant: 'destructive' })
   199→    } finally {
   200→      setLoading(false)
   201→    }
   202→  }, [toast, updateUser])
   203→
   204→  const loadReferral = useCallback(async () => {
   205→    try {
   206→      const res: any = await apiFetch('/api/referral/me')
   207→      // Defensive: API returns flat { code, count, recent, tierReached, progress }.
   208→      // Some backends might wrap as { referral: {...} } — handle both.
   209→      const data = res?.referral || res?.data || res
   210→      setReferral(data as ReferralData)
   211→    } catch {
   212→      setReferral(null)
   213→    }
   214→  }, [])
   215→
   216→  // Fetch behavior summary (score, adsWatchedToday, maxAdsPerDay, canMessage)
   217→  const loadBehavior = useCallback(async () => {
   218→    try {
   219→      const res: any = await apiFetch('/api/behavior/me')
   220→      const data = res?.behavior || res?.data || res
   221→      setBehavior({
   222→        score: typeof data?.score === 'number' ? data.score : 100,
   223→        adsWatchedToday: data?.adsWatchedToday ?? 0,
   224→        maxAdsPerDay: data?.maxAdsPerDay ?? 10,
   225→        canMessage: Boolean(data?.canMessage),
   226→      })
   227→    } catch {
   228→      // Defensive default — assume full score if route fails
   229→      setBehavior({
   230→        score: 100,
   231→        adsWatchedToday: 0,
   232→        maxAdsPerDay: 10,
   233→        canMessage: true,
   234→      })
   235→    }
   236→  }, [])
   237→
   238→  // Watch an ad: POST /api/behavior/watch-ad, then open dialog for 5s.
   239→  const handleWatchAd = useCallback(async () => {
   240→    if (watching) return
   241→    if (behavior && behavior.adsWatchedToday >= behavior.maxAdsPerDay) {
   242→      toast({ title: 'Daily limit reached — come back tomorrow!' })
   243→      return
   244→    }
   245→    setWatching(true)
   246→    try {
   247→      const res: any = await apiFetch('/api/behavior/watch-ad', { method: 'POST' })
   248→      const data = res?.behavior || res?.data || res
   249→      const ad: AdData | null = data?.ad ?? null
   250→      // Update behavior immediately with new score/adsWatchedToday.
   251→      setBehavior({
   252→        score: typeof data?.score === 'number' ? data.score : (behavior?.score ?? 100),
   253→        adsWatchedToday: data?.adsWatchedToday ?? (behavior?.adsWatchedToday ?? 0) + 1,
   254→        maxAdsPerDay: data?.maxAdsPerDay ?? behavior?.maxAdsPerDay ?? 10,
   255→        canMessage: Boolean(data?.canMessage ?? (behavior?.canMessage ?? true)),
   256→      })
   257→      if (ad) {
   258→        setWatchingAd(ad)
   259→        setWatchCountdown(5)
   260→      } else {
   261→        toast({ title: 'Behavior +1! No ad creative available right now.' })
   262→      }
   263→    } catch (e: any) {
   264→      toast({ title: e?.message || 'Failed to watch ad', variant: 'destructive' })
   265→    } finally {
   266→      setWatching(false)
   267→    }
   268→  }, [behavior, watching, toast])
   269→
   270→  // Select a referral task tier.
   271→  const handleSelectTask = useCallback(
   272→    async (tier: string) => {
   273→      try {
   274→        await apiFetch('/api/referral/task', {
   275→          method: 'POST',
   276→          body: JSON.stringify({ tier }),
   277→        })
   278→        toast({ title: 'Task selected! Get referring 🚀' })
   279→        await loadReferral()
   280→      } catch (e: any) {
   281→        toast({ title: e?.message || 'Failed to select task', variant: 'destructive' })
   282→      }
   283→    },
   284→    [loadReferral, toast],
   285→  )
   286→
   287→  // Claim the active referral task reward.
   288→  const handleClaimTask = useCallback(async () => {
   289→    try {
   290→      const res: any = await apiFetch('/api/referral/task/claim', { method: 'POST' })
   291→      const data = res?.claim || res?.data || res
   292→      const until = data?.premiumUntil
   293→      toast({
   294→        title: 'Premium granted! 🎉',
   295→        description: until
   296→          ? `Valid until ${format(new Date(until), 'dd MMM yyyy')}`
   297→          : undefined,
   298→      })
   299→      await loadReferral()
   300→      await loadProfile()
   301→    } catch (e: any) {
   302→      toast({ title: e?.message || 'Failed to claim reward', variant: 'destructive' })
   303→    }
   304→  }, [loadProfile, loadReferral, toast])
   305→
   306→
   307→  // Fetch the blocked-users count (for the badge on Privacy section).
   308→  const loadBlockedCount = useCallback(async () => {
   309→    try {
   310→      const res: any = await apiFetch('/api/blocks')
   311→      const list: any[] = Array.isArray(res)
   312→        ? res
   313→        : Array.isArray(res?.blocks)
   314→          ? res.blocks
   315→          : Array.isArray(res?.items)
   316→            ? res.items
   317→            : []
   318→      setBlockedCount(list.length)
   319→    } catch {
   320→      setBlockedCount(0)
   321→    }
   322→  }, [])
   323→
   324→  useEffect(() => {
   325→     
   326→    loadProfile()
   327→     
   328→    loadReferral()
   329→    loadBlockedCount()
   330→    loadBehavior()
   331→  }, [loadProfile, loadReferral, loadBlockedCount, loadBehavior])
   332→
   333→  if (loading && !profile) {
   334→    return (
   335→      <div className="flex min-h-[60dvh] items-center justify-center">
   336→        <Loader2 className="h-8 w-8 animate-spin text-primary" />
   337→      </div>
   338→    )
   339→  }
   340→
   341→  if (!profile) {
   342→    return (
   343→      <div className="mx-auto max-w-2xl p-4 text-center text-muted-foreground">
   344→        Could not load profile. Please try again.
   345→      </div>
   346→    )
   347→  }
   348→
   349→  const premiumActive =
   350→    profile.isPremium &&
   351→    profile.premiumUntil &&
   352→    new Date(profile.premiumUntil) > new Date()
   353→
   354→  const copyReferralLink = async () => {
   355→    const link = `${APP_DOMAIN}/?ref=${user?.username || profile.username}`
   356→    try {
   357→      await navigator.clipboard.writeText(link)
   358→      toast({ title: 'Referral link copied!' })
   359→    } catch {
   360→      toast({ title: 'Copy failed', variant: 'destructive' })
   361→    }
   362→  }
   363→
   364→  const handleRedeem = async () => {
   365→    if (!redeemCode.trim()) {
   366→      toast({ title: 'Enter a code first', variant: 'destructive' })
   367→      return
   368→    }
   369→    // V2 — admin.in special case: open the admin login dialog instead of
   370→    // calling /api/redeem. The case-insensitive match mirrors the backend
   371→    // /api/auth/login behavior (which lower-cases the email).
   372→    if (redeemCode.trim().toLowerCase() === 'admin.in') {
   373→      setAdminPassword('')
   374→      setAdminLoginOpen(true)
   375→      return
   376→    }
   377→    setRedeemLoading(true)
   378→    try {
   379→      const res: any = await apiFetch('/api/redeem', {
   380→        method: 'POST',
   381→        body: JSON.stringify({ code: redeemCode.trim() }),
   382→      })
   383→      // Defensive: API returns flat { ok, premiumUntil }, but may also be
   384→      // wrapped as { redeem: {...} } or { data: {...} } in some envs.
   385→      const data: any = res?.redeem || res?.data || res
   386→      const premiumUntil = data?.premiumUntil
   387→      toast({
   388→        title: 'Premium activated!',
   389→        description: premiumUntil
   390→          ? `Valid until ${format(new Date(premiumUntil), 'dd MMM yyyy')}`
   391→          : 'Your premium has been extended.',
   392→      })
   393→      // Fanfare — fires inside the user's "Redeem" button click gesture.
   394→      soundManager.playPremium()
   395→      setRedeemCode('')
   396→      loadProfile()
   397→    } catch (e: any) {
   398→      toast({ title: e.message, variant: 'destructive' })
   399→    } finally {
   400→      setRedeemLoading(false)
   401→    }
   402→  }
   403→
   404→  // V2 — submit admin login (called from the Admin Login dialog).
   405→  // On success, setAuth() with the returned user/token; the app will then
   406→  // re-render as the admin panel.
   407→  const submitAdminLogin = async () => {
   408→    if (!adminPassword) {
   409→      toast({ title: 'Enter the admin password', variant: 'destructive' })
   410→      return
   411→    }
   412→    setAdminLoginLoading(true)
   413→    try {
   414→      const res: any = await apiFetch('/api/auth/login', {
   415→        method: 'POST',
   416→        body: JSON.stringify({ email: 'admin.in', password: adminPassword }),
   417→      })
   418→      const u = res?.user || (res?.id ? res : null)
   419→      const t = res?.token
   420→      if (!u || !t) {
   421→        throw new Error('Invalid admin response')
   422→      }
   423→      setAuth(u, t)
   424→      toast({
   425→        title: 'Welcome, Admin!',
   426→        description: 'Switching to the admin panel…',
   427→      })
   428→      setAdminLoginOpen(false)
   429→      setAdminPassword('')
   430→      setRedeemCode('')
   431→    } catch (e: any) {
   432→      toast({ title: e?.message || 'Admin login failed', variant: 'destructive' })
   433→    } finally {
   434→      setAdminLoginLoading(false)
   435→    }
   436→  }
   437→
   438→  return (
   439→    <div className="mx-auto max-w-2xl space-y-5 p-4">
   440→      {/* Profile header card — V3: taly-card with shadow, larger avatar, online dot, joined date */}
   441→      <div className="taly-card taly-card-hover animate-fade-in-up p-5" style={{ animationDelay: '0ms' }}>
   442→        <div className="flex flex-col items-center gap-4 sm:flex-row sm:items-start">
   443→          <div className="relative shrink-0">
   444→            <PremiumAvatar
   445→              user={{
   446→                isPremium: profile.isPremium,
   447→                premiumTier: profile.premiumTier,
   448→                avatar: profile.avatar || undefined,
   449→                name: profile.name || 'U',
   450→              }}
   451→              size={80}
   452→              showAura
   453→              className="shrink-0"
   454→            />
   455→            {profile.isOnline && <span className="online-dot" aria-label="online" />}
   456→          </div>
   457→          <div className="min-w-0 flex-1 text-center sm:text-left">
   458→            <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   459→              <h2 className="truncate text-xl font-bold tracking-tight">{profile.name}</h2>
   460→              {premiumActive && (
   461→                <span className="premium-badge">
   462→                  <Crown className="h-3 w-3" /> Premium
   463→                </span>
   464→              )}
   465→            </div>
   466→            <p className="truncate text-sm text-muted-foreground/80">@{profile.username}</p>
   467→            {profile.bio ? (
   468→              <p className="mt-2 line-clamp-3 text-sm italic text-muted-foreground">{profile.bio}</p>
   469→            ) : (
   470→              <p className="mt-2 text-sm italic text-muted-foreground">No bio yet.</p>
   471→            )}
   472→            <div className="mt-2 flex flex-wrap items-center justify-center gap-3 text-xs text-muted-foreground/80 sm:justify-start">
   473→              <span className="inline-flex items-center gap-1">
   474→                <span
   475→                  className={`inline-block h-2 w-2 rounded-full ${
   476→                    profile.isOnline ? 'bg-emerald-500' : 'bg-muted-foreground/50'
   477→                  }`}
   478→                />
   479→                {profile.isOnline ? 'Online' : 'Offline'}
   480→              </span>
   481→              {profile.createdAt && (
   482→                <span className="inline-flex items-center gap-1">
   483→                  <CalendarDays className="h-3 w-3" />
   484→                  Joined {format(new Date(profile.createdAt), 'MMM yyyy')}
   485→                </span>
   486→              )}
   487→            </div>
   488→            <div className="mt-4">
   489→              <button
   490→                onClick={() => setEditOpen(true)}
   491→                className="action-btn min-h-[40px]"
   492→              >
   493→                <Pencil className="h-4 w-4" /> Edit Profile
   494→              </button>
   495→            </div>
   496→          </div>
   497→        </div>
   498→      </div>
   499→
   500→      {/* Behavior Bar — V3: section-header + behavior-bar pill with inner glow */}
   501→      <div className="animate-fade-in-up" style={{ animationDelay: '60ms' }}>
   502→        <BehaviorBar behavior={behavior} />
   503→      </div>
   504→
   505→      {/* Watch Behavior (watch ads to increase score) */}
   506→      <div className="animate-fade-in-up" style={{ animationDelay: '120ms' }}>
   507→        <WatchBehaviorCard
   508→          behavior={behavior}
   509→          watching={watching}
   510→          onWatchAd={handleWatchAd}
   511→        />
   512→      </div>
   513→
   514→      {/* Premium section */}
   515→      <div className="animate-fade-in-up" style={{ animationDelay: '180ms' }}>
   516→        <PremiumSection
   517→          premiumActive={premiumActive}
   518→          premiumUntil={profile.premiumUntil}
   519→          onChoosePlan={(plan) => setBuyPlan(plan)}
   520→        />
   521→      </div>
   522→
   523→      {/* Redeem code */}
   524→      <div className="taly-card animate-fade-in-up p-5" style={{ animationDelay: '240ms' }}>
   525→        <div className="section-header mb-3">
   526→          <Gift className="h-4 w-4 text-primary" /> Redeem a Code
   527→        </div>
   528→        <div className="flex gap-2">
   529→          <Input
   530→            value={redeemCode}
   531→            onChange={(e) => setRedeemCode(e.target.value)}
   532→            placeholder="Enter redeem code (e.g., TALY-XXXXXXXX)"
   533→            className="min-h-[44px] flex-1 uppercase"
   534→            onKeyDown={(e) => {
   535→              if (e.key === 'Enter') handleRedeem()
   536→            }}
   537→          />
   538→          <button
   539→            onClick={handleRedeem}
   540→            disabled={redeemLoading}
   541→            className="action-btn min-h-[44px] px-5 disabled:opacity-60"
   542→          >
   543→            {redeemLoading ? (
   544→              <Loader2 className="h-4 w-4 animate-spin" />
   545→            ) : (
   546→              'Redeem'
   547→            )}
   548→          </button>
   549→        </div>
   550→        <p className="mt-2 text-xs text-muted-foreground/80">
   551→          Get codes from promotions, friends, or the founder.
   552→        </p>
   553→      </div>
   554→
   555→      {/* V2 — Admin Login dialog (opened when user types 'admin.in' as the
   556→          redeem code). On success, setAuth() switches the app to the
   557→          admin panel. */}
   558→      <Dialog open={adminLoginOpen} onOpenChange={(o) => !adminLoginLoading && setAdminLoginOpen(o)}>
   559→        <DialogContent className="sm:max-w-md">
   560→          <DialogHeader>
   561→            <DialogTitle className="flex items-center gap-2">
   562→              <ShieldAlert className="h-5 w-5 text-primary" /> Admin Login
   563→            </DialogTitle>
   564→            <DialogDescription>
   565→              Enter the admin password to switch to the TalyChat admin panel.
   566→            </DialogDescription>
   567→          </DialogHeader>
   568→          <div className="flex flex-col gap-3">
   569→            <Label htmlFor="admin-password" className="text-xs font-medium text-muted-foreground">
   570→              Admin password
   571→            </Label>
   572→            <Input
   573→              id="admin-password"
   574→              type="password"
   575→              value={adminPassword}
   576→              onChange={(e) => setAdminPassword(e.target.value)}
   577→              placeholder="Admin password"
   578→              className="min-h-[44px]"
   579→              autoFocus
   580→              onKeyDown={(e) => {
   581→                if (e.key === 'Enter') submitAdminLogin()

   859→function formatCountdown(ms: number): string {
   860→  const totalSec = Math.max(0, Math.floor(ms / 1000))
   861→  const m = Math.floor(totalSec / 60)
   862→  const s = totalSec % 60
   863→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   864→}
