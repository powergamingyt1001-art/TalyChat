   243→   243→// ============================================================
   244→   244→// R8-11 — CapsuleCategorySlider
   245→   245→// Horizontal scrolling capsule / pill chips for the Discover category
   246→   246→// filter. Each capsule is a rounded-full button with scroll-snap-align
   247→   247→// start so the slider snaps one capsule at a time. The active capsule
   248→   248→// uses the brand emerald gradient + white text; inactive capsules are
   249→   249→// subtle tinted pills. The whole row is horizontally scrollable with
   250→   250→// hidden scrollbar (.no-scrollbar) so it stays clean on mobile.
   251→   251→// ============================================================
   252→   252→function CapsuleCategorySlider({
   253→   253→  categories,
   254→   254→  activeCategory,
   255→   255→  onSelect,
   256→   256→}: {
   257→   257→  categories: string[]
   258→   258→  activeCategory: string | null
   259→   259→  onSelect: (cat: string) => void
   260→   260→}) {
   261→   261→  return (
   262→   262→    <div
   263→   263→      className="no-scrollbar scroll-pan-y mt-3 w-full overflow-x-auto pb-1"
   264→   264→      style={{
   265→   265→        scrollSnapType: 'x mandatory',
   266→   266→        scrollPaddingLeft: '8px',
   267→   267→        scrollPaddingRight: '8px',
   268→   268→      }}
   269→   269→    >
   270→   270→      <div className="flex min-w-0 gap-1.5 pb-1">
   271→   271→        {categories.map((cat) => {
   272→   272→          const active = activeCategory === cat
   273→   273→          return (
   274→   274→            <button
   275→   275→              key={cat}
   276→   276→              type="button"
   277→   277→              onClick={() => onSelect(cat)}
   278→   278→              aria-pressed={active}
   279→   279→              className={
   280→   280→                'min-h-[36px] shrink-0 snap-start rounded-full border px-3 py-1.5 text-xs font-medium transition-colors ' +
   281→   281→                (active
   282→   282→                  ? 'btn-brand border-transparent text-primary-foreground shadow-sm'
   283→   283→                  : 'border-border bg-card text-foreground/80 hover:bg-accent hover:text-foreground')
   284→   284→              }
   285→   285→            >
   286→   286→              {cat}
   287→   287→            </button>
   288→   288→          )
   289→   289→        })}
   290→   290→      </div>
   291→   291→    </div>
   292→   292→  )
   293→   293→}

   295→   295→export function DiscoverScreen({ onOpenChat }: DiscoverScreenProps = {}) {
   296→   296→  const { toast } = useToast()
   297→   297→  const [trending, setTrending] = useState<GroupItem[]>([])
   298→   298→  const [popular, setPopular] = useState<GroupItem[]>([])
   299→   299→  const [newGroups, setNewGroups] = useState<GroupItem[]>([])
   300→   300→  const [sponsored, setSponsored] = useState<any[]>([])
   301→   301→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   302→   302→  const [categoryResults, setCategoryResults] = useState<GroupItem[]>([])
   303→   303→  const [loading, setLoading] = useState(true)
   304→   304→  const [catLoading, setCatLoading] = useState(false)
   305→   305→  const [joining, setJoining] = useState<string | null>(null)
   306→   306→  // V12 — Track the group currently being "left" (used to disable the Leave
   307→   307→  // button + show a spinner during the leave API call).
   308→   308→  const [leaving, setLeaving] = useState<string | null>(null)
   309→   309→  // joinedIds = public joined; requestedIds = private groups with sent requests
   310→   310→  const [joinedIds, setJoinedIds] = useState<Set<string>>(new Set())
   311→   311→  const [requestedIds, setRequestedIds] = useState<Set<string>>(new Set())
   312→   312→  const [preview, setPreview] = useState<GroupItem | null>(null)
   313→   313→
   314→   314→  useEffect(() => {
   315→   315→    let cancelled = false
   316→   316→    ;(async () => {
   317→   317→      setLoading(true)
   318→   318→      try {
   319→   319→        const [t, p, n] = await Promise.all([
   320→   320→          apiFetch('/api/discover?sort=trending').catch(() => null),
   321→   321→          apiFetch('/api/discover?sort=popular').catch(() => null),
   322→   322→          apiFetch('/api/discover?sort=new').catch(() => null),
   323→   323→        ])
   324→   324→        if (cancelled) return
   325→   325→        // Defensive: API returns { groups: [...], sponsored: [...] }.
   326→   326→        // Fall back to bare arrays if the shape is ever flat.
   327→   327→        const tGroups: GroupItem[] = (Array.isArray(t)
   328→   328→          ? t
   329→   329→          : t?.groups || t?.items || []) as GroupItem[]
   330→   330→        const pGroups: GroupItem[] = (Array.isArray(p)
   331→   331→          ? p
   332→   332→          : p?.groups || p?.items || []) as GroupItem[]
   333→   333→        const nGroups: GroupItem[] = (Array.isArray(n)
   334→   334→          ? n
   335→   335→          : n?.groups || n?.items || []) as GroupItem[]
   336→   336→        setTrending(tGroups)
   337→   337→        setPopular(pGroups)
   338→   338→        setNewGroups(nGroups)
   339→   339→        // Sponsored ads: prefer the trending response's sponsored list,
   340→   340→        // fall back to popular's, then to an empty array.
   341→   341→        const sponsoredList: any[] = Array.isArray(t?.sponsored)
   342→   342→          ? t.sponsored
   343→   343→          : Array.isArray(p?.sponsored)
   344→   344→            ? p.sponsored
   345→   345→            : Array.isArray(n?.sponsored)
   346→   346→              ? n.sponsored
   347→   347→              : []
   348→   348→        setSponsored(sponsoredList)
   349→   349→        const joined = new Set<string>([
   350→   350→          ...tGroups.filter((g) => g.isJoined).map((g) => g.id),
   351→   351→          ...pGroups.filter((g) => g.isJoined).map((g) => g.id),
   352→   352→          ...nGroups.filter((g) => g.isJoined).map((g) => g.id),
   353→   353→        ])
   354→   354→        setJoinedIds(joined)
   355→   355→      } finally {
   356→   356→        if (!cancelled) setLoading(false)
   357→   357→      }
   358→   358→    })()
   359→   359→    return () => {
   360→   360→      cancelled = true
   361→   361→    }
   362→   362→  }, [])
   363→   363→
   364→   364→  // Load the user's sent group join requests once on mount so we can mark
   365→   365→  // private groups as "requested".
   366→   366→  useEffect(() => {
   367→   367→    ;(async () => {
   368→   368→      try {
   369→   369→        const res: any = await apiFetch('/api/groups/requests/sent')
   370→   370→        const list: any[] = Array.isArray(res) ? res : (res?.requests || res?.items || [])
   371→   371→        const pending = new Set<string>(
   372→   372→          list
   373→   373→            .filter((r: any) => r.status === 'pending' && r.group?.id)
   374→   374→            .map((r: any) => r.group.id),
   375→   375→        )
   376→   376→        setRequestedIds(pending)
   377→   377→      } catch {
   378→   378→        // ignore — feature best-effort
   379→   379→      }
   380→   380→    })()
   381→   381→  }, [])
   382→   382→
   383→   383→  const fetchCategory = async (cat: string) => {
   384→   384→    setCatLoading(true)
   385→   385→    try {
   386→   386→      const res: any = await apiFetch(
   387→   387→        `/api/discover?category=${encodeURIComponent(cat)}&sort=trending`,
   388→   388→      )
   389→   389→      // Defensive: API returns { groups: [...] }, fall back to bare array.
   390→   390→      const list: GroupItem[] = (Array.isArray(res)
   391→   391→        ? res
   392→   392→        : res?.groups || res?.items || []) as GroupItem[]
   393→   393→      setCategoryResults(list)
   394→   394→    } catch (e: any) {
   395→   395→      toast({ title: e?.message || 'Failed to load category', variant: 'destructive' })
   396→   396→    } finally {
   397→   397→      setCatLoading(false)
   398→   398→    }
   399→   399→  }
   400→   400→
   401→   401→  const handleCategoryClick = (cat: string) => {
   402→   402→    if (activeCategory === cat) {
   403→   403→      setActiveCategory(null)
   404→   404→      setCategoryResults([])
   405→   405→    } else {
   406→   406→      setActiveCategory(cat)
   407→   407→      fetchCategory(cat)
   408→   408→    }
   409→   409→  }
   410→   410→
   411→   411→  // Returns the join state for a group, factoring in joined + requested sets.
   412→   412→  const joinStateFor = (g: GroupItem): JoinState => {
   413→   413→    if (joinedIds.has(g.id)) return 'joined'
   414→   414→    if (!g.isPublic && requestedIds.has(g.id)) return 'requested'
   415→   415→    return 'idle'
   416→   416→  }
   417→   417→
   418→   418→  const handleJoin = async (g: GroupItem) => {
   419→   419→    setJoining(g.id)
   420→   420→    try {
   421→   421→      const res: any = await apiFetch('/api/groups/join', {
   422→   422→        method: 'POST',
   423→   423→        body: JSON.stringify({ groupId: g.id }),
   424→   424→      })
   425→   425→      if (res?.alreadyMember) {
   426→   426→        toast({ title: 'You are already a member' })
   427→   427→        setJoinedIds((prev) => new Set(prev).add(g.id))
   428→   428→      } else if (res?.requested) {
   429→   429→        // Private group → request was sent
   430→   430→        toast({
   431→   431→          title: 'Join request sent! Wait for admin approval.',
   432→   432→        })
   433→   433→        setRequestedIds((prev) => new Set(prev).add(g.id))
   434→   434→      } else {
   435→   435→        // Public group → direct join
   436→   436→        toast({ title: `Joined ${g.name} 🎉` })
   437→   437→        setJoinedIds((prev) => new Set(prev).add(g.id))
   438→   438→      }
   439→   439→    } catch (e: any) {
   440→   440→      toast({ title: e?.message || 'Failed to join', variant: 'destructive' })
   441→   441→    } finally {
   442→   442→      setJoining(null)
   443→   443→    }
   444→   444→  }
   445→   445→
   446→   446→  // V12 — Open the chat for a joined group. Calls onOpenChat (wired from
   447→   447→  // TalyApp) when a conversationId is linked to the group. Falls back to
   448→   448→  // a friendly toast if the callback is missing (defensive — should never
   449→   449→  // happen in production wiring, but avoids a no-op click).
   450→   450→  const handleOpen = (g: GroupItem) => {
   451→   451→    if (!g.conversationId) {
   452→   452→      toast({
   453→   453→        title: 'No chat linked to this group yet',
   454→   454→        description: 'Please try again in a moment.',
   455→   455→        variant: 'destructive',
   456→   456→      })
   457→   457→      return
   458→   458→    }
   459→   459→    if (onOpenChat) {
   460→   460→      onOpenChat({
   461→   461→        conversationId: g.conversationId,
   462→   462→        name: g.name,
   463→   463→        avatar: g.logo || undefined,
   464→   464→        isGroup: true,
   465→   465→      })
   466→   466→    } else {
   467→   467→      toast({ title: 'Open coming soon' })
   468→   468→    }
   469→   469→  }
   470→   470→
   471→   471→  // V12 — Leave a joined group. Calls POST /api/groups/[id]/leave, then
   472→   472→  // removes the group id from joinedIds so the button reverts to Join.
   473→   473→  const handleLeave = async (g: GroupItem) => {
   474→   474→    setLeaving(g.id)
   475→   475→    try {
   476→   476→      await apiFetch(`/api/groups/${g.id}/leave`, { method: 'POST' })
   477→   477→      toast({ title: `Left ${g.name}` })
   478→   478→      setJoinedIds((prev) => {
   479→   479→        const next = new Set(prev)
   480→   480→        next.delete(g.id)
   481→   481→        return next
   482→   482→      })
   483→   483→    } catch (e: any) {
   484→   484→      toast({ title: e?.message || 'Failed to leave group', variant: 'destructive' })
   485→   485→    } finally {
   486→   486→      setLeaving(null)
   487→   487→    }
   488→   488→  }

  1017→  1017→    derived.push(initials[i][0]?.toUpperCase() || 'U')

  1040→  1040→        {visible.map((letter, i) => {
  1041→  1041→          const z = visible.length - i
  1042→  1042→          return (
  1043→  1043→            <span

  1047→  1047→            >
  1048→  1048→              {i === 0 && logo ? (
  1049→  1049→                <img
  1050→  1050→                  src={logo}
  1051→  1051→                  alt=""
  1052→  1052→                  className="h-full w-full rounded-full object-cover"
  1053→  1053→                />
  1054→  1054→              ) : (
  1055→  1055→                letter
  1056→  1056→              )}
  1057→  1057→            </span>
  1058→  1058→          )
  1059→  1059→        })}
  1060→  1060→        {extra > 0 && (
  1061→  1061→          <span
  1062→  1062→            className={`absolute flex ${sizeClass} items-center justify-center rounded-full bg-muted ${textSize} font-bold text-muted-foreground ring-1 ring-card`}

  1069→  1069→    </span>
  1070→  1070→  )
