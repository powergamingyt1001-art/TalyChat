   111→// ---------------------------------------------------------------------------
   112→// Safe customizer hook (works with or without the CustomizerProvider).
   113→// ---------------------------------------------------------------------------
   114→
   115→function useCustomizerSafe(preferences: any) {
   116→  try {
   117→    const ctx = useCustomizer()
   118→    return ctx
   119→  } catch {
   120→    // CustomizerProvider not available — derive from preferences prop.
   121→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   122→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   123→    return {
   124→      preferences,
   125→      wallpaper,
   126→      fontFamily,
   127→      messageStyle: preferences?.messageStyle || 'bubble',
   128→      fontSize: preferences?.fontSize || 14,
   129→      setPreference: async () => {},
   130→    }
   131→  }
   132→}
