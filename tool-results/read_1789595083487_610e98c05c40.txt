   863→

   865→    const compute = () => {
   866→      const now = Date.now()

   868→      if (!start || now - start > OFFER_LOOP_MS) {
   869→        // restart offer cycle
   870→        start = now
   871→        localStorage.setItem(OFFER_KEY, String(start))
   872→      }
   873→      const elapsed = now - start
   874→      if (elapsed < OFFER_WINDOW_MS) {
   875→        setOfferMsLeft(OFFER_WINDOW_MS - elapsed)
   876→        setOfferPrice(OFFER_PRICE)
   877→      } else {
   878→        setOfferMsLeft(0)
   879→        setOfferPrice(REGULAR_PRICE)
   880→      }
   881→    }
   882→    compute()
   883→    timerRef.current = setInterval(compute, 1000)
   884→    return () => {
   885→      if (timerRef.current) clearInterval(timerRef.current)
   886→    }
   887→  }, [])

   907→            </p>
   908→          </div>
   909→        </div>
   910→      </div>
   911→    )
   912→  }
   913→

   915→    { id: '2mo', label: '2 Months', price: 49, months: 2, best: false },

   918→  ]
   919→
   920→  return (
   921→    <div className="taly-card animate-fade-in-up p-5">
   922→      <div className="mb-3 flex items-center justify-between gap-2">

   939→          Offer ended — 1-year plan is back to ₹{REGULAR_PRICE}

   941→      )}
   942→      <div className="grid gap-3 sm:grid-cols-3 sm:items-center">
   943→        {plans.map((p) => {
   944→          // V8 — Best Value elevation: scale-105, emerald bg tint, larger

   947→          const isBest = p.best
   948→          return (

   950→              key={p.id}
   951→              className={`taly-card relative flex flex-col gap-1 overflow-hidden p-4 transition-all ${
   952→                isBest

   954→                  : 'taly-card-hover opacity-95 scale-[0.98] border-border bg-card'

   957→              {/* Best value ribbon + shimmer overlay (larger + crown icon) */}
   958→              {isBest && (
   959→                <>
   960→                  <div className="premium-shimmer pointer-events-none absolute inset-0 opacity-60" />
   961→                  <span className="absolute right-0 top-0 inline-flex items-center gap-1 rounded-bl-xl bg-gradient-to-r from-amber-400 to-amber-500 px-2.5 py-1.5 text-[11px] font-bold uppercase tracking-wide text-white shadow-md">

   999→            </div>

  1002→      </div>
  1003→    </div>
  1004→  )
  1005→}

  1007→function formatCountdown(ms: number): string {
  1008→  const totalSec = Math.max(0, Math.floor(ms / 1000))
  1009→  const m = Math.floor(totalSec / 60)
  1010→  const s = totalSec % 60
  1011→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  1012→}
