   428→function PremiumSection({
   429→  premiumActive,
   430→  premiumUntil,
   431→  onChoosePlan,
   432→}: {
   433→  premiumActive: boolean
   434→  premiumUntil: string | null
   435→  onChoosePlan: (plan: { id: string; label: string; price: number; months: number }) => void
   436→}) {
   437→  const [offerMsLeft, setOfferMsLeft] = useState<number | null>(null)
   438→  const [offerPrice, setOfferPrice] = useState(OFFER_PRICE)
   439→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   440→
   441→  useEffect(() => {
   442→    const compute = () => {
   443→      const now = Date.now()
   444→      let start = Number(localStorage.getItem(OFFER_KEY) || 0)
   445→      if (!start || now - start > OFFER_LOOP_MS) {
   446→        // restart offer cycle
   447→        start = now
   448→        localStorage.setItem(OFFER_KEY, String(start))
   449→      }
   450→      const elapsed = now - start
   451→      if (elapsed < OFFER_WINDOW_MS) {
   452→        setOfferMsLeft(OFFER_WINDOW_MS - elapsed)
   453→        setOfferPrice(OFFER_PRICE)
   454→      } else {
   455→        setOfferMsLeft(0)
   456→        setOfferPrice(REGULAR_PRICE)
   457→      }
   458→    }
   459→    compute()
   460→    timerRef.current = setInterval(compute, 1000)
   461→    return () => {
   462→      if (timerRef.current) clearInterval(timerRef.current)
   463→    }
   464→  }, [])
   465→
   466→  if (premiumActive) {
   467→    return (
   468→      <Card className="p-4">
   469→        <div className="flex items-center gap-3">
   470→          <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-amber-400 to-amber-500">
   471→            <Crown className="h-6 w-6 text-white" />
   472→          </div>
   473→          <div className="min-w-0 flex-1">
   474→            <p className="flex items-center gap-2 text-base font-semibold">
   475→              Premium active
   476→              <span className="premium-badge">
   477→                <Crown className="h-3 w-3" /> Pro
   478→              </span>
   479→            </p>
   480→            <p className="text-sm text-muted-foreground">
   481→              {premiumUntil
   482→                ? `Valid until ${format(new Date(premiumUntil), 'dd MMM yyyy')}`
   483→                : 'Lifetime'}
   484→            </p>
   485→          </div>
   486→        </div>
   487→      </Card>
   488→    )
   489→  }
   490→
   491→  const plans = [
   492→    { id: '2mo', label: '2 Months', price: 49, months: 2, best: false },
   493→    { id: '6mo', label: '6 Months', price: 99, months: 6, best: false },
   494→    { id: '1yr', label: '1 Year', price: offerPrice, months: 12, best: true },
   495→  ]
   496→
   497→  return (
   498→    <Card className="p-4">
   499→      <div className="mb-3 flex items-center justify-between gap-2">
   500→        <h3 className="flex items-center gap-2 text-base font-semibold">
   501→          <Sparkles className="h-4 w-4 text-primary" /> Upgrade to Premium
   502→        </h3>
   503→        {offerMsLeft && offerMsLeft > 0 && (
   504→          <span className="inline-flex items-center gap-1 rounded-full bg-primary/10 px-2 py-1 text-xs font-semibold text-primary">
   505→            <Clock className="h-3 w-3" />
   506→            {formatCountdown(offerMsLeft)}
   507→          </span>
   508→        )}
   509→      </div>
   510→      {offerMsLeft && offerMsLeft > 0 ? (
   511→        <p className="mb-3 text-xs text-primary">
   512→          Limited-time offer! 1-year plan for ₹{OFFER_PRICE} (was ₹{REGULAR_PRICE})
   513→        </p>
   514→      ) : (
   515→        <p className="mb-3 text-xs text-muted-foreground">
   516→          Offer ended — 1-year plan is back to ₹{REGULAR_PRICE}
   517→        </p>
   518→      )}
   519→      <div className="grid gap-2 sm:grid-cols-3">
   520→        {plans.map((p) => (
   521→          <div
   522→            key={p.id}
   523→            className={`relative flex flex-col gap-1 rounded-xl border-2 p-3 ${
   524→              p.best ? 'border-primary bg-primary/5' : 'border-border'
   525→            }`}
   526→          >
   527→            {p.best && (
   528→              <span className="absolute -top-2 left-1/2 -translate-x-1/2 rounded-full bg-primary px-2 py-0.5 text-[10px] font-bold text-primary-foreground">
   529→                Best Value
   530→              </span>
   531→            )}
   532→            <p className="text-xs font-medium text-muted-foreground">{p.label}</p>
   533→            <p className="text-2xl font-bold text-primary">₹{p.price}</p>
   534→            <p className="text-[10px] text-muted-foreground">
   535→              ₹{Math.round((p.price / p.months) * 100) / 100}/mo
   536→            </p>
   537→            <Button
   538→              size="sm"
   539→              variant={p.best ? 'default' : 'outline'}
   540→              className={`mt-2 min-h-[36px] ${p.best ? 'btn-brand' : ''}`}
   541→              onClick={() => onChoosePlan(p)}
   542→            >
   543→              Choose
   544→            </Button>
   545→          </div>
   546→        ))}
   547→      </div>
   548→    </Card>
   549→  )
   550→}
   551→
   552→function formatCountdown(ms: number): string {
   553→  const totalSec = Math.max(0, Math.floor(ms / 1000))
   554→  const m = Math.floor(totalSec / 60)
   555→  const s = totalSec % 60
   556→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   557→}
