   843→function PremiumSection({
   844→  premiumActive,
   845→  premiumUntil,
   846→  onChoosePlan,
   847→}: {
   848→  premiumActive: boolean
   849→  premiumUntil: string | null
   850→  onChoosePlan: (plan: { id: string; label: string; price: number; months: number }) => void
   851→}) {
   852→  const [offerMsLeft, setOfferMsLeft] = useState<number | null>(null)
   853→  const [offerPrice, setOfferPrice] = useState(OFFER_PRICE)
   854→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   855→
   856→  useEffect(() => {
   857→    const compute = () => {
   858→      const now = Date.now()
   859→      let start = Number(localStorage.getItem(OFFER_KEY) || 0)
   860→      if (!start || now - start > OFFER_LOOP_MS) {
   861→        // restart offer cycle
   862→        start = now
   863→        localStorage.setItem(OFFER_KEY, String(start))
   864→      }
   865→      const elapsed = now - start
   866→      if (elapsed < OFFER_WINDOW_MS) {
   867→        setOfferMsLeft(OFFER_WINDOW_MS - elapsed)
   868→        setOfferPrice(OFFER_PRICE)
   869→      } else {
   870→        setOfferMsLeft(0)
   871→        setOfferPrice(REGULAR_PRICE)
   872→      }
   873→    }
   874→    compute()
   875→    timerRef.current = setInterval(compute, 1000)
   876→    return () => {
   877→      if (timerRef.current) clearInterval(timerRef.current)
   878→    }
   879→  }, [])
   880→
   881→  if (premiumActive) {
   882→    return (
   883→      <div className="taly-card taly-card-hover animate-fade-in-up p-5">
   884→        <div className="flex items-center gap-3">
   885→          <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-amber-400 to-amber-500">
   886→            <Crown className="h-6 w-6 text-white" />

   889→            <p className="flex items-center gap-2 text-base font-semibold">

   894→            </p>
   895→            <p className="text-sm text-muted-foreground/80">
   896→              {premiumUntil
   897→                ? `Valid until ${format(new Date(premiumUntil), 'dd MMM yyyy')}`
   898→                : 'Lifetime'}
   899→            </p>
   900→          </div>
   901→        </div>
   902→      </div>
   903→    )
   904→  }
   905→
   906→  const plans = [
   907→    { id: '2mo', label: '2 Months', price: 49, months: 2, best: false },
   908→    { id: '6mo', label: '6 Months', price: 99, months: 6, best: false },
   909→    { id: '1yr', label: '1 Year', price: offerPrice, months: 12, best: true },
   910→  ]
   911→
   912→  return (
   913→    <div className="taly-card animate-fade-in-up p-5">
   914→      <div className="mb-3 flex items-center justify-between gap-2">
   915→        <div className="section-header">
   916→          <Sparkles className="h-4 w-4 text-primary" /> Upgrade to Premium
   917→        </div>
   918→        {offerMsLeft && offerMsLeft > 0 && (
   919→          <span className="inline-flex items-center gap-1 rounded-full bg-primary/10 px-2 py-1 text-xs font-semibold text-primary">
   920→            <Clock className="h-3 w-3" />
   921→            {formatCountdown(offerMsLeft)}
   922→          </span>
   923→        )}
   924→      </div>
   925→      {offerMsLeft && offerMsLeft > 0 ? (
   926→        <p className="mb-3 text-xs text-primary">
   927→          Limited-time offer! 1-year plan for ₹{OFFER_PRICE} (was ₹{REGULAR_PRICE})
   928→        </p>
   929→      ) : (
   930→        <p className="mb-3 text-xs text-muted-foreground/80">
   931→          Offer ended — 1-year plan is back to ₹{REGULAR_PRICE}
   932→        </p>
   933→      )}
   934→      <div className="grid gap-3 sm:grid-cols-3">
   935→        {plans.map((p) => (
   936→          <div
   937→            key={p.id}
   938→            className={`taly-card taly-card-hover relative flex flex-col gap-1 overflow-hidden p-4 ${
   939→              p.best
   940→                ? 'border border-emerald-500/40 bg-emerald-50 shadow-lg shadow-emerald-500/10 dark:bg-emerald-950/20'
   941→                : ''
   942→            }`}
   943→          >
   944→            {/* Best value ribbon + shimmer overlay */}
   945→            {p.best && (
   946→              <>
   947→                <div className="premium-shimmer pointer-events-none absolute inset-0 opacity-60" />
   948→                <span className="absolute right-0 top-0 rounded-bl-lg bg-gradient-to-r from-amber-400 to-amber-500 px-2 py-1 text-[10px] font-bold uppercase tracking-wide text-white shadow-sm">
   949→                  Best Value
   950→                </span>
   951→              </>
   952→            )}
   953→            <p className="text-xs font-medium text-muted-foreground/80">{p.label}</p>
   954→            <p className={`text-2xl font-bold ${p.best ? 'text-emerald-600 dark:text-emerald-400' : 'text-primary'}`}>₹{p.price}</p>
   955→            <p className="text-[10px] text-muted-foreground/80">
   956→              ₹{Math.round((p.price / p.months) * 100) / 100}/mo
   957→            </p>
   958→            <button
   959→              onClick={() => onChoosePlan(p)}
   960→              className={`mt-2 min-h-[36px] text-sm font-semibold ${
   961→                p.best
   962→                  ? 'action-btn relative overflow-hidden'
   963→                  : 'ghost-btn'
   964→              }`}
   965→            >
   966→              {p.best && (
   967→                <span className="premium-shimmer pointer-events-none absolute inset-0 opacity-40" />
   968→              )}
   969→              <span className="relative">Choose</span>
   970→            </button>
   971→          </div>
   972→        ))}
   973→      </div>
   974→    </div>
   975→  )
   976→}
   977→
   978→function formatCountdown(ms: number): string {
   979→  const totalSec = Math.max(0, Math.floor(ms / 1000))
   980→  const m = Math.floor(totalSec / 60)
   981→  const s = totalSec % 60
   982→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   983→}
