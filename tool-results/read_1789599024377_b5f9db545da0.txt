   913→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   914→

   916→    const compute = () => {
   917→      const now = Date.now()
   918→      let start = Number(localStorage.getItem(OFFER_KEY) || 0)
   919→      if (!start || now - start > OFFER_LOOP_MS) {
   920→        // restart offer cycle
   921→        start = now
   922→        localStorage.setItem(OFFER_KEY, String(start))
   923→      }
   924→      const elapsed = now - start
   925→      if (elapsed < OFFER_WINDOW_MS) {
   926→        setOfferMsLeft(OFFER_WINDOW_MS - elapsed)
   927→        setOfferPrice(OFFER_PRICE)

   929→        setOfferMsLeft(0)
