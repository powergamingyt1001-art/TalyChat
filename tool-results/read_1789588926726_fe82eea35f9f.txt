   109→// ---------------------------------------------------------------------------
   110→// Safe customizer hook (works with or without the CustomizerProvider).
   111→// ---------------------------------------------------------------------------
   112→
   113→function useCustomizerSafe(preferences: any) {
   114→  try {
   115→    const ctx = useCustomizer()
   116→    return ctx
   117→  } catch {
   118→    // CustomizerProvider not available — derive from preferences prop.
   119→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   120→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   121→    return {
   122→      preferences,
   123→      wallpaper,
   124→      fontFamily,
   125→      messageStyle: preferences?.messageStyle || 'bubble',
   126→      fontSize: preferences?.fontSize || 14,
   127→      setPreference: async () => {},
   128→    }
   129→  }
   130→}
