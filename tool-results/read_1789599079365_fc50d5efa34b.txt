   191→   191→export function DiscoverScreen({ onOpenChat, onCreateGroup }: DiscoverScreenProps = {}) {
   192→   192→  const { toast } = useToast()
   193→   193→  const [trending, setTrending] = useState<GroupItem[]>([])
   194→   194→  const [popular, setPopular] = useState<GroupItem[]>([])
   195→   195→  const [newGroups, setNewGroups] = useState<GroupItem[]>([])
   196→   196→  const [sponsored, setSponsored] = useState<any[]>([])
   197→   197→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   198→   198→  const [categoryResults, setCategoryResults] = useState<GroupItem[]>([])
   199→   199→  const [loading, setLoading] = useState(true)
   200→   200→  const [catLoading, setCatLoading] = useState(false)
   201→   201→  const [joining, setJoining] = useState<string | null>(null)
   202→   202→  // V12 — Track the group currently being "left" (used to disable the Leave
   203→   203→  // button + show a spinner during the leave API call).
   204→   204→  const [leaving, setLeaving] = useState<string | null>(null)
   205→   205→  // joinedIds = public joined; requestedIds = private groups with sent requests
   206→   206→  const [joinedIds, setJoinedIds] = useState<Set<string>>(new Set())
   207→   207→  const [requestedIds, setRequestedIds] = useState<Set<string>>(new Set())
   208→   208→  const [preview, setPreview] = useState<GroupItem | null>(null)
   209→   209→
   210→   210→  useEffect(() => {
   211→   211→    let cancelled = false
   212→   212→    ;(async () => {
   213→   213→      setLoading(true)
   214→   214→      try {
   215→   215→        const [t, p, n] = await Promise.all([
   216→   216→          apiFetch('/api/discover?sort=trending').catch(() => null),
   217→   217→          apiFetch('/api/discover?sort=popular').catch(() => null),
   218→   218→          apiFetch('/api/discover?sort=new').catch(() => null),
   219→   219→        ])
   220→   220→        if (cancelled) return
   221→   221→        // Defensive: API returns { groups: [...], sponsored: [...] }.
   222→   222→        // Fall back to bare arrays if the shape is ever flat.
   223→   223→        const tGroups: GroupItem[] = (Array.isArray(t)
   224→   224→          ? t
   225→   225→          : t?.groups || t?.items || []) as GroupItem[]
   226→   226→        const pGroups: GroupItem[] = (Array.isArray(p)
   227→   227→          ? p
   228→   228→          : p?.groups || p?.items || []) as GroupItem[]
   229→   229→        const nGroups: GroupItem[] = (Array.isArray(n)
   230→   230→          ? n
   231→   231→          : n?.groups || n?.items || []) as GroupItem[]
   232→   232→        setTrending(tGroups)
   233→   233→        setPopular(pGroups)
   234→   234→        setNewGroups(nGroups)
   235→   235→        // Sponsored ads: prefer the trending response's sponsored list,
   236→   236→        // fall back to popular's, then to an empty array.
   237→   237→        const sponsoredList: any[] = Array.isArray(t?.sponsored)
   238→   238→          ? t.sponsored
   239→   239→          : Array.isArray(p?.sponsored)
   240→   240→            ? p.sponsored
   241→   241→            : Array.isArray(n?.sponsored)
   242→   242→              ? n.sponsored
   243→   243→              : []
   244→   244→        setSponsored(sponsoredList)
   245→   245→        const joined = new Set<string>([
   246→   246→          ...tGroups.filter((g) => g.isJoined).map((g) => g.id),
   247→   247→          ...pGroups.filter((g) => g.isJoined).map((g) => g.id),
   248→   248→          ...nGroups.filter((g) => g.isJoined).map((g) => g.id),
   249→   249→        ])
   250→   250→        setJoinedIds(joined)
   251→   251→      } finally {
   252→   252→        if (!cancelled) setLoading(false)
   253→   253→      }
   254→   254→    })()
   255→   255→    return () => {
   256→   256→      cancelled = true
   257→   257→    }
   258→   258→  }, [])
   259→   259→
   260→   260→  // Load the user's sent group join requests once on mount so we can mark
   261→   261→  // private groups as "requested".
   262→   262→  useEffect(() => {
   263→   263→    ;(async () => {
   264→   264→      try {
   265→   265→        const res: any = await apiFetch('/api/groups/requests/sent')
   266→   266→        const list: any[] = Array.isArray(res) ? res : (res?.requests || res?.items || [])
   267→   267→        const pending = new Set<string>(
   268→   268→          list
   269→   269→            .filter((r: any) => r.status === 'pending' && r.group?.id)
   270→   270→            .map((r: any) => r.group.id),
   271→   271→        )
   272→   272→        setRequestedIds(pending)
   273→   273→      } catch {
   274→   274→        // ignore — feature best-effort
   275→   275→      }
   276→   276→    })()
   277→   277→  }, [])
   278→   278→
   279→   279→  const fetchCategory = async (cat: string) => {
   280→   280→    setCatLoading(true)
   281→   281→    try {
   282→   282→      const res: any = await apiFetch(
   283→   283→        `/api/discover?category=${encodeURIComponent(cat)}&sort=trending`,
   284→   284→      )
   285→   285→      // Defensive: API returns { groups: [...] }, fall back to bare array.
   286→   286→      const list: GroupItem[] = (Array.isArray(res)
   287→   287→        ? res
   288→   288→        : res?.groups || res?.items || []) as GroupItem[]
   289→   289→      setCategoryResults(list)
   290→   290→    } catch (e: any) {
   291→   291→      toast({ title: e?.message || 'Failed to load category', variant: 'destructive' })
   292→   292→    } finally {
   293→   293→      setCatLoading(false)
   294→   294→    }
   295→   295→  }
   296→   296→
   297→   297→  const handleCategoryClick = (cat: string) => {
   298→   298→    if (activeCategory === cat) {
   299→   299→      setActiveCategory(null)
   300→   300→      setCategoryResults([])
   301→   301→    } else {
   302→   302→      setActiveCategory(cat)
   303→   303→      fetchCategory(cat)
   304→   304→    }
   305→   305→  }
   306→   306→
   307→   307→  // Returns the join state for a group, factoring in joined + requested sets.
   308→   308→  const joinStateFor = (g: GroupItem): JoinState => {
   309→   309→    if (joinedIds.has(g.id)) return 'joined'
   310→   310→    if (!g.isPublic && requestedIds.has(g.id)) return 'requested'
   311→   311→    return 'idle'
   312→   312→  }
   313→   313→
   314→   314→  const handleJoin = async (g: GroupItem) => {
   315→   315→    setJoining(g.id)
   316→   316→    try {
   317→   317→      const res: any = await apiFetch('/api/groups/join', {
   318→   318→        method: 'POST',
   319→   319→        body: JSON.stringify({ groupId: g.id }),
   320→   320→      })
   321→   321→      if (res?.alreadyMember) {
   322→   322→        toast({ title: 'You are already a member' })
   323→   323→        setJoinedIds((prev) => new Set(prev).add(g.id))
   324→   324→      } else if (res?.requested) {
   325→   325→        // Private group → request was sent
   326→   326→        toast({
   327→   327→          title: 'Join request sent! Wait for admin approval.',
   328→   328→        })
   329→   329→        setRequestedIds((prev) => new Set(prev).add(g.id))
   330→   330→      } else {
   331→   331→        // Public group → direct join
   332→   332→        toast({ title: `Joined ${g.name} 🎉` })
   333→   333→        setJoinedIds((prev) => new Set(prev).add(g.id))
   334→   334→      }
   335→   335→    } catch (e: any) {
   336→   336→      toast({ title: e?.message || 'Failed to join', variant: 'destructive' })
   337→   337→    } finally {
   338→   338→      setJoining(null)
   339→   339→    }
   340→   340→  }
   341→   341→
   342→   342→  // V12 — Open the chat for a joined group. Calls onOpenChat (wired from
   343→   343→  // TalyApp) when a conversationId is linked to the group. Falls back to
   344→   344→  // a friendly toast if the callback is missing (defensive — should never
   345→   345→  // happen in production wiring, but avoids a no-op click).
   346→   346→  const handleOpen = (g: GroupItem) => {
   347→   347→    if (!g.conversationId) {
   348→   348→      toast({
   349→   349→        title: 'No chat linked to this group yet',
   350→   350→        description: 'Please try again in a moment.',
   351→   351→        variant: 'destructive',
   352→   352→      })
   353→   353→      return
   354→   354→    }
   355→   355→    if (onOpenChat) {
   356→   356→      onOpenChat({
   357→   357→        conversationId: g.conversationId,
   358→   358→        name: g.name,
   359→   359→        avatar: g.logo || undefined,
   360→   360→        isGroup: true,
   361→   361→      })
   362→   362→    } else {
   363→   363→      toast({ title: 'Open coming soon' })
   364→   364→    }
   365→   365→  }
   366→   366→
   367→   367→  // V12 — Leave a joined group. Calls POST /api/groups/[id]/leave, then
   368→   368→  // removes the group id from joinedIds so the button reverts to Join.
   369→   369→  const handleLeave = async (g: GroupItem) => {
   370→   370→    setLeaving(g.id)
   371→   371→    try {
   372→   372→      await apiFetch(`/api/groups/${g.id}/leave`, { method: 'POST' })
   373→   373→      toast({ title: `Left ${g.name}` })
   374→   374→      setJoinedIds((prev) => {
   375→   375→        const next = new Set(prev)
   376→   376→        next.delete(g.id)
   377→   377→        return next
   378→   378→      })
   379→   379→    } catch (e: any) {
   380→   380→      toast({ title: e?.message || 'Failed to leave group', variant: 'destructive' })
   381→   381→    } finally {
   382→   382→      setLeaving(null)
   383→   383→    }
   384→   384→  }
   385→   385→
   386→   386→  // V12 — Reset the active category filter (used by the "Browse Trending
   387→   387→  // instead" link in the empty state and by the active-filter banner's
   388→   388→  // clear (✕) button).
   389→   389→  const clearCategory = () => {
   390→   390→    setActiveCategory(null)
   391→   391→    setCategoryResults([])
   392→   392→  }
   393→   393→
   394→   394→  // V12 — Handle the "Create {Category} Group" CTA in the no-results
   395→   395→  // empty state. If the parent wired onCreateGroup, delegate to it
   396→   396→  // (TalyApp navigates to the Groups tab). Otherwise show a friendly
   397→   397→  // toast so the click is never a no-op.
   398→   398→  const handleCreateGroup = () => {
   399→   399→    if (onCreateGroup) {
   400→   400→      onCreateGroup(activeCategory || undefined)
   401→   401→    } else {
   402→   402→      toast({
   403→   403→        title: 'Open the Groups tab',
   404→   404→        description: activeCategory
   405→   405→          ? `Tap the + button there to start a ${activeCategory} community.`
   406→   406→          : 'Tap the + button there to start a new community.',
   407→   407→      })
   408→   408→    }
   409→   409→  }
   410→   410→
   411→   411→  const totalGroups = trending.length + popular.length + newGroups.length
   412→   412→
   413→   413→  return (
   414→   414→    <div className="mx-auto max-w-2xl px-4 pb-20 pt-6 lg:pb-6 lg:pt-8">
   415→   415→      {/* Header */}
   416→   416→      <motion.div
   417→   417→        initial={{ opacity: 0, y: -8 }}
   418→   418→        animate={{ opacity: 1, y: 0 }}
   419→   419→        transition={{ duration: 0.3 }}
   420→   420→      >
   421→   421→        <h1 className="section-header">Discover</h1>
   422→   422→        <p className="mt-1 text-sm text-muted-foreground">
   423→   423→          Find communities that match your interests
   424→   424→        </p>
   425→   425→      </motion.div>
   426→   426→
   427→   427→      {/* Category chips — solid emerald active state + horizontal scroll snap */}
   428→   428→      <div
   429→   429→        className="no-scrollbar scroll-pan-y -mx-4 mt-3 w-full overflow-x-auto px-4"
   430→   430→        style={{ scrollSnapType: 'x proximity' }}
   431→   431→      >
   432→   432→        <div className="flex min-w-0 gap-2 pb-1">
   433→   433→          {CATEGORIES.map((cat) => {
   434→   434→            const active = activeCategory === cat
   435→   435→            return (
   436→   436→              <button
   437→   437→                key={cat}
   438→   438→                type="button"
   439→   439→                onClick={() => handleCategoryClick(cat)}
   440→   440→                className={`category-chip min-h-[36px] ${active ? 'active' : ''}`}
   441→   441→                style={{ scrollSnapAlign: 'start' }}
   442→   442→              >
   443→   443→                {cat}
   444→   444→              </button>
   445→   445→            )
   446→   446→          })}
   447→   447→        </div>
   448→   448→      </div>
   449→   449→
   450→   450→      {/* V12 — Active filter banner: shows below the chips row when a
   451→   451→          category filter is selected, with a clear (✕) button to reset. */}
   452→   452→      {activeCategory && (
   453→   453→        <div className="mt-3 flex items-center justify-between gap-2 rounded-lg bg-emerald-50 px-3 py-2 text-xs font-medium text-emerald-700 dark:bg-emerald-950/20 dark:text-emerald-300">
   454→   454→          <span className="truncate">
   455→   455→            Showing results for{' '}
   456→   456→            <span className="font-bold">&lsquo;{activeCategory}&rsquo;</span>
   457→   457→          </span>
   458→   458→          <button
   459→   459→            type="button"
   460→   460→            onClick={clearCategory}
   461→   461→            aria-label="Clear category filter"
   462→   462→            className="inline-flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-emerald-700 transition-colors hover:bg-emerald-500/15 dark:text-emerald-300"
   463→   463→          >
   464→   464→            <X className="h-3.5 w-3.5" />
   465→   465→          </button>
   466→   466→        </div>
   467→   467→      )}
   468→   468→
   469→   469→      {/* Empty state — no groups loaded */}
   470→   470→      {!loading && totalGroups === 0 && !activeCategory && (

   903→   903→// MiniStackedAvatars — 2-3 small overlapping letter-avatars with +N.

   906→   906→function MiniStackedAvatars({
   907→   907→  logo,
   908→   908→  name,
   909→   909→  memberCount,
   910→   910→  small = false,
   911→   911→}: {
   912→   912→  logo?: string | null
   913→   913→  name: string
   914→   914→  memberCount: number
   915→   915→  small?: boolean
   916→   916→}) {
   917→   917→  const baseName = name || 'G'
   918→   918→  const initials = (baseName.replace(/\s+/g, ' ').trim().split(' ').filter(Boolean))
   919→   919→  const derived: string[] = []
   920→   920→  for (let i = 0; i < initials.length && derived.length < 3; i++) {
   921→   921→    derived.push(initials[i][0]?.toUpperCase() || 'U')
   922→   922→  }
   923→   923→  while (derived.length < 3) {
   924→   924→    derived.push(baseName[0]?.toUpperCase() || 'U')
   925→   925→  }
   926→   926→  const visible = derived.slice(0, 3)
   927→   927→  const extra = Math.max(0, memberCount - 3)
   928→   928→  const sizeClass = small ? 'h-4 w-4' : 'h-5 w-5'
   929→   929→  const textSize = small ? 'text-[7px]' : 'text-[8px]'
   930→   930→  const offset = small ? 8 : 10
   931→   931→  const slotClass = (i: number) => {
   932→   932→    switch (i % 3) {
   933→   933→      case 0:
