   101→const AD_RELATED_KEYS = new Set([
   102→  'ads_enabled',
   103→  'ads_private_interval',
   104→  'ads_group_interval',

   107→function AppSettingsSection() {
   108→  const { toast } = useToast()
   109→  const [map, setMap] = React.useState<Record<string, string>>({})
   110→  const [draft, setDraft] = React.useState<Record<string, string>>({})
   111→  const [loading, setLoading] = React.useState(true)
   112→  const [saving, setSaving] = React.useState(false)
   113→
   114→  const load = React.useCallback(async () => {
   115→    setLoading(true)
   116→    try {
   117→      const res: any = await apiFetch('/api/admin/settings')
   118→      setMap(res?.settings || {})
   119→      setDraft(res?.settings || {})
   120→    } catch (e: any) {
   121→      toast({ title: 'Failed to load settings', description: e?.message, variant: 'destructive' })
   122→    } finally {
   123→      setLoading(false)
   124→    }
   125→  }, [toast])
   126→
   127→  React.useEffect(() => {
   128→    load()
   129→  }, [load])
   130→
   131→  const saveAll = async () => {
   132→    const changed = Object.fromEntries(
   133→      Object.entries(draft).filter(([k, v]) => map[k] !== v)
   134→    )
   135→    if (Object.keys(changed).length === 0) {
   136→      toast({ title: 'No changes to save' })
   137→      return
   138→    }
   139→    setSaving(true)
   140→    try {
   141→      await apiFetch('/api/admin/settings', {
   142→        method: 'PUT',
   143→        body: JSON.stringify(changed),
   144→      })
   145→      toast({ title: 'Settings saved', description: `${Object.keys(changed).length} updated` })
   146→      load()
   147→    } catch (e: any) {
   148→      toast({ title: 'Failed to save', description: e?.message, variant: 'destructive' })
   149→    } finally {
   150→      setSaving(false)
   151→    }
   152→  }
   153→
   154→  const addNew = () => {
   155→    const key = prompt('New setting key:')
   156→    if (!key) return
   157→    if (draft[key] !== undefined) {
   158→      toast({ title: 'Key already exists' })
   159→      return
   160→    }
   161→    setDraft((d) => ({ ...d, [key]: '' }))
   162→  }
   163→
   164→  const setVal = (k: string, v: string) => setDraft((d) => ({ ...d, [k]: v }))
   165→
   166→  const adEntries = Object.entries(draft).filter(([k]) => AD_RELATED_KEYS.has(k))
   167→  const otherEntries = Object.entries(draft).filter(([k]) => !AD_RELATED_KEYS.has(k))
   168→
   169→  return (
   170→    <section className="flex flex-col gap-4">
   171→      <div className="flex items-center justify-between gap-3">
   172→        <div>
   173→          <h3 className="flex items-center gap-2 text-base font-semibold">
   174→            <Settings className="size-4 text-primary" /> App settings
   175→          </h3>
   176→          <p className="text-xs text-muted-foreground">Key/value pairs from <code className="font-mono">AppSetting</code></p>
   177→        </div>
   178→        <div className="flex gap-2">
   179→          <Button variant="outline" size="sm" className="h-9" onClick={addNew}>
   180→            <Plus className="size-4" /> New
   181→          </Button>
   182→          <Button className="btn-brand h-9" onClick={saveAll} disabled={saving}>
   183→            {saving ? <Loader2 className="size-4 animate-spin" /> : <Save className="size-4" />}
   184→            Save All
   185→          </Button>
   186→        </div>
   187→      </div>
   188→
   189→      {/* Ad-specific quick controls */}
   190→      {adEntries.length > 0 && (
   191→        <Card>
   192→          <CardHeader className="pb-2">
   193→            <CardTitle className="text-sm">Ad controls</CardTitle>
   194→          </CardHeader>
   195→          <CardContent className="flex flex-col gap-3">
   196→            {adEntries.map(([k, v]) => (
   197→              <div key={k} className="flex items-center justify-between gap-3">
   198→                <div>
   199→                  <Label className="text-sm font-medium">{humanizeKey(k)}</Label>
   200→                  <p className="text-xs text-muted-foreground">{k}</p>
   201→                </div>
   202→                {k === 'ads_enabled' ? (
   203→                  <Switch
   204→                    checked={v === 'true' || v === '1'}
   205→                    onCheckedChange={(c) => setVal(k, c ? 'true' : 'false')}
   206→                  />
   207→                ) : (
   208→                  <Input
   209→                    type="number"
   210→                    value={v}
   211→                    onChange={(e) => setVal(k, e.target.value)}
   212→                    className="h-9 w-32"
   213→                  />
   214→                )}
   215→              </div>
   216→            ))}
   217→          </CardContent>
   218→        </Card>
   219→      )}
   220→
   221→      {/* All settings */}
   222→      <Card>
   223→        <CardContent>
   224→          {loading ? (
   225→            <div className="space-y-2">
   226→              {Array.from({ length: 6 }).map((_, i) => (
   227→                <Skeleton key={i} className="h-12 w-full rounded-lg" />
   228→              ))}
   229→            </div>
   230→          ) : otherEntries.length === 0 && adEntries.length === 0 ? (
   231→            <EmptyState title="No settings yet" hint="Add one with the New button" icon={Settings} />
   232→          ) : (
   233→            <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
   234→              {otherEntries.map(([k, v]) => (
   235→                <div key={k} className="flex flex-col gap-1">
   236→                  <Label className="text-xs font-medium">{k}</Label>
   237→                  <Input
   238→                    value={v}
   239→                    onChange={(e) => setVal(k, e.target.value)}
   240→                    className="h-10"
   241→                  />
   242→                </div>
   243→              ))}
   244→            </div>
   245→          )}
   246→        </CardContent>
   247→      </Card>
   248→    </section>
   249→  )
   250→}
   251→
   252→function humanizeKey(k: string): string {
   253→  return k
   254→    .replace(/_/g, ' ')
   255→    .replace(/\b\w/g, (c) => c.toUpperCase())
   256→}

   264→function AdsSection() {
   265→  const { toast } = useToast()
   266→  const isMobile = useIsMobile()
   267→  const [ads, setAds] = React.useState<AdminAd[]>([])
   268→  const [loading, setLoading] = React.useState(true)
   269→  const [editing, setEditing] = React.useState<AdminAd | null>(null)
   270→  const [creating, setCreating] = React.useState(false)
   271→
   272→  const load = React.useCallback(async () => {
   273→    setLoading(true)
   274→    try {
   275→      const res: any = await apiFetch('/api/admin/ads')
   276→      setAds(res?.ads || [])
   277→    } catch (e: any) {
   278→      toast({ title: 'Failed to load ads', description: e?.message, variant: 'destructive' })
   279→    } finally {
   280→      setLoading(false)
   281→    }
   282→  }, [toast])
   283→
   284→  React.useEffect(() => {
   285→    load()
   286→  }, [load])
   287→
   288→  const remove = async (id: string) => {
   289→    if (!confirm('Delete this ad permanently?')) return
   290→    try {
   291→      await apiFetch('/api/admin/ads', {
   292→        method: 'DELETE',
   293→        body: JSON.stringify({ id }),
   294→      })
   295→      toast({ title: 'Ad deleted' })
   296→      load()
   297→    } catch (e: any) {
   298→      toast({ title: 'Delete failed', description: e?.message, variant: 'destructive' })
   299→    }
   300→  }
   301→
   302→  const toggleActive = async (ad: AdminAd) => {
   303→    try {
   304→      await apiFetch('/api/admin/ads', {
   305→        method: 'PATCH',
   306→        body: JSON.stringify({ id: ad.id, isActive: !ad.isActive }),
   307→      })
   308→      toast({ title: `Ad ${!ad.isActive ? 'enabled' : 'disabled'}` })
   309→      load()
   310→    } catch (e: any) {
   311→      toast({ title: 'Update failed', description: e?.message, variant: 'destructive' })
   312→    }
   313→  }
   314→
   315→  return (
   316→    <section className="flex flex-col gap-4">
   317→      <div className="flex items-center justify-between gap-3">
   318→        <div>
   319→          <h3 className="flex items-center gap-2 text-base font-semibold">
   320→            <ImageIcon className="size-4 text-primary" /> Ad management
   321→          </h3>
   322→          <p className="text-xs text-muted-foreground">Create, edit and toggle ad units</p>
   323→        </div>
   324→        <Button className="btn-brand h-9" onClick={() => setCreating(true)}>
   325→          <Plus className="size-4" /> New Ad
   326→        </Button>
   327→      </div>
   328→
   329→      <Card>
   330→        <CardContent className="p-0">
   331→          {loading ? (
   332→            <div className="space-y-2 p-4">
   333→              {Array.from({ length: 4 }).map((_, i) => (
   334→                <Skeleton key={i} className="h-16 w-full rounded-lg" />
   335→              ))}
   336→            </div>
   337→          ) : ads.length === 0 ? (
   338→            <EmptyState title="No ads yet" hint="Create one to start serving ads" icon={ImageIcon} />
   339→          ) : isMobile ? (
   340→            <div className="divide-y">
   341→              {ads.map((ad) => (
   342→                <AdCard key={ad.id} ad={ad} onEdit={() => setEditing(ad)} onToggle={() => toggleActive(ad)} onDelete={() => remove(ad.id)} />
   343→              ))}
   344→            </div>
   345→          ) : (
   346→            <Table>
   347→              <TableHeader>
   348→                <TableRow>
   349→                  <TableHead>Brand</TableHead>
   350→                  <TableHead>Headline</TableHead>
   351→                  <TableHead>Placement</TableHead>
   352→                  <TableHead>Stats</TableHead>
   353→                  <TableHead>Status</TableHead>
   354→                  <TableHead className="text-right">Actions</TableHead>
   355→                </TableRow>
   356→              </TableHeader>
   357→              <TableBody>
   358→                {ads.map((ad) => (
   359→                  <TableRow key={ad.id}>
   360→                    <TableCell className="font-medium">{ad.brandName}</TableCell>
   361→                    <TableCell className="max-w-[200px] truncate text-sm text-muted-foreground">
   362→                      {ad.headline}
   363→                    </TableCell>
   364→                    <TableCell>
   365→                      <Badge variant="secondary" className="capitalize">{ad.placement}</Badge>
   366→                    </TableCell>
   367→                    <TableCell className="text-xs text-muted-foreground">
   368→                      {ad.impressions.toLocaleString('en-IN')} imp · {ad.clicks.toLocaleString('en-IN')} clicks
   369→                    </TableCell>
   370→                    <TableCell>
   371→                      <div className="flex items-center gap-2">
   372→                        <Switch checked={ad.isActive} onCheckedChange={() => toggleActive(ad)} />
   373→                        <span className="text-xs">{ad.isActive ? 'Active' : 'Off'}</span>
   374→                      </div>
   375→                    </TableCell>
   376→                    <TableCell className="text-right">
   377→                      <div className="flex justify-end gap-1">
   378→                        <Button size="icon" variant="ghost" className="size-9" onClick={() => setEditing(ad)} aria-label="Edit ad">
   379→                          <Pencil className="size-4" />
   380→                        </Button>
   381→                        <Button size="icon" variant="ghost" className="size-9 text-destructive" onClick={() => remove(ad.id)} aria-label="Delete ad">
   382→                          <Trash2 className="size-4" />
   383→                        </Button>
   384→                      </div>
   385→                    </TableCell>
   386→                  </TableRow>
   387→                ))}
   388→              </TableBody>
   389→            </Table>
   390→          )}
   391→        </CardContent>
   392→      </Card>
   393→
   394→      <AdEditor
   395→        open={creating || !!editing}
   396→        ad={editing}
   397→        onClose={() => {
   398→          setCreating(false)
   399→          setEditing(null)
   400→        }}
   401→        onSaved={() => {
   402→          setCreating(false)
   403→          setEditing(null)
   404→          load()
   405→        }}
   406→      />
   407→    </section>
   408→  )
   409→}

   461→function AdEditor({
   462→  open,
   463→  ad,
   464→  onClose,
   465→  onSaved,
   466→}: {
   467→  open: boolean
   468→  ad: AdminAd | null
   469→  onClose: () => void
   470→  onSaved: () => void
   471→}) {
   472→  const { toast } = useToast()
   473→  const [saving, setSaving] = React.useState(false)
   474→  const [form, setForm] = React.useState<any>({})
   475→
   476→  React.useEffect(() => {
   477→    if (!open) return
   478→    if (ad) {
   479→      setForm({
   480→        brandName: ad.brandName,
   481→        headline: ad.headline,
   482→        description: ad.description || '',
   483→        imageUrl: ad.imageUrl || '',
   484→        ctaText: ad.ctaText || 'Learn More',
   485→        ctaUrl: ad.ctaUrl || '',
   486→        placement: ad.placement,
   487→        category: ad.category || '',
   488→        startDate: ad.startDate ? ad.startDate.slice(0, 10) : '',
   489→        endDate: ad.endDate ? ad.endDate.slice(0, 10) : '',
   490→        isActive: ad.isActive,
   491→      })
   492→    } else {
   493→      setForm({
   494→        brandName: '',
   495→        headline: '',
   496→        description: '',
   497→        imageUrl: '',
   498→        ctaText: 'Learn More',
   499→        ctaUrl: '',
   500→        placement: 'in-chat',
   501→        category: '',
   502→        startDate: new Date().toISOString().slice(0, 10),
   503→        endDate: '',
   504→        isActive: true,
   505→      })
   506→    }
   507→  }, [ad, open])
   508→
   509→  const submit = async () => {
   510→    if (!form.brandName || !form.headline) {
   511→      toast({ title: 'brandName and headline are required', variant: 'destructive' })
   512→      return
   513→    }
   514→    setSaving(true)
   515→    try {
   516→      const payload = {
   517→        ...form,
   518→        category: form.category || null,
   519→        endDate: form.endDate || null,
   520→        startDate: form.startDate || new Date().toISOString(),
   521→      }
   522→      if (ad) {
   523→        await apiFetch('/api/admin/ads', {
   524→          method: 'PATCH',
   525→          body: JSON.stringify({ id: ad.id, ...payload }),
   526→        })
   527→      } else {
   528→        await apiFetch('/api/admin/ads', {
   529→          method: 'POST',
   530→          body: JSON.stringify(payload),
   531→        })
   532→      }
   533→      toast({ title: ad ? 'Ad updated' : 'Ad created' })
   534→      onSaved()
   535→    } catch (e: any) {
   536→      toast({ title: 'Save failed', description: e?.message, variant: 'destructive' })
   537→    } finally {
   538→      setSaving(false)
   539→    }
   540→  }
   541→
   542→  return (
   543→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   544→      <DialogContent className="sm:max-w-lg">
   545→        <DialogHeader>
   546→          <DialogTitle>{ad ? 'Edit ad' : 'Create ad'}</DialogTitle>
   547→          <DialogDescription>Configure creative, placement and schedule</DialogDescription>
   548→        </DialogHeader>
   549→
   550→        <div className="grid max-h-[60vh] gap-3 overflow-y-auto pr-1">
   551→          <LabeledInput label="Brand name" value={form.brandName} onChange={(v) => setForm((f: any) => ({ ...f, brandName: v }))} placeholder="Acme" />
   552→          <LabeledInput label="Headline" value={form.headline} onChange={(v) => setForm((f: any) => ({ ...f, headline: v }))} placeholder="Check out our deal" />
   553→          <LabeledTextarea label="Description" value={form.description} onChange={(v) => setForm((f: any) => ({ ...f, description: v }))} placeholder="Optional body copy" />
   554→          <LabeledInput label="Image URL" value={form.imageUrl} onChange={(v) => setForm((f: any) => ({ ...f, imageUrl: v }))} placeholder="/uploads/…" />
   555→          <div className="grid grid-cols-2 gap-3">
   556→            <LabeledInput label="CTA text" value={form.ctaText} onChange={(v) => setForm((f: any) => ({ ...f, ctaText: v }))} />
   557→            <LabeledInput label="CTA URL" value={form.ctaUrl} onChange={(v) => setForm((f: any) => ({ ...f, ctaUrl: v }))} />
   558→          </div>
   559→          <div className="grid grid-cols-2 gap-3">
   560→            <Field label="Placement">
   561→              <Select value={form.placement} onValueChange={(v) => setForm((f: any) => ({ ...f, placement: v }))}>
   562→                <SelectTrigger className="h-10 w-full">
   563→                  <SelectValue />
   564→                </SelectTrigger>
   565→                <SelectContent>
   566→                  {AD_PLACEMENTS.map((p) => (
   567→                    <SelectItem key={p} value={p}>{p}</SelectItem>
   568→                  ))}
   569→                </SelectContent>
   570→              </Select>
   571→            </Field>
   572→            <LabeledInput label="Category (optional)" value={form.category} onChange={(v) => setForm((f: any) => ({ ...f, category: v }))} />
   573→          </div>
   574→          <div className="grid grid-cols-2 gap-3">

   601→// ============================================================
   602→// Section 3 — Change admin password
   603→// ============================================================
   604→
   605→function PasswordSection() {
   606→  const { toast } = useToast()
   607→  const [current, setCurrent] = React.useState('')
   608→  const [next, setNext] = React.useState('')
   609→  const [confirm, setConfirm] = React.useState('')
   610→  const [saving, setSaving] = React.useState(false)
   611→
   612→  const submit = async () => {
   613→    if (!current || !next || !confirm) {
   614→      toast({ title: 'All fields required', variant: 'destructive' })
   615→      return
   616→    }
   617→    if (next !== confirm) {
   618→      toast({ title: 'New passwords do not match', variant: 'destructive' })
   619→      return
   620→    }
   621→    if (next.length < 6) {
   622→      toast({ title: 'New password must be at least 6 characters', variant: 'destructive' })
   623→      return
   624→    }
   625→    setSaving(true)
   626→    try {
   627→      await apiFetch('/api/admin/password', {
   628→        method: 'POST',
   629→        body: JSON.stringify({ currentPassword: current, newPassword: next }),
   630→      })
   631→      toast({ title: 'Password updated' })
   632→      setCurrent('')
   633→      setNext('')
   634→      setConfirm('')
   635→    } catch (e: any) {
   636→      toast({ title: 'Failed to update', description: e?.message, variant: 'destructive' })
   637→    } finally {
   638→      setSaving(false)
   639→    }
   640→  }
   641→
   642→  return (
   643→    <section className="flex flex-col gap-3">
   644→      <h3 className="flex items-center gap-2 text-base font-semibold">
   645→        <KeyRound className="size-4 text-primary" /> Change admin password
   646→      </h3>
   647→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   648→        <LabeledInput label="Current password" type="password" value={current} onChange={setCurrent} />
   649→        <LabeledInput label="New password" type="password" value={next} onChange={setNext} />
   650→        <LabeledInput label="Confirm new" type="password" value={confirm} onChange={setConfirm} />
   651→      </div>
   652→      <div className="flex justify-end">
   653→        <Button className="btn-brand h-10" onClick={submit} disabled={saving}>
   654→          {saving ? <Loader2 className="size-4 animate-spin" /> : <Save className="size-4" />} Update password
   655→        </Button>
   656→      </div>
   657→    </section>
   658→  )
   659→}

   665→function PaymentsSection() {
   666→  const { toast } = useToast()
   667→  const [payments, setPayments] = React.useState<AdminPayment[]>([])
   668→  const [loading, setLoading] = React.useState(true)
   669→  const [actingId, setActingId] = React.useState<string | null>(null)
   670→  const [statusFilter, setStatusFilter] = React.useState<'pending' | 'approved' | 'rejected'>('pending')
   671→
   672→  const load = React.useCallback(async (s: 'pending' | 'approved' | 'rejected') => {
   673→    setLoading(true)
   674→    try {
   675→      const res: any = await apiFetch(`/api/admin/payments?status=${s}`)
   676→      setPayments(res?.payments || [])
   677→    } catch (e: any) {
   678→      toast({ title: 'Failed to load payments', description: e?.message, variant: 'destructive' })
   679→    } finally {
   680→      setLoading(false)
   681→    }
   682→  }, [toast])
   683→
   684→  React.useEffect(() => {
   685→    load(statusFilter)
   686→  }, [statusFilter, load])
   687→
   688→  const act = async (id: string, action: 'approve' | 'reject') => {
   689→    setActingId(id)
   690→    try {
   691→      await apiFetch('/api/admin/payments', {
   692→        method: 'POST',
   693→        body: JSON.stringify({ id, action }),
   694→      })
   695→      toast({ title: action === 'approve' ? 'Payment approved — premium granted' : 'Payment rejected' })
   696→      // remove from current list if filter is pending
   697→      if (statusFilter === 'pending') {
   698→        setPayments((prev) => prev.filter((p) => p.id !== id))
   699→      } else {
   700→        load(statusFilter)
   701→      }
   702→    } catch (e: any) {
   703→      toast({ title: 'Action failed', description: e?.message, variant: 'destructive' })
   704→    } finally {
   705→      setActingId(null)
   706→    }
   707→  }
   708→
   709→  return (
   710→    <section className="flex flex-col gap-4">
   711→      <div className="flex items-center justify-between gap-3">
   712→        <h3 className="flex items-center gap-2 text-base font-semibold">
   713→          <CreditCard className="size-4 text-primary" /> Payment proofs
   714→        </h3>
   715→        <Select value={statusFilter} onValueChange={(v) => setStatusFilter(v as any)}>
   716→          <SelectTrigger className="h-9 w-[150px]">
   717→            <SelectValue />
   718→          </SelectTrigger>
   719→          <SelectContent>
   720→            <SelectItem value="pending">Pending</SelectItem>
   721→            <SelectItem value="approved">Approved</SelectItem>
   722→            <SelectItem value="rejected">Rejected</SelectItem>
   723→          </SelectContent>
   724→        </Select>
   725→      </div>
   726→
   727→      {loading ? (
   728→        <div className="space-y-2">
   729→          {Array.from({ length: 3 }).map((_, i) => (
   730→            <Skeleton key={i} className="h-24 w-full rounded-lg" />
   731→          ))}
   732→        </div>
   733→      ) : payments.length === 0 ? (
   734→        <EmptyState title="No payments" hint={`No ${statusFilter} payment proofs`} icon={CreditCard} />
   735→      ) : (
   736→        <div className="grid grid-cols-1 gap-3 md:grid-cols-2">
   737→          {payments.map((p) => (
   738→            <PaymentCard key={p.id} payment={p} acting={actingId === p.id} onAct={act} />
   739→          ))}
   740→        </div>
   741→      )}
   742→    </section>
   743→  )
