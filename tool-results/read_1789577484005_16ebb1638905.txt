   154→export function ProfileScreen() {
   155→  const { user, updateUser, logout, setAuth } = useAuth()
   156→  const { toast } = useToast()

   158→  const [profile, setProfile] = useState<any>(null)
   159→  const [editOpen, setEditOpen] = useState(false)
   160→  const [settingsOpen, setSettingsOpen] = useState(false)
   161→  const [buyPlan, setBuyPlan] = useState<null | { id: string; label: string; price: number; months: number }>(null)
   162→  const [redeemCode, setRedeemCode] = useState('')
   163→  const [redeemLoading, setRedeemLoading] = useState(false)
   164→  const [referral, setReferral] = useState<ReferralData | null>(null)
   165→
   166→  // V2 — admin login via Redeem section.
   167→  // When the user types "admin.in" (case-insensitive) as the redeem code
   168→  // and taps Redeem, we open an admin-login password dialog instead of
   169→  // calling the normal /api/redeem endpoint. On success we setAuth() and
   170→  // the app re-renders as the admin panel (see src/app/page.tsx).
   171→  const [adminLoginOpen, setAdminLoginOpen] = useState(false)
   172→  const [adminPassword, setAdminPassword] = useState('')
   173→  const [adminLoginLoading, setAdminLoginLoading] = useState(false)
   174→
   175→  // Block list state
   176→  const [blockedOpen, setBlockedOpen] = useState(false)
   177→  const [blockedCount, setBlockedCount] = useState<number>(0)
   178→
   179→  // Behavior + watch-ad state
   180→  const [behavior, setBehavior] = useState<BehaviorData | null>(null)
   181→  const [watchingAd, setWatchingAd] = useState<AdData | null>(null)
   182→  const [watching, setWatching] = useState(false)
   183→  const [watchCountdown, setWatchCountdown] = useState(5)
   184→
   185→  // Team sheet state
   186→  const [teamSheetOpen, setTeamSheetOpen] = useState(false)
   187→
   188→  const loadProfile = useCallback(async () => {
   189→    setLoading(true)
   190→    try {
   191→      const res: any = await apiFetch('/api/users/me')
   192→      // Defensive: API returns { user: {...}, preferences: {...} }.
   193→      // Fall back to bare object if shape changes.
   194→      const u = res?.user || (res?.id ? res : null)
   195→      setProfile(u)
   196→      if (u) updateUser(u)
   197→    } catch (e: any) {
   198→      toast({ title: e.message, variant: 'destructive' })
   199→    } finally {
   200→      setLoading(false)
   201→    }
   202→  }, [toast, updateUser])
   203→
   204→  const loadReferral = useCallback(async () => {
   205→    try {
   206→      const res: any = await apiFetch('/api/referral/me')
   207→      // Defensive: API returns flat { code, count, recent, tierReached, progress }.
   208→      // Some backends might wrap as { referral: {...} } — handle both.
   209→      const data = res?.referral || res?.data || res
   210→      setReferral(data as ReferralData)
   211→    } catch {
   212→      setReferral(null)
   213→    }
   214→  }, [])
   215→
   216→  // Fetch behavior summary (score, adsWatchedToday, maxAdsPerDay, canMessage)
   217→  const loadBehavior = useCallback(async () => {
   218→    try {
   219→      const res: any = await apiFetch('/api/behavior/me')
   220→      const data = res?.behavior || res?.data || res
   221→      setBehavior({
   222→        score: typeof data?.score === 'number' ? data.score : 100,
   223→        adsWatchedToday: data?.adsWatchedToday ?? 0,
   224→        maxAdsPerDay: data?.maxAdsPerDay ?? 10,
   225→        canMessage: Boolean(data?.canMessage),
   226→      })
   227→    } catch {
   228→      // Defensive default — assume full score if route fails
   229→      setBehavior({
   230→        score: 100,
   231→        adsWatchedToday: 0,
   232→        maxAdsPerDay: 10,
   233→        canMessage: true,
   234→      })
   235→    }
   236→  }, [])
   237→
   238→  // Watch an ad: POST /api/behavior/watch-ad, then open dialog for 5s.
   239→  const handleWatchAd = useCallback(async () => {
   240→    if (watching) return
   241→    if (behavior && behavior.adsWatchedToday >= behavior.maxAdsPerDay) {
   242→      toast({ title: 'Daily limit reached — come back tomorrow!' })
   243→      return
   244→    }
   245→    setWatching(true)
   246→    try {
   247→      const res: any = await apiFetch('/api/behavior/watch-ad', { method: 'POST' })
   248→      const data = res?.behavior || res?.data || res
   249→      const ad: AdData | null = data?.ad ?? null
   250→      // Update behavior immediately with new score/adsWatchedToday.
   251→      setBehavior({
   252→        score: typeof data?.score === 'number' ? data.score : (behavior?.score ?? 100),
   253→        adsWatchedToday: data?.adsWatchedToday ?? (behavior?.adsWatchedToday ?? 0) + 1,
   254→        maxAdsPerDay: data?.maxAdsPerDay ?? behavior?.maxAdsPerDay ?? 10,
   255→        canMessage: Boolean(data?.canMessage ?? (behavior?.canMessage ?? true)),
   256→      })
   257→      if (ad) {
   258→        setWatchingAd(ad)
   259→        setWatchCountdown(5)
   260→      } else {
   261→        toast({ title: 'Behavior +1! No ad creative available right now.' })
   262→      }
   263→    } catch (e: any) {
   264→      toast({ title: e?.message || 'Failed to watch ad', variant: 'destructive' })
   265→    } finally {
   266→      setWatching(false)
   267→    }
   268→  }, [behavior, watching, toast])
   269→
   270→  // Select a referral task tier.
   271→  const handleSelectTask = useCallback(
   272→    async (tier: string) => {
   273→      try {
   274→        await apiFetch('/api/referral/task', {
   275→          method: 'POST',
   276→          body: JSON.stringify({ tier }),
   277→        })
   278→        toast({ title: 'Task selected! Get referring 🚀' })
   279→        await loadReferral()
   280→      } catch (e: any) {
   281→        toast({ title: e?.message || 'Failed to select task', variant: 'destructive' })
   282→      }
   283→    },
   284→    [loadReferral, toast],
   285→  )
   286→
   287→  // Claim the active referral task reward.
   288→  const handleClaimTask = useCallback(async () => {
   289→    try {
   290→      const res: any = await apiFetch('/api/referral/task/claim', { method: 'POST' })
   291→      const data = res?.claim || res?.data || res
   292→      const until = data?.premiumUntil
   293→      toast({
   294→        title: 'Premium granted! 🎉',
   295→        description: until
   296→          ? `Valid until ${format(new Date(until), 'dd MMM yyyy')}`
   297→          : undefined,
   298→      })
   299→      await loadReferral()
   300→      await loadProfile()
   301→    } catch (e: any) {
   302→      toast({ title: e?.message || 'Failed to claim reward', variant: 'destructive' })
   303→    }
   304→  }, [loadProfile, loadReferral, toast])
   305→
   306→
   307→  // Fetch the blocked-users count (for the badge on Privacy section).
   308→  const loadBlockedCount = useCallback(async () => {
   309→    try {
   310→      const res: any = await apiFetch('/api/blocks')
   311→      const list: any[] = Array.isArray(res)
   312→        ? res
   313→        : Array.isArray(res?.blocks)
   314→          ? res.blocks
   315→          : Array.isArray(res?.items)
   316→            ? res.items
   317→            : []
   318→      setBlockedCount(list.length)
   319→    } catch {
   320→      setBlockedCount(0)
   321→    }
   322→  }, [])
   323→
   324→  useEffect(() => {
   325→     
   326→    loadProfile()
   327→     
   328→    loadReferral()
   329→    loadBlockedCount()
   330→    loadBehavior()
   331→  }, [loadProfile, loadReferral, loadBlockedCount, loadBehavior])
   332→
   333→  if (loading && !profile) {
   334→    return (
   335→      <div className="flex min-h-[60dvh] items-center justify-center">
   336→        <Loader2 className="h-8 w-8 animate-spin text-primary" />
   337→      </div>
   338→    )
   339→  }
   340→
   341→  if (!profile) {
   342→    return (
   343→      <div className="mx-auto max-w-2xl p-4 text-center text-muted-foreground">
   344→        Could not load profile. Please try again.
   345→      </div>
   346→    )
   347→  }
   348→
   349→  const premiumActive =
   350→    profile.isPremium &&
   351→    profile.premiumUntil &&
   352→    new Date(profile.premiumUntil) > new Date()
   353→
   354→  const copyReferralLink = async () => {
   355→    const link = `${APP_DOMAIN}/?ref=${user?.username || profile.username}`
   356→    try {
   357→      await navigator.clipboard.writeText(link)
   358→      toast({ title: 'Referral link copied!' })
   359→    } catch {
   360→      toast({ title: 'Copy failed', variant: 'destructive' })
   361→    }
   362→  }
   363→
   364→  const handleRedeem = async () => {
   365→    if (!redeemCode.trim()) {
   366→      toast({ title: 'Enter a code first', variant: 'destructive' })
   367→      return
   368→    }
   369→    // V2 — admin.in special case: open the admin login dialog instead of
   370→    // calling /api/redeem. The case-insensitive match mirrors the backend
   371→    // /api/auth/login behavior (which lower-cases the email).
   372→    if (redeemCode.trim().toLowerCase() === 'admin.in') {
   373→      setAdminPassword('')
   374→      setAdminLoginOpen(true)
   375→      return
   376→    }
   377→    setRedeemLoading(true)
   378→    try {
   379→      const res: any = await apiFetch('/api/redeem', {
   380→        method: 'POST',
   381→        body: JSON.stringify({ code: redeemCode.trim() }),
   382→      })
   383→      // Defensive: API returns flat { ok, premiumUntil }, but may also be
   384→      // wrapped as { redeem: {...} } or { data: {...} } in some envs.
   385→      const data: any = res?.redeem || res?.data || res
   386→      const premiumUntil = data?.premiumUntil
   387→      toast({
   388→        title: 'Premium activated!',
   389→        description: premiumUntil
   390→          ? `Valid until ${format(new Date(premiumUntil), 'dd MMM yyyy')}`
   391→          : 'Your premium has been extended.',
   392→      })
   393→      setRedeemCode('')
   394→      loadProfile()
   395→    } catch (e: any) {
   396→      toast({ title: e.message, variant: 'destructive' })
   397→    } finally {
   398→      setRedeemLoading(false)
   399→    }
   400→  }
   401→
   402→  // V2 — submit admin login (called from the Admin Login dialog).
   403→  // On success, setAuth() with the returned user/token; the app will then
   404→  // re-render as the admin panel.
   405→  const submitAdminLogin = async () => {
   406→    if (!adminPassword) {
   407→      toast({ title: 'Enter the admin password', variant: 'destructive' })

   409→    }
   410→    setAdminLoginLoading(true)
   411→    try {
   412→      const res: any = await apiFetch('/api/auth/login', {
   413→        method: 'POST',
   414→        body: JSON.stringify({ email: 'admin.in', password: adminPassword }),
   415→      })
   416→      const u = res?.user || (res?.id ? res : null)
   417→      const t = res?.token
   418→      if (!u || !t) {
   419→        throw new Error('Invalid admin response')
   420→      }
   421→      setAuth(u, t)
   422→      toast({
   423→        title: 'Welcome, Admin!',
   424→        description: 'Switching to the admin panel…',
   425→      })
   426→      setAdminLoginOpen(false)
   427→      setAdminPassword('')
   428→      setRedeemCode('')
   429→    } catch (e: any) {
   430→      toast({ title: e?.message || 'Admin login failed', variant: 'destructive' })
   431→    } finally {
   432→      setAdminLoginLoading(false)
   433→    }
   434→  }

   441→          <PremiumAvatar

   443→              isPremium: profile.isPremium,

   453→            <div className="flex items-center justify-center gap-2 sm:justify-start">

   455→              {premiumActive && (
   456→                <span className="premium-badge">
   457→                  <Crown className="h-3 w-3" /> Premium

   459→              )}
   460→            </div>
   461→            <p className="truncate text-sm text-muted-foreground">@{profile.username}</p>
   462→            {profile.bio ? (
   463→              <p className="mt-2 line-clamp-3 text-sm">{profile.bio}</p>
   464→            ) : (
   465→              <p className="mt-2 text-sm italic text-muted-foreground">No bio yet.</p>
   466→            )}
   467→            <div className="mt-2 flex flex-wrap items-center justify-center gap-3 text-xs text-muted-foreground sm:justify-start">
   468→              <span className="inline-flex items-center gap-1">
   469→                <span
   470→                  className={`inline-block h-2 w-2 rounded-full ${
   471→                    profile.isOnline ? 'bg-green-500' : 'bg-muted-foreground'
   472→                  }`}
   473→                />
   474→                {profile.isOnline ? 'Online' : 'Offline'}
   475→              </span>
   476→              {profile.createdAt && (
   477→                <span className="inline-flex items-center gap-1">
   478→                  <CalendarDays className="h-3 w-3" />
   479→                  Joined {format(new Date(profile.createdAt), 'MMM yyyy')}
   480→                </span>
   481→              )}
   482→            </div>
   483→            <div className="mt-4">
   484→              <Button
   485→                onClick={() => setEditOpen(true)}
   486→                variant="outline"
   487→                size="sm"
   488→                className="min-h-[40px]"
   489→              >
   490→                <Pencil className="mr-1 h-4 w-4" /> Edit Profile
   491→              </Button>
   492→            </div>
   493→          </div>
   494→        </div>
   495→      </Card>
   496→
   497→      {/* Behavior Bar */}
   498→      <Separator />
   499→      <BehaviorBar behavior={behavior} />
   500→
   501→      {/* Watch Behavior (watch ads to increase score) */}
   502→      <Separator />
   503→      <WatchBehaviorCard
   504→        behavior={behavior}
   505→        watching={watching}
   506→        onWatchAd={handleWatchAd}
   507→      />
   508→
   509→      {/* Premium section */}
   510→      <Separator />
   511→      <PremiumSection
   512→        premiumActive={premiumActive}
   513→        premiumUntil={profile.premiumUntil}
   514→        onChoosePlan={(plan) => setBuyPlan(plan)}
   515→      />
   516→
   517→      {/* Redeem code */}
   518→      <Separator />
   519→      <Card className="p-4">
   520→        <h3 className="mb-3 flex items-center gap-2 text-base font-semibold">
   521→          <Gift className="h-4 w-4 text-primary" /> Redeem a Code
   522→        </h3>
   523→        <div className="flex gap-2">
   524→          <Input
   525→            value={redeemCode}
   526→            onChange={(e) => setRedeemCode(e.target.value)}
   527→            placeholder="Enter redeem code (e.g., TALY-XXXXXXXX)"
   528→            className="min-h-[44px] flex-1 uppercase"
   529→            onKeyDown={(e) => {
   530→              if (e.key === 'Enter') handleRedeem()
   531→            }}
   532→          />
   533→          <Button
   534→            onClick={handleRedeem}
   535→            disabled={redeemLoading}
   536→            className="btn-brand min-h-[44px] px-5"
   537→          >
   538→            {redeemLoading ? (
   539→              <Loader2 className="h-4 w-4 animate-spin" />
   540→            ) : (
   541→              'Redeem'
   542→            )}
   543→          </Button>
   544→        </div>
   545→        <p className="mt-2 text-xs text-muted-foreground">
   546→          Get codes from promotions, friends, or the founder.
   547→        </p>
   548→      </Card>
   549→
   550→      {/* V2 — Admin Login dialog (opened when user types 'admin.in' as the

   553→      <Dialog open={adminLoginOpen} onOpenChange={(o) => !adminLoginLoading && setAdminLoginOpen(o)}>
   554→        <DialogContent className="sm:max-w-md">
   555→          <DialogHeader>
   556→            <DialogTitle className="flex items-center gap-2">
   557→              <ShieldAlert className="h-5 w-5 text-primary" /> Admin Login
   558→            </DialogTitle>
   559→            <DialogDescription>
   560→              Enter the admin password to switch to the TalyChat admin panel.
   561→            </DialogDescription>
   562→          </DialogHeader>
   563→          <div className="flex flex-col gap-3">
   564→            <Label htmlFor="admin-password" className="text-xs font-medium text-muted-foreground">
   565→              Admin password
   566→            </Label>
   567→            <Input
   568→              id="admin-password"
   569→              type="password"
   570→              value={adminPassword}
   571→              onChange={(e) => setAdminPassword(e.target.value)}
   572→              placeholder="Admin password"
   573→              className="min-h-[44px]"
   574→              autoFocus

   576→                if (e.key === 'Enter') submitAdminLogin()
   577→              }}
   578→            />
   579→            <p className="text-xs text-muted-foreground">
   580→              You entered <code className="font-mono font-semibold">admin.in</code> as the

   582→            </p>
   583→          </div>
   584→          <DialogFooter className="gap-2">
   585→            <Button

   730→}) {

   733→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   734→

   736→    const compute = () => {
   737→      const now = Date.now()
   738→      let start = Number(localStorage.getItem(OFFER_KEY) || 0)
   739→      if (!start || now - start > OFFER_LOOP_MS) {
   740→        // restart offer cycle
   741→        start = now
   742→        localStorage.setItem(OFFER_KEY, String(start))
   743→      }
   744→      const elapsed = now - start
   745→      if (elapsed < OFFER_WINDOW_MS) {
   746→        setOfferMsLeft(OFFER_WINDOW_MS - elapsed)
   747→        setOfferPrice(OFFER_PRICE)
   748→      } else {
   749→        setOfferMsLeft(0)
   750→        setOfferPrice(REGULAR_PRICE)
   751→      }
   752→    }
   753→    compute()
   754→    timerRef.current = setInterval(compute, 1000)
   755→    return () => {
   756→      if (timerRef.current) clearInterval(timerRef.current)
   757→    }
   758→  }, [])

   779→          </div>
   780→        </div>
   781→      </Card>
   782→    )
   783→  }
   784→
   785→  const plans = [
   786→    { id: '2mo', label: '2 Months', price: 49, months: 2, best: false },
   787→    { id: '6mo', label: '6 Months', price: 99, months: 6, best: false },
   788→    { id: '1yr', label: '1 Year', price: offerPrice, months: 12, best: true },
   789→  ]
   790→
   791→  return (
   792→    <Card className="p-4">
   793→      <div className="mb-3 flex items-center justify-between gap-2">
   794→        <h3 className="flex items-center gap-2 text-base font-semibold">
   795→          <Sparkles className="h-4 w-4 text-primary" /> Upgrade to Premium

   799→            <Clock className="h-3 w-3" />

   801→          </span>
   802→        )}
   803→      </div>
   804→      {offerMsLeft && offerMsLeft > 0 ? (
   805→        <p className="mb-3 text-xs text-primary">
   806→          Limited-time offer! 1-year plan for ₹{OFFER_PRICE} (was ₹{REGULAR_PRICE})

   814→        {plans.map((p) => (
   815→          <div
   816→            key={p.id}
   817→            className={`relative flex flex-col gap-1 rounded-xl border-2 p-3 ${
   818→              p.best ? 'border-primary bg-primary/5' : 'border-border'

   820→          >

   822→              <span className="absolute -top-2 left-1/2 -translate-x-1/2 rounded-full bg-primary px-2 py-0.5 text-[10px] font-bold text-primary-foreground">
   823→                Best Value
   824→              </span>
   825→            )}
   826→            <p className="text-xs font-medium text-muted-foreground">{p.label}</p>
   827→            <p className="text-2xl font-bold text-primary">₹{p.price}</p>
   828→            <p className="text-[10px] text-muted-foreground">
   829→              ₹{Math.round((p.price / p.months) * 100) / 100}/mo

   832→              size="sm"
   833→              variant={p.best ? 'default' : 'outline'}
   834→              className={`mt-2 min-h-[36px] ${p.best ? 'btn-brand' : ''}`}

   841→      </div>
   842→    </Card>
   843→  )

   846→function formatCountdown(ms: number): string {
   847→  const totalSec = Math.max(0, Math.floor(ms / 1000))
   848→  const m = Math.floor(totalSec / 60)
   849→  const s = totalSec % 60
   850→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   851→}
