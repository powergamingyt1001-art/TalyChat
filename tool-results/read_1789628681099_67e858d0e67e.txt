   113→// ---------------------------------------------------------------------------
   114→// Safe customizer hook (works with or without the CustomizerProvider).
   115→// ---------------------------------------------------------------------------
   116→
   117→function useCustomizerSafe(preferences: any) {
   118→  try {
   119→    const ctx = useCustomizer()
   120→    return ctx
   121→  } catch {
   122→    // CustomizerProvider not available — derive from preferences prop.
   123→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   124→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   125→    return {
   126→      preferences,
   127→      wallpaper,
   128→      fontFamily,
   129→      messageStyle: preferences?.messageStyle || 'bubble',
   130→      fontSize: preferences?.fontSize || 14,
   131→      setPreference: async () => {},
   132→    }
   133→  }
   134→}
