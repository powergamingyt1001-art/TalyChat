   469→// ============================================================
   470→// Drilldown dialog — opens when any metric card is clicked.
   471→// Renders a mini chart for that metric.
   472→// ============================================================
   473→function DrilldownDialog({
   474→  metric,
   475→  stats,
   476→  period,
   477→  onClose,
   478→}: {
   479→  metric: MetricDef | UserStatusDef | SubsDef | null
   480→  stats: AdminStats | null
   481→  period: Period
   482→  onClose: () => void
   483→}) {
   484→  if (!metric) return null
   485→
   486→  // Resolve common fields
   487→  const Icon = metric.icon
   488→  const label = metric.label
   489→  const tone = metric.tone
   490→  const rawValue = metric.value(stats as AdminStats)
   491→  const valueStr = typeof rawValue === 'string' ? rawValue : rawValue.toLocaleString('en-IN')
   492→
   493→  const chart = (metric as MetricDef).chart ?? 'none'
   494→  const periodLabel =
   495→    period === '7days' ? '7 Days' : period === '1month' ? '1 Month' : 'All Time'
   496→
   497→  return (
   498→    <Dialog open={!!metric} onOpenChange={(o) => !o && onClose()}>
   499→      <DialogContent className="sm:max-w-lg">
   500→        <DialogHeader>
   501→          <DialogTitle className="flex items-center gap-2">
   502→            <Icon className="h-5 w-5 text-primary" />
   503→            {label}
   504→          </DialogTitle>
   505→          <DialogDescription>
   506→            Period: <span className="font-medium">{periodLabel}</span> · current value:{' '}
   507→            <span className="font-semibold text-foreground">{stats ? valueStr : '—'}</span>
   508→          </DialogDescription>
   509→        </DialogHeader>
   510→
   511→        <div className="h-[280px] w-full">
   512→          {!stats ? (
   513→            <Skeleton className="h-full w-full" />
   514→          ) : chart === 'none' ? (
   515→            <div className="flex h-full flex-col items-center justify-center gap-2 text-muted-foreground">
