   114→// ---------------------------------------------------------------------------
   115→// Safe customizer hook (works with or without the CustomizerProvider).
   116→// ---------------------------------------------------------------------------
   117→
   118→function useCustomizerSafe(preferences: any) {
   119→  try {
   120→    const ctx = useCustomizer()
   121→    return ctx
   122→  } catch {
   123→    // CustomizerProvider not available — derive from preferences prop.
   124→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   125→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   126→    return {
   127→      preferences,
   128→      wallpaper,
   129→      fontFamily,
   130→      messageStyle: preferences?.messageStyle || 'bubble',
   131→      fontSize: preferences?.fontSize || 14,
   132→      setPreference: async () => {},
   133→    }
   134→  }
   135→}

   565→  const { connected } = useSocket({
