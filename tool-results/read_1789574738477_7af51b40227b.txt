    90→// ---------------------------------------------------------------------------
    91→// Safe customizer hook (works with or without the CustomizerProvider).
    92→// ---------------------------------------------------------------------------
    93→
    94→function useCustomizerSafe(preferences: any) {
    95→  try {
    96→    const ctx = useCustomizer()
    97→    return ctx
    98→  } catch {
    99→    // CustomizerProvider not available — derive from preferences prop.
   100→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   101→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   102→    return {
   103→      preferences,
   104→      wallpaper,
   105→      fontFamily,
   106→      messageStyle: preferences?.messageStyle || 'bubble',
   107→      fontSize: preferences?.fontSize || 14,
   108→      setPreference: async () => {},
   109→    }
   110→  }
   111→}

   578→      toast({ title: e.message || 'Failed to send', variant: 'destructive' })

   580→      setSending(false)
   581→    }
   582→  }

   584→  const handleSendSticker = async (sticker: string) => {

   586→      const res: any = await apiFetch('/api/messages', {
   587→        method: 'POST',
   588→        body: JSON.stringify({
