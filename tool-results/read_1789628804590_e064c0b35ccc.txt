    78→// Bubble style class helper
    79→function bubbleStyleClass(style: string, isMine: boolean): string {
    80→  switch (style) {
    81→    case 'sharp':
    82→      return 'bubble-sharp rounded-none'
    83→    case 'tail':
    84→      return isMine ? 'bubble-tail-sent' : 'bubble-tail-received'
    85→    case 'none':
    86→      return 'bubble-none'
    87→    default:
    88→      return '' // default bubble shape
    89→  }
    90→}

    92→// V13 — Determine readable text color (white/black) for a given hex bubble background.
    93→// Used when applying per-conversation sent/received bubble color overrides.
    94→function readableTextOn(hex: string | null): string {
    95→  if (!hex) return '#ffffff'
    96→  const m = /^#([0-9a-fA-F]{6})$/.exec(hex.trim())
    97→  if (!m) return '#ffffff'
    98→  const r = parseInt(m[1].slice(0, 2), 16)
    99→  const g = parseInt(m[1].slice(2, 4), 16)
   100→  const b = parseInt(m[1].slice(4, 6), 16)
   101→  // YIQ contrast formula — light backgrounds get dark text, dark backgrounds get white.
   102→  const yiq = (r * 299 + g * 587 + b * 114) / 1000
   103→  return yiq >= 140 ? '#111111' : '#ffffff'
   104→}

   580→                  />

   590→            {aggregatedReactions.length > 0 && (

   624→            >

   644→        {/* Image lightbox */}
   645→        {message.type === 'image' && message.mediaUrl && (

   674→interface MessageActionMenuProps {

   689→  onForward?: () => void
   690→  /** R1-3 — onSchedule is kept on the interface for backward-compat with

   696→export function MessageActionMenu({
   697→  open,
   698→  message,
   699→  isMine,
   700→  anchorRect,
   701→  onClose,
   702→  onReply,
   703→  onCopy,
   704→  onForward,
   705→  onEdit,
   706→  onDelete,
   707→  onPin,
   708→  onReact,
   709→  onSchedule,
   710→}: MessageActionMenuProps) {
   711→  // R1-3 — onForward / onSchedule are no longer rendered in this menu,
   712→  // but kept on the interface for backward-compat. Mark them as used so
   713→  // eslint doesn't complain.
   714→  void onForward
   715→  void onSchedule
   716→  // Backdrop + a popover positioned above the anchor.
   717→  const [showExpandedReactions, setShowExpandedReactions] = React.useState(false)
   718→  if (!open || !message) return null
   719→
   720→  // Compute position (default: above anchor, centered). Fall back to top center.
   721→  const style: React.CSSProperties = (() => {
   722→    if (anchorRect) {
   723→      const top = Math.max(8, anchorRect.top - 64)
   724→      const left = Math.min(
   725→        Math.max(8, anchorRect.left + anchorRect.width / 2 - 150),
   726→        typeof window !== 'undefined' ? window.innerWidth - 308 : 1000
   727→      )
   728→      return { top, left }
   729→    }
   730→    return { top: 80, left: 8 }
   731→  })()
   732→
   733→  return (
   734→    <div
   735→      className="fixed inset-0 z-50"
   736→      onClick={onClose}
   737→      onContextMenu={(e) => {
   738→        e.preventDefault()
   739→        onClose()
   740→      }}
   741→    >
   742→      <div
   743→        className="absolute"
   744→        style={style}
   745→        onClick={(e) => e.stopPropagation()}
   746→      >
   747→        {/* Reaction emoji row */}
   748→        <div className="mb-1 flex items-center gap-1 rounded-full border bg-popover px-2 py-1 shadow-md">
   749→          {QUICK_REACTIONS.map((emoji) => (
   750→            <button
   751→              key={emoji}
   752→              type="button"
   753→              onClick={() => {
   754→                onReact(emoji)
   755→                onClose()
   756→              }}
   757→              aria-label={`React with ${emoji}`}
   758→              className="flex h-9 w-9 min-h-[44px] min-w-[44px] items-center justify-center rounded-full text-xl transition-transform hover:scale-125"
   759→            >
   760→              {emoji}
   761→            </button>
   762→          ))}
   763→          {/* Expand button for more reactions */}
   764→          <button
   765→            type="button"
   766→            onClick={(e) => {
   767→              e.stopPropagation()
   768→              setShowExpandedReactions((v) => !v)
   769→            }}
   770→            aria-label="More reactions"
   771→            aria-expanded={showExpandedReactions}
   772→            className="flex h-9 w-9 min-h-[44px] min-w-[44px] items-center justify-center rounded-full text-muted-foreground transition-colors hover:bg-accent"
   773→          >
   774→            <SmilePlus className="h-5 w-5" />
   775→          </button>
   776→        </div>
   777→
   778→        {/* Expanded reactions grid */}
   779→        {showExpandedReactions && (
   780→          <div className="mb-1 grid w-[280px] max-w-[calc(100vw-16px)] grid-cols-8 gap-1 rounded-md border bg-popover p-2 shadow-md">
   781→            {EXTENDED_REACTIONS.map((emoji) => (
   782→              <button
   783→                key={emoji}
   784→                type="button"
   785→                onClick={() => {
   786→                  onReact(emoji)
   787→                  onClose()
   788→                }}
   789→                aria-label={`React with ${emoji}`}
   790→                className="flex h-8 w-8 items-center justify-center rounded-full text-lg transition-transform hover:scale-125 hover:bg-accent"
   791→              >
   792→                {emoji}
   793→              </button>
   794→            ))}
   795→          </div>
   796→        )}
   797→
   798→        {/* Action items */}
   799→        <div className="flex w-[300px] max-w-[calc(100vw-16px)] flex-col gap-0.5 rounded-md border bg-popover p-1 shadow-md">
   800→          <ActionMenuItem icon={<Reply className="h-4 w-4" />} label="Reply" onClick={() => { onReply(); onClose() }} />
   801→          <ActionMenuItem icon={<Copy className="h-4 w-4" />} label="Copy" onClick={() => { onCopy(); onClose() }} />
   802→          {isMine && message.type === 'text' && !message.deletedAt && (
   803→            <ActionMenuItem icon={<Pencil className="h-4 w-4" />} label="Edit" onClick={() => { onEdit(); onClose() }} />
   804→          )}
   805→          {isMine && !message.deletedAt && (

   813→          <ActionMenuItem icon={<Pin className="h-4 w-4" />} label={message.pinnedAt ? 'Unpin' : 'Pin'} onClick={() => { onPin(); onClose() }} />
   814→        </div>
   815→      </div>
   816→    </div>
