   100→// ---------------------------------------------------------------------------
   101→// Safe customizer hook (works with or without the CustomizerProvider).
   102→// ---------------------------------------------------------------------------
   103→
   104→function useCustomizerSafe(preferences: any) {
   105→  try {
   106→    const ctx = useCustomizer()
   107→    return ctx
   108→  } catch {
   109→    // CustomizerProvider not available — derive from preferences prop.
   110→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   111→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   112→    return {
   113→      preferences,
   114→      wallpaper,
   115→      fontFamily,
   116→      messageStyle: preferences?.messageStyle || 'bubble',
   117→      fontSize: preferences?.fontSize || 14,
   118→      setPreference: async () => {},
   119→    }
   120→  }
   121→}

   589→      }
   590→    } catch (e: any) {
   591→      toast({ title: e.message || 'Failed to send', variant: 'destructive' })
   592→    } finally {
   593→      setSending(false)
   594→    }
   595→  }

  1756→  const [profile, setProfile] = React.useState<any>(null)
  1757→  const [group, setGroup] = React.useState<any>(null)
  1758→
  1759→  // Reset + fetch whenever the dialog opens.

  1761→    if (!open) return

  1763→    setGroup(null)

  1767→      try {
  1768→        if (isGroup) {
  1769→          if (!groupId) return
  1770→          const res: any = await apiFetch(`/api/groups/${groupId}`)
  1771→          const g = res?.group || (res?.id ? res : null)
  1772→          if (!cancelled) setGroup(g)

  1774→          if (!otherUserId) return

  1783→      }

  1786→      cancelled = true
  1787→    }

  1789→
