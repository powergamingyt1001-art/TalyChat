   235→  onOpenChat?: (c: {
   236→    conversationId: string

   243→// ============================================================
   244→// R8-11 — CapsuleCategorySlider
   245→// Horizontal scrolling capsule / pill chips for the Discover category
   246→// filter. Each capsule is a rounded-full button with scroll-snap-align
   247→// start so the slider snaps one capsule at a time. The active capsule
   248→// uses the brand emerald gradient + white text; inactive capsules are
   249→// subtle tinted pills. The whole row is horizontally scrollable with
   250→// hidden scrollbar (.no-scrollbar) so it stays clean on mobile.
   251→// ============================================================
   252→function CapsuleCategorySlider({
   253→  categories,
   254→  activeCategory,
   255→  onSelect,
   256→}: {
   257→  categories: string[]
   258→  activeCategory: string | null
   259→  onSelect: (cat: string) => void
   260→}) {
   261→  return (
   262→    <div
   263→      className="no-scrollbar scroll-pan-y mt-3 w-full overflow-x-auto pb-1"
   264→      style={{
   265→        scrollSnapType: 'x mandatory',
   266→        scrollPaddingLeft: '8px',
   267→        scrollPaddingRight: '8px',
   268→      }}
   269→    >
   270→      <div className="flex min-w-0 gap-1.5 pb-1">
   271→        {categories.map((cat) => {
   272→          const active = activeCategory === cat
   273→          return (
   274→            <button
   275→              key={cat}
   276→              type="button"
   277→              onClick={() => onSelect(cat)}
   278→              aria-pressed={active}
   279→              className={
   280→                'min-h-[36px] shrink-0 snap-start rounded-full border px-3 py-1.5 text-xs font-medium transition-colors ' +
   281→                (active
   282→                  ? 'btn-brand border-transparent text-primary-foreground shadow-sm'
   283→                  : 'border-border bg-card text-foreground/80 hover:bg-accent hover:text-foreground')
   284→              }
   285→            >
   286→              {cat}
   287→            </button>
   288→          )
   289→        })}
   290→      </div>
   291→    </div>
   292→  )
   293→}
   294→
   295→export function DiscoverScreen({ onOpenChat }: DiscoverScreenProps = {}) {
   296→  const { toast } = useToast()
   297→  const [trending, setTrending] = useState<GroupItem[]>([])
   298→  const [popular, setPopular] = useState<GroupItem[]>([])
   299→  const [newGroups, setNewGroups] = useState<GroupItem[]>([])
   300→  const [sponsored, setSponsored] = useState<any[]>([])
   301→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   302→  const [categoryResults, setCategoryResults] = useState<GroupItem[]>([])
   303→  const [loading, setLoading] = useState(true)
   304→  const [catLoading, setCatLoading] = useState(false)
   305→  const [joining, setJoining] = useState<string | null>(null)
   306→  // V12 — Track the group currently being "left" (used to disable the Leave
   307→  // button + show a spinner during the leave API call).
   308→  const [leaving, setLeaving] = useState<string | null>(null)
   309→  // joinedIds = public joined; requestedIds = private groups with sent requests
   310→  const [joinedIds, setJoinedIds] = useState<Set<string>>(new Set())
   311→  const [requestedIds, setRequestedIds] = useState<Set<string>>(new Set())
   312→  const [preview, setPreview] = useState<GroupItem | null>(null)
   313→
   314→  useEffect(() => {
   315→    let cancelled = false
   316→    ;(async () => {
   317→      setLoading(true)
   318→      try {
   319→        const [t, p, n] = await Promise.all([
   320→          apiFetch('/api/discover?sort=trending').catch(() => null),
   321→          apiFetch('/api/discover?sort=popular').catch(() => null),
   322→          apiFetch('/api/discover?sort=new').catch(() => null),
   323→        ])
   324→        if (cancelled) return
   325→        // Defensive: API returns { groups: [...], sponsored: [...] }.
   326→        // Fall back to bare arrays if the shape is ever flat.
   327→        const tGroups: GroupItem[] = (Array.isArray(t)
   328→          ? t
   329→          : t?.groups || t?.items || []) as GroupItem[]
   330→        const pGroups: GroupItem[] = (Array.isArray(p)
   331→          ? p
   332→          : p?.groups || p?.items || []) as GroupItem[]
   333→        const nGroups: GroupItem[] = (Array.isArray(n)
   334→          ? n
   335→          : n?.groups || n?.items || []) as GroupItem[]
   336→        setTrending(tGroups)
   337→        setPopular(pGroups)
   338→        setNewGroups(nGroups)
   339→        // Sponsored ads: prefer the trending response's sponsored list,
   340→        // fall back to popular's, then to an empty array.
   341→        const sponsoredList: any[] = Array.isArray(t?.sponsored)
   342→          ? t.sponsored
   343→          : Array.isArray(p?.sponsored)
   344→            ? p.sponsored
   345→            : Array.isArray(n?.sponsored)
   346→              ? n.sponsored
   347→              : []
   348→        setSponsored(sponsoredList)
   349→        const joined = new Set<string>([
   350→          ...tGroups.filter((g) => g.isJoined).map((g) => g.id),
   351→          ...pGroups.filter((g) => g.isJoined).map((g) => g.id),
   352→          ...nGroups.filter((g) => g.isJoined).map((g) => g.id),
   353→        ])
   354→        setJoinedIds(joined)
   355→      } finally {
   356→        if (!cancelled) setLoading(false)
   357→      }
   358→    })()
   359→    return () => {
   360→      cancelled = true
   361→    }
   362→  }, [])
   363→
   364→  // Load the user's sent group join requests once on mount so we can mark
   365→  // private groups as "requested".
   366→  useEffect(() => {
   367→    ;(async () => {
   368→      try {
   369→        const res: any = await apiFetch('/api/groups/requests/sent')
   370→        const list: any[] = Array.isArray(res) ? res : (res?.requests || res?.items || [])
   371→        const pending = new Set<string>(
   372→          list
   373→            .filter((r: any) => r.status === 'pending' && r.group?.id)
   374→            .map((r: any) => r.group.id),
   375→        )
   376→        setRequestedIds(pending)
   377→      } catch {
   378→        // ignore — feature best-effort
   379→      }
   380→    })()
   381→  }, [])
   382→
   383→  const fetchCategory = async (cat: string) => {
   384→    setCatLoading(true)
   385→    try {
   386→      const res: any = await apiFetch(
   387→        `/api/discover?category=${encodeURIComponent(cat)}&sort=trending`,
   388→      )
   389→      // Defensive: API returns { groups: [...] }, fall back to bare array.
   390→      const list: GroupItem[] = (Array.isArray(res)
   391→        ? res
   392→        : res?.groups || res?.items || []) as GroupItem[]
   393→      setCategoryResults(list)
   394→    } catch (e: any) {
   395→      toast({ title: e?.message || 'Failed to load category', variant: 'destructive' })
   396→    } finally {
   397→      setCatLoading(false)
   398→    }
   399→  }
   400→
   401→  const handleCategoryClick = (cat: string) => {
   402→    if (activeCategory === cat) {
   403→      setActiveCategory(null)
   404→      setCategoryResults([])
   405→    } else {
   406→      setActiveCategory(cat)
   407→      fetchCategory(cat)
   408→    }
   409→  }
   410→
   411→  // Returns the join state for a group, factoring in joined + requested sets.
   412→  const joinStateFor = (g: GroupItem): JoinState => {
   413→    if (joinedIds.has(g.id)) return 'joined'
   414→    if (!g.isPublic && requestedIds.has(g.id)) return 'requested'
   415→    return 'idle'
   416→  }
   417→
   418→  const handleJoin = async (g: GroupItem) => {
   419→    setJoining(g.id)
   420→    try {
   421→      const res: any = await apiFetch('/api/groups/join', {
   422→        method: 'POST',
   423→        body: JSON.stringify({ groupId: g.id }),
   424→      })
   425→      if (res?.alreadyMember) {
   426→        toast({ title: 'You are already a member' })
   427→        setJoinedIds((prev) => new Set(prev).add(g.id))
   428→      } else if (res?.requested) {
   429→        // Private group → request was sent
   430→        toast({
   431→          title: 'Join request sent! Wait for admin approval.',
   432→        })
   433→        setRequestedIds((prev) => new Set(prev).add(g.id))
   434→      } else {
   435→        // Public group → direct join
   436→        toast({ title: `Joined ${g.name} 🎉` })
   437→        setJoinedIds((prev) => new Set(prev).add(g.id))
   438→      }
   439→    } catch (e: any) {
   440→      toast({ title: e?.message || 'Failed to join', variant: 'destructive' })
   441→    } finally {
   442→      setJoining(null)
   443→    }
   444→  }
   445→
   446→  // V12 — Open the chat for a joined group. Calls onOpenChat (wired from
   447→  // TalyApp) when a conversationId is linked to the group. Falls back to
   448→  // a friendly toast if the callback is missing (defensive — should never
   449→  // happen in production wiring, but avoids a no-op click).
   450→  const handleOpen = (g: GroupItem) => {
   451→    if (!g.conversationId) {
   452→      toast({
   453→        title: 'No chat linked to this group yet',
   454→        description: 'Please try again in a moment.',
   455→        variant: 'destructive',
   456→      })
   457→      return
   458→    }
   459→    if (onOpenChat) {
   460→      onOpenChat({
   461→        conversationId: g.conversationId,
   462→        name: g.name,
   463→        avatar: g.logo || undefined,
   464→        isGroup: true,
   465→      })
   466→    } else {
   467→      toast({ title: 'Open coming soon' })
   468→    }
   469→  }
   470→
   471→  // V12 — Leave a joined group. Calls POST /api/groups/[id]/leave, then
   472→  // removes the group id from joinedIds so the button reverts to Join.
   473→  const handleLeave = async (g: GroupItem) => {
   474→    setLeaving(g.id)
   475→    try {
   476→      await apiFetch(`/api/groups/${g.id}/leave`, { method: 'POST' })
   477→      toast({ title: `Left ${g.name}` })
   478→      setJoinedIds((prev) => {
   479→        const next = new Set(prev)
   480→        next.delete(g.id)
   481→        return next
   482→      })
   483→    } catch (e: any) {
   484→      toast({ title: e?.message || 'Failed to leave group', variant: 'destructive' })
   485→    } finally {
   486→      setLeaving(null)
   487→    }
   488→  }
   489→
   490→  // V12 — Reset the active category filter (used by the "Browse Trending
   491→  // instead" link in the empty state and by the active-filter banner's
   492→  // clear (✕) button).
   493→  const clearCategory = () => {
   494→    setActiveCategory(null)
   495→    setCategoryResults([])
   496→  }
   497→
   498→  const totalGroups = trending.length + popular.length + newGroups.length
   499→
   500→  return (
   501→    <div className="w-full px-2 pb-20 pt-4 lg:pb-6 lg:pt-6">
   502→      {/* Header */}
   503→      <motion.div
   504→        initial={{ opacity: 0, y: -8 }}
   505→        animate={{ opacity: 1, y: 0 }}
   506→        transition={{ duration: 0.3 }}
   507→      >
   508→        <h1 className="section-header">Discover</h1>
   509→        <p className="mt-1 text-sm text-muted-foreground">
   510→          Find communities that match your interests
   511→        </p>
   512→      </motion.div>
   513→
   514→      {/* R8-11 — Capsule slider: horizontal scrolling capsule/pill chips for
   515→          the new category list (AI, Game, Fun, …). Each capsule is a pill
   516→          with rounded-full + scroll-snap-align start so the slider snaps
   517→          one capsule at a time. The active capsule uses the brand gradient
   518→          (btn-brand look) so it stands out from the rest. */}
   519→      <CapsuleCategorySlider
   520→        categories={CAPSULE_CATEGORIES as unknown as string[]}
   521→        activeCategory={activeCategory}
   522→        onSelect={handleCategoryClick}
   523→      />
   524→
   525→      {/* V12 — Active filter banner: shows below the chips row when a
   526→          category filter is selected, with a clear (✕) button to reset. */}
   527→      {activeCategory && (
   528→        <div className="mt-3 flex items-center justify-between gap-2 rounded-lg bg-emerald-50 px-3 py-2 text-xs font-medium text-emerald-700 dark:bg-emerald-950/20 dark:text-emerald-300">
   529→          <span className="truncate">
   530→            Showing results for{' '}
   531→            <span className="font-bold">&lsquo;{activeCategory}&rsquo;</span>
   532→          </span>
   533→          <button
   534→            type="button"
   535→            onClick={clearCategory}
   536→            aria-label="Clear category filter"
   537→            className="inline-flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-emerald-700 transition-colors hover:bg-emerald-500/15 dark:text-emerald-300"
   538→          >
   539→            <X className="h-3.5 w-3.5" />
   540→          </button>
   541→        </div>
   542→      )}
   543→
   544→      {/* Empty state — no groups loaded */}
   545→      {!loading && totalGroups === 0 && !activeCategory && (
   546→        <div className="dotted-bg mt-6 flex flex-col items-center justify-center rounded-2xl border border-border bg-card p-8 text-center">
   547→          <Compass className="h-12 w-12 text-muted-foreground/60" />
   548→          <p className="mt-3 text-sm font-semibold text-foreground">
   549→            No communities yet
   550→          </p>
   551→          <p className="mt-1 text-xs text-muted-foreground">
   552→            Check back soon — new communities are added every day.
   553→          </p>
   554→        </div>
   555→      )}
   556→
   557→      {/* Category results — when a capsule is selected, show every group
   558→          in that category as a vertical list (no slider). */}
   559→      {activeCategory && (
   560→        <section className="mt-4 animate-fade-in-up">
   561→          <h2 className="section-header">
   562→            {activeCategory} communities
   563→          </h2>
   564→          <div className="mt-3">
   565→            {catLoading ? (
   566→              <InlineLoadingRow />
   567→            ) : categoryResults.length === 0 ? (
   568→              // Friendly no-results empty state: magnifying glass + clear
   569→              // filter CTA. The old "Create {Category} Group" button has
   570→              // been removed because group creation is no longer
   571→              // available from the Groups tab (R8-11).
   572→              <div className="flex flex-col items-center justify-center rounded-2xl border border-dashed border-border bg-card p-8 text-center">
   573→                <Search className="h-12 w-12 text-muted-foreground/40" />
   574→                <p className="mt-3 text-sm font-semibold text-foreground">
   575→                  No groups found in {activeCategory}
   576→                </p>

   999→// MiniStackedAvatars — 2-3 small overlapping letter-avatars with +N.
  1000→// Used inside group cards/rows to humanize the member count display.
  1001→// ============================================================
  1002→function MiniStackedAvatars({
  1003→  logo,
  1004→  name,
  1005→  memberCount,
  1006→  small = false,
  1007→}: {
  1008→  logo?: string | null
  1009→  name: string
  1010→  memberCount: number
  1011→  small?: boolean
  1012→}) {
  1013→  const baseName = name || 'G'
  1014→  const initials = (baseName.replace(/\s+/g, ' ').trim().split(' ').filter(Boolean))
  1015→  const derived: string[] = []
  1016→  for (let i = 0; i < initials.length && derived.length < 3; i++) {
  1017→    derived.push(initials[i][0]?.toUpperCase() || 'U')
  1018→  }
  1019→  while (derived.length < 3) {
  1020→    derived.push(baseName[0]?.toUpperCase() || 'U')
  1021→  }
  1022→  const visible = derived.slice(0, 3)
  1023→  const extra = Math.max(0, memberCount - 3)
  1024→  const sizeClass = small ? 'h-4 w-4' : 'h-5 w-5'
  1025→  const textSize = small ? 'text-[7px]' : 'text-[8px]'
  1026→  const offset = small ? 8 : 10
  1027→  const slotClass = (i: number) => {
  1028→    switch (i % 3) {
  1029→      case 0:
  1030→        return 'bg-emerald-500/15 text-emerald-600'
  1031→      case 1:
  1032→        return 'bg-amber-500/15 text-amber-600'
  1033→      default:
  1034→        return 'bg-sky-500/15 text-sky-600'
  1035→    }
  1036→  }
  1037→  return (
  1038→    <span className="relative inline-flex items-center" aria-hidden>
  1039→      <span className="relative inline-flex" style={{ minWidth: `${offset * 2 + 8}px` }}>
  1040→        {visible.map((letter, i) => {
  1041→          const z = visible.length - i
  1042→          return (
  1043→            <span
  1044→              key={i}
  1045→              className={`absolute flex ${sizeClass} items-center justify-center rounded-full ${textSize} font-bold ring-1 ring-card ${slotClass(i)}`}
  1046→              style={{ left: `${i * offset}px`, zIndex: z }}
  1047→            >
  1048→              {i === 0 && logo ? (
  1049→                <img
  1050→                  src={logo}
  1051→                  alt=""
  1052→                  className="h-full w-full rounded-full object-cover"
  1053→                />
  1054→              ) : (
  1055→                letter
  1056→              )}
  1057→            </span>
  1058→          )
  1059→        })}
  1060→        {extra > 0 && (
  1061→          <span
  1062→            className={`absolute flex ${sizeClass} items-center justify-center rounded-full bg-muted ${textSize} font-bold text-muted-foreground ring-1 ring-card`}
  1063→            style={{ left: `${visible.length * offset}px`, zIndex: visible.length + 1 }}
  1064→          >
  1065→            +{extra > 99 ? '99' : extra}
  1066→          </span>
  1067→        )}
  1068→      </span>
  1069→    </span>
  1070→  )
  1071→}
