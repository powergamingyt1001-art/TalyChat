   113→   113→// ---------------------------------------------------------------------------
   114→   114→// Safe customizer hook (works with or without the CustomizerProvider).
   115→   115→// ---------------------------------------------------------------------------
   116→   116→
   117→   117→function useCustomizerSafe(preferences: any) {
   118→   118→  try {
   119→   119→    const ctx = useCustomizer()
   120→   120→    return ctx
   121→   121→  } catch {
   122→   122→    // CustomizerProvider not available — derive from preferences prop.
   123→   123→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   124→   124→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   125→   125→    return {
   126→   126→      preferences,
   127→   127→      wallpaper,
   128→   128→      fontFamily,
   129→   129→      messageStyle: preferences?.messageStyle || 'bubble',
   130→   130→      fontSize: preferences?.fontSize || 14,
   131→   131→      setPreference: async () => {},
   132→   132→    }
   133→   133→  }
   134→   134→}
