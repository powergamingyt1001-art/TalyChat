    91→    91→// Safe customizer hook (works with or without the CustomizerProvider).
    92→    92→// ---------------------------------------------------------------------------
    93→    93→
    94→    94→function useCustomizerSafe(preferences: any) {
    95→    95→  try {
    96→    96→    const ctx = useCustomizer()
    97→    97→    return ctx
    98→    98→  } catch {
    99→    99→    // CustomizerProvider not available — derive from preferences prop.
   100→   100→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   101→   101→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   102→   102→    return {
   103→   103→      preferences,
   104→   104→      wallpaper,
   105→   105→      fontFamily,
   106→   106→      messageStyle: preferences?.messageStyle || 'bubble',
   107→   107→      fontSize: preferences?.fontSize || 14,
   108→   108→      setPreference: async () => {},
   109→   109→    }
   110→   110→  }
   111→   111→}

   486→   486→  const scheduleNextAd = React.useCallback(() => {
