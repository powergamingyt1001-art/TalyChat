   416→function PremiumSection({
   417→  premiumActive,
   418→  premiumUntil,
   419→  onChoosePlan,
   420→}: {
   421→  premiumActive: boolean
   422→  premiumUntil: string | null
   423→  onChoosePlan: (plan: { id: string; label: string; price: number; months: number }) => void
   424→}) {
   425→  const [offerMsLeft, setOfferMsLeft] = useState<number | null>(null)
   426→  const [offerPrice, setOfferPrice] = useState(OFFER_PRICE)
   427→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   428→
   429→  useEffect(() => {
   430→    const compute = () => {
   431→      const now = Date.now()
   432→      let start = Number(localStorage.getItem(OFFER_KEY) || 0)
   433→      if (!start || now - start > OFFER_LOOP_MS) {
   434→        // restart offer cycle
   435→        start = now
   436→        localStorage.setItem(OFFER_KEY, String(start))
   437→      }
   438→      const elapsed = now - start
   439→      if (elapsed < OFFER_WINDOW_MS) {
   440→        setOfferMsLeft(OFFER_WINDOW_MS - elapsed)
   441→        setOfferPrice(OFFER_PRICE)
   442→      } else {
   443→        setOfferMsLeft(0)
   444→        setOfferPrice(REGULAR_PRICE)
   445→      }
   446→    }
   447→    compute()
   448→    timerRef.current = setInterval(compute, 1000)
   449→    return () => {
   450→      if (timerRef.current) clearInterval(timerRef.current)
   451→    }
   452→  }, [])
   453→
   454→  if (premiumActive) {
   455→    return (
   456→      <Card className="p-4">
   457→        <div className="flex items-center gap-3">
   458→          <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-amber-400 to-amber-500">
   459→            <Crown className="h-6 w-6 text-white" />
   460→          </div>
   461→          <div className="min-w-0 flex-1">
   462→            <p className="flex items-center gap-2 text-base font-semibold">
   463→              Premium active
   464→              <span className="premium-badge">
   465→                <Crown className="h-3 w-3" /> Pro
   466→              </span>
   467→            </p>
   468→            <p className="text-sm text-muted-foreground">
   469→              {premiumUntil
   470→                ? `Valid until ${format(new Date(premiumUntil), 'dd MMM yyyy')}`
   471→                : 'Lifetime'}
   472→            </p>
   473→          </div>
   474→        </div>
   475→      </Card>
   476→    )
   477→  }
   478→
   479→  const plans = [
   480→    { id: '2mo', label: '2 Months', price: 49, months: 2, best: false },
   481→    { id: '6mo', label: '6 Months', price: 99, months: 6, best: false },
   482→    { id: '1yr', label: '1 Year', price: offerPrice, months: 12, best: true },
   483→  ]
   484→
   485→  return (
   486→    <Card className="p-4">
   487→      <div className="mb-3 flex items-center justify-between gap-2">
   488→        <h3 className="flex items-center gap-2 text-base font-semibold">
   489→          <Sparkles className="h-4 w-4 text-primary" /> Upgrade to Premium
   490→        </h3>
   491→        {offerMsLeft && offerMsLeft > 0 && (
   492→          <span className="inline-flex items-center gap-1 rounded-full bg-primary/10 px-2 py-1 text-xs font-semibold text-primary">
   493→            <Clock className="h-3 w-3" />
   494→            {formatCountdown(offerMsLeft)}
   495→          </span>
   496→        )}
   497→      </div>
   498→      {offerMsLeft && offerMsLeft > 0 ? (
   499→        <p className="mb-3 text-xs text-primary">
   500→          Limited-time offer! 1-year plan for ₹{OFFER_PRICE} (was ₹{REGULAR_PRICE})
   501→        </p>
   502→      ) : (
   503→        <p className="mb-3 text-xs text-muted-foreground">
   504→          Offer ended — 1-year plan is back to ₹{REGULAR_PRICE}
   505→        </p>
   506→      )}
   507→      <div className="grid gap-2 sm:grid-cols-3">
   508→        {plans.map((p) => (
   509→          <div
   510→            key={p.id}
   511→            className={`relative flex flex-col gap-1 rounded-xl border-2 p-3 ${
   512→              p.best ? 'border-primary bg-primary/5' : 'border-border'
   513→            }`}
   514→          >
   515→            {p.best && (
   516→              <span className="absolute -top-2 left-1/2 -translate-x-1/2 rounded-full bg-primary px-2 py-0.5 text-[10px] font-bold text-primary-foreground">
   517→                Best Value
   518→              </span>
   519→            )}
   520→            <p className="text-xs font-medium text-muted-foreground">{p.label}</p>
   521→            <p className="text-2xl font-bold text-primary">₹{p.price}</p>
   522→            <p className="text-[10px] text-muted-foreground">
   523→              ₹{Math.round((p.price / p.months) * 100) / 100}/mo
   524→            </p>
   525→            <Button
   526→              size="sm"
   527→              variant={p.best ? 'default' : 'outline'}
   528→              className={`mt-2 min-h-[36px] ${p.best ? 'btn-brand' : ''}`}
   529→              onClick={() => onChoosePlan(p)}
   530→            >
   531→              Choose
   532→            </Button>
   533→          </div>
   534→        ))}
   535→      </div>
   536→    </Card>
   537→  )
   538→}
   539→
   540→function formatCountdown(ms: number): string {
   541→  const totalSec = Math.max(0, Math.floor(ms / 1000))
   542→  const m = Math.floor(totalSec / 60)
   543→  const s = totalSec % 60
   544→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   545→}

   602→      toast({ title: 'Username is required', variant: 'destructive' })

   604→    }

   609→        username: username.trim(),

   611→        gender,
   612→        avatar: avatar || '',
   613→        dob: dob ? format(dob, 'yyyy-MM-dd') : '',

   621→      onClose()
   622→      void res
   623→    } catch (e: any) {
   624→      toast({ title: e.message, variant: 'destructive' })

   626→      setSaving(false)
   627→    }
