    92→// Apply a theme on the document root (light/dark/system)
    93→function applyTheme(theme: string) {
    94→  if (typeof document === 'undefined') return
    95→  const root = document.documentElement
    96→  const isDark =
    97→    theme === 'dark' ||
    98→    (theme === 'system' &&
    99→      typeof window !== 'undefined' &&
   100→      window.matchMedia('(prefers-color-scheme: dark)').matches)
   101→  root.classList.toggle('dark', isDark)
   102→}

   104→// Bubble preview style (for message-style cards and live preview)
   105→function previewBubbleClass(style: string, isMine: boolean): string {
   106→  switch (style) {
   107→    case 'sharp':
   108→      return 'rounded-none'
   109→    case 'tail':
   110→      return isMine
   111→        ? 'rounded-2xl rounded-br-md'
   112→        : 'rounded-2xl rounded-bl-md'
   113→    case 'none':
   114→      return 'rounded-none bg-transparent border-0 shadow-none'
   115→    default:
   116→      return 'rounded-2xl'
   117→  }
   118→}

   120→// Determine readable text color (white/black) for a given hex bubble background.
   121→function readableTextOn(hex: string | null): string {
   122→  if (!hex) return ''
   123→  const m = /^#([0-9a-fA-F]{6})$/.exec(hex.trim())
   124→  if (!m) return ''
   125→  const r = parseInt(m[1].slice(0, 2), 16)
   126→  const g = parseInt(m[1].slice(2, 4), 16)
   127→  const b = parseInt(m[1].slice(4, 6), 16)
   128→  // YIQ contrast formula
   129→  const yiq = (r * 299 + g * 587 + b * 114) / 1000
   130→  return yiq >= 140 ? '#111111' : '#ffffff'
   131→}

   496→function FontImport({
   497→  onUploaded,
   498→}: {
   499→  onUploaded: (url: string) => void
   500→}) {
   501→  const { toast } = useToast()
   502→  const inputRef = React.useRef<HTMLInputElement | null>(null)
   503→  const [uploading, setUploading] = React.useState(false)
   504→
   505→  const handleFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   506→    const file = e.target.files?.[0]
   507→    if (!file) return
   508→    setUploading(true)
   509→    try {
   510→      const { apiUpload } = await import('@/lib/api')
   511→      const res: any = await apiUpload('/api/upload', file, 'file')
   512→      const url = res?.url
   513→      if (!url) throw new Error('Upload did not return a URL')
   514→      onUploaded(url)
   515→      toast({ title: 'Custom font applied' })
   516→    } catch (err: any) {
   517→      toast({
   518→        title: err?.message || 'Font upload failed',
   519→        variant: 'destructive',
   520→      })
   521→    } finally {
   522→      setUploading(false)
   523→      if (inputRef.current) inputRef.current.value = ''
   524→    }
   525→  }
   526→
   527→  return (
   528→    <div className="flex items-center gap-2">
   529→      <input
   530→        ref={inputRef}
   531→        type="file"
   532→        accept=".ttf,.otf"
   533→        className="hidden"
   534→        onChange={handleFile}
   535→      />
   536→      <Button
   537→        type="button"
   538→        variant="outline"
   539→        onClick={() => inputRef.current?.click()}
   540→        disabled={uploading}
   541→        className="min-h-[40px]"
   542→      >
   543→        {uploading ? (
   544→          <Loader2 className="mr-1 h-4 w-4 animate-spin" />
   545→        ) : (
   546→          <Upload className="mr-1 h-4 w-4" />
   547→        )}
   548→        Upload .ttf / .otf
   549→      </Button>
   550→      <p className="text-xs text-muted-foreground">
   551→        Upload a font file to apply as your custom chat font.
   552→      </p>
   553→    </div>
   554→  )
   555→}

   637→// ---------------------------------------------------------------------------
   638→// Main component
   639→// ---------------------------------------------------------------------------
   640→
   641→export function CustomizeDialog({
   642→  open,
   643→  onClose,
   644→  onOpenChange,
   645→  conversationId,
   646→  isGroup: _isGroup,
   647→  currentThemeColor,
   648→  onApplyThemeColor,
   649→}: CustomizeDialogProps) {
   650→  const { user } = useAuth()
   651→  const { toast } = useToast()
   652→  const customizer = useCustomizerSafe()
   653→
   654→  // Local state mirrors customizer preferences so the dialog can work even
   655→  // when the provider has not been mounted yet.
   656→  const [prefs, setPrefs] = React.useState<any>(customizer?.preferences || null)
   657→  const [saving, setSaving] = React.useState<string | null>(null)
   658→  const [fontImportUrl, setFontImportUrl] = React.useState<string | null>(null)
   659→  const [applyingColor, setApplyingColor] = React.useState(false)
   660→
   661→  // V13 — per-conversation color state (chat background, sent + received bubbles).
   662→  const [localBgColor, setLocalBgColor] = React.useState<string | null>(
   663→    currentThemeColor || null,
   664→  )
   665→
   666→  // Per-conversation bubble colors (stored on the Conversation row, optimistically
   667→  // mirrored locally so the swatch reflects the current state).
   668→  const [localSentColor, setLocalSentColor] = React.useState<string | null>(null)
   669→  const [localReceivedColor, setLocalReceivedColor] = React.useState<string | null>(null)
   670→
   671→  // Sync from customizer context when it changes
   672→  React.useEffect(() => {
   673→    if (customizer?.preferences) setPrefs(customizer.preferences)
   674→  }, [customizer?.preferences])
   675→
   676→  // Sync per-conversation color state when the dialog opens or themeColor changes.
   677→  React.useEffect(() => {
   678→    setLocalBgColor(currentThemeColor || null)
   679→  }, [open, currentThemeColor])
   680→
   681→  // Fetch fresh prefs + per-conversation bubble colors when the dialog opens.
   682→  React.useEffect(() => {
   683→    if (!open) return
   684→    let cancelled = false
   685→    ;(async () => {
   686→      try {
   687→        const res: any = await apiFetch('/api/preferences').catch(() => null)
   688→        const p = res && (res.preferences || res)
   689→        if (!cancelled && p) setPrefs(p)
   690→      } catch {}
   691→      if (conversationId) {
   692→        try {
   693→          const convRes: any = await apiFetch(`/api/conversations/${conversationId}`)
   694→          const conv = convRes?.conversation || convRes
   695→          if (!cancelled) {
   696→            setLocalSentColor(conv?.sentBubbleColor || null)
   697→            setLocalReceivedColor(conv?.receivedBubbleColor || null)
   698→          }
   699→        } catch {}
   700→      }
   701→    })()
   702→    return () => {
   703→      cancelled = true
   704→    }
   705→  }, [open, conversationId])
   706→
   707→  const isPremium = !!(user as any)?.isPremium
   708→
   709→  const handleOpenChange = (o: boolean) => {
   710→    if (o) {
   711→      onOpenChange?.(true)
   712→      return
   713→    }
   714→    // Closing — prefer canonical `onClose`, then `onOpenChange(false)`.
   715→    if (onClose) onClose()
   716→    else onOpenChange?.(false)
   717→  }
   718→
   719→  // Save a single preference key — uses customizer context when available,
   720→  // otherwise falls back to apiFetch directly. Also updates local state.
   721→  const savePref = async (patch: Record<string, any>) => {
   722→    setPrefs((p: any) => ({ ...(p || {}), ...patch }))
   723→    if (customizer) {
   724→      try {
   725→        await customizer.setPreference(patch)
   726→      } catch {}
   727→      return
   728→    }
   729→    setSaving(Object.keys(patch).join(','))
   730→    try {
   731→      await apiFetch('/api/preferences', {
   732→        method: 'PUT',
   733→        body: JSON.stringify(patch),
   734→      })
   735→    } catch (e: any) {
   736→      toast({ title: e?.message || 'Failed to save', variant: 'destructive' })
   737→    } finally {
   738→      setSaving(null)
   739→    }
   740→  }
   741→
   742→  // V13 — Apply per-conversation chat background color (immediately on tap).
   743→  const applyBgColor = async (hex: string | null) => {
   744→    setLocalBgColor(hex)
   745→    onApplyThemeColor?.(hex)
   746→    if (!conversationId) return
   747→    setApplyingColor(true)
   748→    try {
   749→      await apiFetch(`/api/conversations/${conversationId}`, {
   750→        method: 'PATCH',
   751→        body: JSON.stringify({ themeColor: hex }),
   752→      })
   753→    } catch (e: any) {
   754→      const err = e as ApiError
   755→      toast({
   756→        title: err?.message || 'Failed to update chat background',
   757→        variant: 'destructive',
   758→      })
   759→    } finally {
   760→      setApplyingColor(false)
   761→    }
   762→  }
   763→
   764→  // V13 — Apply per-conversation sent bubble color.
   765→  const applySentColor = async (hex: string | null) => {
   766→    setLocalSentColor(hex)
   767→    if (!conversationId) return
   768→    try {
   769→      await apiFetch(`/api/conversations/${conversationId}`, {
   770→        method: 'PATCH',
   771→        body: JSON.stringify({ sentBubbleColor: hex }),
   772→      })
   773→    } catch (e: any) {
   774→      const err = e as ApiError
   775→      toast({
   776→        title: err?.message || 'Failed to update sent bubble color',
   777→        variant: 'destructive',
   778→      })
   779→    }
   780→  }
   781→
   782→  // V13 — Apply per-conversation received bubble color.
   783→  const applyReceivedColor = async (hex: string | null) => {
   784→    setLocalReceivedColor(hex)
   785→    if (!conversationId) return
   786→    try {
   787→      await apiFetch(`/api/conversations/${conversationId}`, {
   788→        method: 'PATCH',
   789→        body: JSON.stringify({ receivedBubbleColor: hex }),
   790→      })
   791→    } catch (e: any) {
   792→      const err = e as ApiError
   793→      toast({
   794→        title: err?.message || 'Failed to update received bubble color',
   795→        variant: 'destructive',
   796→      })
   797→    }
   798→  }
   799→
   800→  // ----- Derived selections -----
   801→  const theme: string = prefs?.theme || 'system'
   802→  const wallpaperId: string = prefs?.wallpaper || 'default'
   803→  const messageStyle: string = prefs?.messageStyle || 'bubble'
   804→  const fontSize: number = Number(prefs?.fontSize) || 14
   805→  const fontFamilyId: string = prefs?.fontFamily || 'sans'
   806→
   807→  const wallpaper =
   808→    WALLPAPERS.find((w) => w.id === wallpaperId) || WALLPAPERS[0]
   809→  const fontFamily =
   810→    FONT_OPTIONS.find((f) => f.id === fontFamilyId) || FONT_OPTIONS[0]
   811→
   812→  const setTheme = (v: string) => {
   813→    applyTheme(v)
   814→    savePref({ theme: v })
   815→  }
   816→
   817→  const setWallpaper = (id: string) => savePref({ wallpaper: id })
   818→  const setMessageStyle = (v: string) => savePref({ messageStyle: v })
   819→  const setFontFamily = (id: string) => savePref({ fontFamily: id })
   820→  const setFontSize = (v: number) => savePref({ fontSize: v })
   821→
   822→  const handleFontImport = async (url: string) => {
   823→    setFontImportUrl(url)
   824→    await savePref({ customFontUrl: url, fontFamily: 'custom' })
   825→  }
   826→
   827→  return (
   828→    <Dialog open={open} onOpenChange={handleOpenChange}>
   829→      <DialogContent className="max-w-2xl gap-0 p-0 sm:max-w-2xl">
   830→        <DialogHeader className="border-b p-4">
   831→          <DialogTitle className="flex items-center gap-2">
   832→            <Palette className="h-4 w-4 text-primary" />
   833→            Customize
   834→            {applyingColor && (
   835→              <Loader2 className="ml-1 h-3.5 w-3.5 animate-spin text-muted-foreground" />
   836→            )}
   837→          </DialogTitle>
   838→          <DialogDescription className="sr-only">
   839→            Personalize your chat with chat background colors, bubble colors,
   840→            and image wallpapers.
   841→          </DialogDescription>
   842→        </DialogHeader>
   843→
   844→        <ScrollArea className="max-h-[78dvh]">
   845→          <div className="space-y-6 p-4">
   846→            {/* Saving indicator */}
   847→            {saving && (
   848→              <div className="flex items-center justify-end gap-1 text-xs text-muted-foreground">
   849→                <Loader2 className="h-3 w-3 animate-spin" /> Saving…
   850→              </div>
   851→            )}
   852→
   853→            {/* ====================================================
   854→                Section 1 (V13): Chat background — 4 color swatches
   855→                Available to ALL users (per-conversation override).
   856→                ==================================================== */}
   857→            <section className="space-y-2">
   858→              <div className="flex items-center justify-between">
   859→                <SectionTitle>Chat Background</SectionTitle>
   860→                <span className="text-[11px] text-muted-foreground">
   861→                  Per-conversation
   862→                </span>
   863→              </div>
   864→              <p className="text-xs text-muted-foreground">
   865→                Pick a color to tint this chat&apos;s background.
   866→              </p>
   867→              <ChatBackgroundPicker
   868→                selected={localBgColor}
   869→                onSelect={applyBgColor}
   870→              />
   871→            </section>
   872→
   873→            {/* ====================================================
   874→                Section 2 (V13): Bubble colors — sent + received
   875→                Available to ALL users (per-conversation override).
   876→                ==================================================== */}
   877→            <section className="space-y-3">
   878→              <div className="flex items-center justify-between">
   879→                <SectionTitle>Message Bubble Colors</SectionTitle>
   880→                <span className="text-[11px] text-muted-foreground">
   881→                  Per-conversation
   882→                </span>
   883→              </div>
   884→              <div className="space-y-4 rounded-lg border border-border bg-muted/20 p-3">
   885→                <BubbleColorPicker
   886→                  label="Sent bubble color"
   887→                  selected={localSentColor}
   888→                  onSelect={applySentColor}
   889→                  isMine
   890→                />
   891→                <BubbleColorPicker
   892→                  label="Received bubble color"
   893→                  selected={localReceivedColor}
   894→                  onSelect={applyReceivedColor}
   895→                  isMine={false}
   896→                />
   897→              </div>
   898→            </section>
   899→
   900→            {/* ====================================================
   901→                Section 3 (V13): Image wallpapers — LOCKED for non-premium.
   902→                Per-task: lock icon overlay + Premium badge per item.
   903→                ==================================================== */}
   904→            <section className="space-y-2">
   905→              <div className="flex items-center justify-between">
   906→                <SectionTitle>Image Wallpapers</SectionTitle>
   907→                {!isPremium && (
   908→                  <span className="inline-flex items-center gap-1 rounded-full bg-amber-500/15 px-2 py-0.5 text-[10px] font-bold uppercase text-amber-600 dark:text-amber-400">
   909→                    <Lock className="h-2.5 w-2.5" /> Premium
   910→                  </span>
   911→                )}
   912→              </div>
   913→              <WallpaperGrid
   914→                selected={wallpaperId}
   915→                onSelect={setWallpaper}
   916→                isPremium={isPremium}
   917→              />
   918→            </section>
   919→
   920→            {/* ====================================================
   921→                Sections 4-8: Premium-only global customizations
   922→                (theme, message style, font size, font family, font import)
   923→                ==================================================== */}
   924→            {!isPremium ? (
   925→              <section className="space-y-2">
   926→                <SectionTitle>Global Themes &amp; Fonts</SectionTitle>
   927→                <PremiumGateOverlay />
   928→              </section>
   929→            ) : (
   930→              <>
   931→                {/* Section 4: Theme */}
   932→                <section className="space-y-2">
   933→                  <SectionTitle>Theme</SectionTitle>
   934→                  <ThemeSegmented value={theme} onChange={setTheme} />
   935→                </section>
   936→
   937→                {/* Section 5: Message Style */}
   938→                <section className="space-y-2">
   939→                  <SectionTitle>Message Style</SectionTitle>
   940→                  <MessageStyleCards
   941→                    selected={messageStyle}
   942→                    onSelect={setMessageStyle}
   943→                    fontSizePx={fontSize}
   944→                    fontFamilyClass={fontFamily.className}
   945→                  />
   946→                </section>
   947→
   948→                {/* Section 6: Font Size */}
   949→                <section className="space-y-2">
   950→                  <div className="flex items-center justify-between">
   951→                    <SectionTitle>Font Size</SectionTitle>
   952→                    <span className="text-xs font-medium tabular-nums text-muted-foreground">
   953→                      {fontSize}px
   954→                    </span>
   955→                  </div>
   956→                  <Slider
   957→                    min={FONT_SIZE_MIN}
   958→                    max={FONT_SIZE_MAX}
   959→                    step={1}
   960→                    value={[fontSize]}
   961→                    onValueChange={(v) => setFontSize(Number(v[0]) || fontSize)}
   962→                  />
   963→                  <div className="flex justify-between text-[10px] text-muted-foreground">
   964→                    <span>{FONT_SIZE_MIN}px</span>
   965→                    <span>{FONT_SIZE_MAX}px</span>
   966→                  </div>
   967→                </section>
   968→
   969→                {/* Section 7: Font Family */}
   970→                <section className="space-y-2">
   971→                  <SectionTitle>Font Family</SectionTitle>
   972→                  <FontList selected={fontFamilyId} onSelect={setFontFamily} />
   973→                  {fontImportUrl && (
   974→                    <p className="text-xs text-primary">
   975→                      Custom font applied: <code>{fontImportUrl}</code>
   976→                    </p>
   977→                  )}
   978→                </section>
   979→
   980→                {/* Section 8: Font Import */}
   981→                <section className="space-y-2">
   982→                  <SectionTitle>Import Custom Font</SectionTitle>
   983→                  <FontImport onUploaded={handleFontImport} />
   984→                </section>
   985→
   986→                {/* Live Preview */}
   987→                <section className="space-y-2">
   988→                  <SectionTitle>Live Preview</SectionTitle>
   989→                  <LivePreview
   990→                    wallpaper={wallpaper}
   991→                    messageStyle={messageStyle}
   992→                    fontFamily={fontFamily}
   993→                    fontSize={fontSize}
   994→                  />
   995→                </section>
   996→              </>
   997→            )}
   998→          </div>
   999→        </ScrollArea>
  1000→      </DialogContent>
  1001→    </Dialog>
  1002→  )
