   108→// ---------------------------------------------------------------------------
   109→// Safe customizer hook (works with or without the CustomizerProvider).
   110→// ---------------------------------------------------------------------------
   111→
   112→function useCustomizerSafe(preferences: any) {
   113→  try {
   114→    const ctx = useCustomizer()
   115→    return ctx
   116→  } catch {
   117→    // CustomizerProvider not available — derive from preferences prop.
   118→    const wallpaper = WALLPAPERS.find((w) => w.id === preferences?.wallpaper) || WALLPAPERS[0]
   119→    const fontFamily = FONT_OPTIONS.find((f) => f.id === preferences?.fontFamily) || FONT_OPTIONS[0]
   120→    return {
   121→      preferences,
   122→      wallpaper,
   123→      fontFamily,
   124→      messageStyle: preferences?.messageStyle || 'bubble',
   125→      fontSize: preferences?.fontSize || 14,
   126→      setPreference: async () => {},
   127→    }
   128→  }
   129→}
