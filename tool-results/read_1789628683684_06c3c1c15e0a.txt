   236→  onOpenChat?: (c: {
   237→    conversationId: string
   238→    name: string

   244→// ============================================================
   245→// R8-11 — CapsuleCategorySlider
   246→// Horizontal scrolling capsule / pill chips for the Discover category
   247→// filter. Each capsule is a rounded-full button with scroll-snap-align
   248→// start so the slider snaps one capsule at a time. The active capsule
   249→// uses the brand emerald gradient + white text; inactive capsules are
   250→// subtle tinted pills. The whole row is horizontally scrollable with
   251→// hidden scrollbar (.no-scrollbar) so it stays clean on mobile.
   252→// ============================================================
   253→function CapsuleCategorySlider({
   254→  categories,
   255→  activeCategory,
   256→  onSelect,
   257→}: {
   258→  categories: string[]
   259→  activeCategory: string | null
   260→  onSelect: (cat: string) => void
   261→}) {
   262→  return (
   263→    <div
   264→      className="no-scrollbar scroll-pan-y -mx-4 mt-3 w-full overflow-x-auto px-4"
   265→      style={{
   266→        scrollSnapType: 'x proximity',
   267→        scrollPaddingLeft: '16px',
   268→        scrollPaddingRight: '16px',
   269→      }}
   270→    >
   271→      <div className="flex min-w-0 gap-2 pb-1">
   272→        {categories.map((cat) => {
   273→          const active = activeCategory === cat
   274→          return (
   275→            <button
   276→              key={cat}
   277→              type="button"
   278→              onClick={() => onSelect(cat)}
   279→              aria-pressed={active}
   280→              className={
   281→                'min-h-[40px] shrink-0 rounded-full border px-4 py-2 text-sm font-medium transition-colors ' +
   282→                (active
   283→                  ? 'btn-brand border-transparent text-primary-foreground shadow-sm'
   284→                  : 'border-border bg-card text-foreground/80 hover:bg-accent hover:text-foreground')
   285→              }
   286→              style={{ scrollSnapAlign: 'start' }}
   287→            >
   288→              {cat}
   289→            </button>
   290→          )
   291→        })}
   292→      </div>
   293→    </div>
   294→  )
   295→}

   297→export function DiscoverScreen({ onOpenChat }: DiscoverScreenProps = {}) {
   298→  const { toast } = useToast()
   299→  const [trending, setTrending] = useState<GroupItem[]>([])
   300→  const [popular, setPopular] = useState<GroupItem[]>([])
   301→  const [newGroups, setNewGroups] = useState<GroupItem[]>([])
   302→  const [sponsored, setSponsored] = useState<any[]>([])
   303→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   304→  const [categoryResults, setCategoryResults] = useState<GroupItem[]>([])
   305→  const [loading, setLoading] = useState(true)
   306→  const [catLoading, setCatLoading] = useState(false)
   307→  const [joining, setJoining] = useState<string | null>(null)
   308→  // V12 — Track the group currently being "left" (used to disable the Leave
   309→  // button + show a spinner during the leave API call).
   310→  const [leaving, setLeaving] = useState<string | null>(null)
   311→  // joinedIds = public joined; requestedIds = private groups with sent requests
   312→  const [joinedIds, setJoinedIds] = useState<Set<string>>(new Set())
   313→  const [requestedIds, setRequestedIds] = useState<Set<string>>(new Set())
   314→  const [preview, setPreview] = useState<GroupItem | null>(null)
   315→
   316→  useEffect(() => {
   317→    let cancelled = false
   318→    ;(async () => {
   319→      setLoading(true)
   320→      try {
   321→        const [t, p, n] = await Promise.all([
   322→          apiFetch('/api/discover?sort=trending').catch(() => null),
   323→          apiFetch('/api/discover?sort=popular').catch(() => null),
   324→          apiFetch('/api/discover?sort=new').catch(() => null),
   325→        ])
   326→        if (cancelled) return
   327→        // Defensive: API returns { groups: [...], sponsored: [...] }.
   328→        // Fall back to bare arrays if the shape is ever flat.
   329→        const tGroups: GroupItem[] = (Array.isArray(t)
   330→          ? t
   331→          : t?.groups || t?.items || []) as GroupItem[]
   332→        const pGroups: GroupItem[] = (Array.isArray(p)
   333→          ? p
   334→          : p?.groups || p?.items || []) as GroupItem[]
   335→        const nGroups: GroupItem[] = (Array.isArray(n)
   336→          ? n
   337→          : n?.groups || n?.items || []) as GroupItem[]
   338→        setTrending(tGroups)
   339→        setPopular(pGroups)
   340→        setNewGroups(nGroups)
   341→        // Sponsored ads: prefer the trending response's sponsored list,
   342→        // fall back to popular's, then to an empty array.
   343→        const sponsoredList: any[] = Array.isArray(t?.sponsored)
   344→          ? t.sponsored
   345→          : Array.isArray(p?.sponsored)
   346→            ? p.sponsored
   347→            : Array.isArray(n?.sponsored)
   348→              ? n.sponsored
   349→              : []
   350→        setSponsored(sponsoredList)
   351→        const joined = new Set<string>([
   352→          ...tGroups.filter((g) => g.isJoined).map((g) => g.id),
   353→          ...pGroups.filter((g) => g.isJoined).map((g) => g.id),
   354→          ...nGroups.filter((g) => g.isJoined).map((g) => g.id),
   355→        ])
   356→        setJoinedIds(joined)
   357→      } finally {
   358→        if (!cancelled) setLoading(false)
   359→      }
   360→    })()
   361→    return () => {
   362→      cancelled = true
   363→    }
   364→  }, [])
   365→
   366→  // Load the user's sent group join requests once on mount so we can mark
   367→  // private groups as "requested".
   368→  useEffect(() => {
   369→    ;(async () => {
   370→      try {
   371→        const res: any = await apiFetch('/api/groups/requests/sent')
   372→        const list: any[] = Array.isArray(res) ? res : (res?.requests || res?.items || [])
   373→        const pending = new Set<string>(
   374→          list
   375→            .filter((r: any) => r.status === 'pending' && r.group?.id)
   376→            .map((r: any) => r.group.id),
   377→        )
   378→        setRequestedIds(pending)
   379→      } catch {
   380→        // ignore — feature best-effort
   381→      }
   382→    })()
   383→  }, [])
   384→
   385→  const fetchCategory = async (cat: string) => {
   386→    setCatLoading(true)
   387→    try {
   388→      const res: any = await apiFetch(
   389→        `/api/discover?category=${encodeURIComponent(cat)}&sort=trending`,
   390→      )
   391→      // Defensive: API returns { groups: [...] }, fall back to bare array.
   392→      const list: GroupItem[] = (Array.isArray(res)
   393→        ? res
   394→        : res?.groups || res?.items || []) as GroupItem[]
   395→      setCategoryResults(list)
   396→    } catch (e: any) {
   397→      toast({ title: e?.message || 'Failed to load category', variant: 'destructive' })
   398→    } finally {
   399→      setCatLoading(false)
   400→    }
   401→  }
   402→
   403→  const handleCategoryClick = (cat: string) => {
   404→    if (activeCategory === cat) {
   405→      setActiveCategory(null)
   406→      setCategoryResults([])
   407→    } else {
   408→      setActiveCategory(cat)
   409→      fetchCategory(cat)
   410→    }
   411→  }
   412→
   413→  // Returns the join state for a group, factoring in joined + requested sets.
   414→  const joinStateFor = (g: GroupItem): JoinState => {
   415→    if (joinedIds.has(g.id)) return 'joined'
   416→    if (!g.isPublic && requestedIds.has(g.id)) return 'requested'
   417→    return 'idle'
   418→  }
   419→
   420→  const handleJoin = async (g: GroupItem) => {
   421→    setJoining(g.id)
   422→    try {
   423→      const res: any = await apiFetch('/api/groups/join', {
   424→        method: 'POST',
   425→        body: JSON.stringify({ groupId: g.id }),
   426→      })
   427→      if (res?.alreadyMember) {
   428→        toast({ title: 'You are already a member' })
   429→        setJoinedIds((prev) => new Set(prev).add(g.id))
   430→      } else if (res?.requested) {
   431→        // Private group → request was sent
   432→        toast({
   433→          title: 'Join request sent! Wait for admin approval.',
   434→        })
   435→        setRequestedIds((prev) => new Set(prev).add(g.id))
   436→      } else {
   437→        // Public group → direct join
   438→        toast({ title: `Joined ${g.name} 🎉` })
   439→        setJoinedIds((prev) => new Set(prev).add(g.id))
   440→      }
   441→    } catch (e: any) {
   442→      toast({ title: e?.message || 'Failed to join', variant: 'destructive' })
   443→    } finally {
   444→      setJoining(null)
   445→    }
   446→  }
   447→
   448→  // V12 — Open the chat for a joined group. Calls onOpenChat (wired from
   449→  // TalyApp) when a conversationId is linked to the group. Falls back to
   450→  // a friendly toast if the callback is missing (defensive — should never
   451→  // happen in production wiring, but avoids a no-op click).
   452→  const handleOpen = (g: GroupItem) => {
   453→    if (!g.conversationId) {
   454→      toast({
   455→        title: 'No chat linked to this group yet',
   456→        description: 'Please try again in a moment.',
   457→        variant: 'destructive',
   458→      })
   459→      return
   460→    }
   461→    if (onOpenChat) {
   462→      onOpenChat({
   463→        conversationId: g.conversationId,
   464→        name: g.name,
   465→        avatar: g.logo || undefined,
   466→        isGroup: true,
   467→      })
   468→    } else {
   469→      toast({ title: 'Open coming soon' })
   470→    }
   471→  }
   472→
   473→  // V12 — Leave a joined group. Calls POST /api/groups/[id]/leave, then
   474→  // removes the group id from joinedIds so the button reverts to Join.
   475→  const handleLeave = async (g: GroupItem) => {
   476→    setLeaving(g.id)
   477→    try {
   478→      await apiFetch(`/api/groups/${g.id}/leave`, { method: 'POST' })
   479→      toast({ title: `Left ${g.name}` })
   480→      setJoinedIds((prev) => {
   481→        const next = new Set(prev)
   482→        next.delete(g.id)
   483→        return next
   484→      })
   485→    } catch (e: any) {
   486→      toast({ title: e?.message || 'Failed to leave group', variant: 'destructive' })
   487→    } finally {
   488→      setLeaving(null)
   489→    }
   490→  }
   491→
   492→  // V12 — Reset the active category filter (used by the "Browse Trending
   493→  // instead" link in the empty state and by the active-filter banner's
   494→  // clear (✕) button).
   495→  const clearCategory = () => {
   496→    setActiveCategory(null)
   497→    setCategoryResults([])
   498→  }
   499→
   500→  const totalGroups = trending.length + popular.length + newGroups.length
   501→
   502→  return (
   503→    <div className="mx-auto max-w-2xl px-4 pb-20 pt-6 lg:pb-6 lg:pt-8">
   504→      {/* Header */}
   505→      <motion.div
   506→        initial={{ opacity: 0, y: -8 }}
   507→        animate={{ opacity: 1, y: 0 }}
   508→        transition={{ duration: 0.3 }}
   509→      >
   510→        <h1 className="section-header">Discover</h1>
   511→        <p className="mt-1 text-sm text-muted-foreground">
   512→          Find communities that match your interests
   513→        </p>
   514→      </motion.div>
   515→
   516→      {/* R8-11 — Capsule slider: horizontal scrolling capsule/pill chips for
   517→          the new category list (AI, Game, Fun, …). Each capsule is a pill
   518→          with rounded-full + scroll-snap-align start so the slider snaps
   519→          one capsule at a time. The active capsule uses the brand gradient
   520→          (btn-brand look) so it stands out from the rest. */}
   521→      <CapsuleCategorySlider
   522→        categories={CAPSULE_CATEGORIES as unknown as string[]}
   523→        activeCategory={activeCategory}
   524→        onSelect={handleCategoryClick}
   525→      />
   526→
   527→      {/* V12 — Active filter banner: shows below the chips row when a
   528→          category filter is selected, with a clear (✕) button to reset. */}
   529→      {activeCategory && (
   530→        <div className="mt-3 flex items-center justify-between gap-2 rounded-lg bg-emerald-50 px-3 py-2 text-xs font-medium text-emerald-700 dark:bg-emerald-950/20 dark:text-emerald-300">
   531→          <span className="truncate">
   532→            Showing results for{' '}
   533→            <span className="font-bold">&lsquo;{activeCategory}&rsquo;</span>
   534→          </span>
   535→          <button
   536→            type="button"
   537→            onClick={clearCategory}
   538→            aria-label="Clear category filter"
   539→            className="inline-flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-emerald-700 transition-colors hover:bg-emerald-500/15 dark:text-emerald-300"
   540→          >
   541→            <X className="h-3.5 w-3.5" />
   542→          </button>
   543→        </div>
   544→      )}
   545→
   546→      {/* Empty state — no groups loaded */}
   547→      {!loading && totalGroups === 0 && !activeCategory && (
   548→        <div className="dotted-bg mt-6 flex flex-col items-center justify-center rounded-2xl border border-border bg-card p-10 text-center">
   549→          <Compass className="h-12 w-12 text-muted-foreground/60" />
   550→          <p className="mt-3 text-sm font-semibold text-foreground">
   551→            No communities yet
   552→          </p>
   553→          <p className="mt-1 text-xs text-muted-foreground">
   554→            Check back soon — new communities are added every day.
   555→          </p>
   556→        </div>
   557→      )}
   558→
   559→      {/* Category results — when a capsule is selected, show every group
   560→          in that category as a vertical list (no slider). */}
   561→      {activeCategory && (
   562→        <section className="mt-4 animate-fade-in-up">
   563→          <h2 className="section-header">
   564→            {activeCategory} communities
   565→          </h2>
   566→          <div className="mt-3">
   567→            {catLoading ? (
   568→              <InlineLoadingRow />
   569→            ) : categoryResults.length === 0 ? (
   570→              // Friendly no-results empty state: magnifying glass + clear
   571→              // filter CTA. The old "Create {Category} Group" button has
   572→              // been removed because group creation is no longer
   573→              // available from the Groups tab (R8-11).
   574→              <div className="flex flex-col items-center justify-center rounded-2xl border border-dashed border-border bg-card p-8 text-center">
   575→                <Search className="h-12 w-12 text-muted-foreground/40" />
   576→                <p className="mt-3 text-sm font-semibold text-foreground">

   727→function Section({
   728→  title,
   729→  icon,
   730→  loading,
   731→  groups,
   732→  joinStateFor,
   733→  joining,
   734→  leaving,
   735→  onJoin,
   736→  onOpen,
   737→  onLeave,
   738→  onPreview,
   739→  delay = 0,
   740→}: {
   741→  title: string
   742→  icon?: string
   743→  loading: boolean
   744→  groups: GroupItem[]
   745→  joinStateFor: (g: GroupItem) => JoinState
   746→  joining: string | null
   747→  leaving: string | null

   749→  onOpen: (g: GroupItem) => void
   750→  onLeave: (g: GroupItem) => void
   751→  onPreview: (g: GroupItem) => void

   754→  // R8-11 — 2-at-a-time horizontal slider. Cards are 50% of the visible
   755→  // container width (calc(50% - 6px) so two cards + the 12px gap fit the
   756→  // viewport exactly). The slider uses CSS scroll-snap so each swipe
   757→  // advances by one card width, revealing the next pair.
   758→  const trackRef = useRef<HTMLDivElement | null>(null)
   759→
   760→  const scrollByCard = (direction: 1 | -1) => {
   761→    const el = trackRef.current
   762→    if (!el) return
   763→    // Advance by ~2 card widths (= ~one full viewport of 2 cards).
   764→    const cardWidth = el.clientWidth * 0.5
   765→    el.scrollBy({ left: direction * cardWidth * 2, behavior: 'smooth' })
   766→  }
   767→
   768→  // Show a chevron on either side only when there's something to scroll to.
   769→  // We use the simpler "more than 2 groups" check — the scroll container's
   770→  // overflow itself indicates scrollability.
   771→  const canScroll = !loading && groups.length > 2
   772→
   773→  return (
   774→    <section
   775→      className="mt-6 animate-fade-in-up"

   782→        {canScroll && (
   783→          <div className="flex items-center gap-1">
   784→            <button
   785→              type="button"
   786→              onClick={() => scrollByCard(-1)}
   787→              aria-label={`Previous ${title} groups`}
   788→              className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full border border-border text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   789→            >
   790→              <ChevronLeft className="h-4 w-4" />
   791→            </button>
   792→            <button
   793→              type="button"
   794→              onClick={() => scrollByCard(1)}
   795→              aria-label={`Next ${title} groups`}
   796→              className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full border border-border text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   797→            >
   798→              <ChevronRight className="h-4 w-4" />
   799→            </button>
   800→          </div>
   801→        )}
   802→      </div>
   803→
   804→      {/* 2-cards-at-a-time horizontal slider with scroll-snap */}
   805→      <div
   806→        ref={trackRef}
   807→        className="no-scrollbar scroll-pan-y -mx-4 w-full overflow-x-auto px-4"
   808→        style={{
   809→          scrollSnapType: 'x mandatory',
   810→          scrollPaddingLeft: '16px',
   811→          scrollPaddingRight: '16px',
   812→        }}

   814→        {loading ? (
   815→          <CardLoadingRow />
   816→        ) : groups.length === 0 ? (
   817→          <p className="px-4 text-sm text-muted-foreground">No groups yet.</p>

   820→            className="flex min-w-0 pb-1"

   828→                  width: 'calc(50% - 6px)',
   829→                  scrollSnapAlign: 'start',
   830→                  scrollSnapStop: 'normal',
   831→                }}
   832→              >
   833→                <GroupCard

  1039→// MiniStackedAvatars — 2-3 small overlapping letter-avatars with +N.
  1040→// Used inside group cards/rows to humanize the member count display.
  1041→// ============================================================
  1042→function MiniStackedAvatars({
  1043→  logo,
  1044→  name,
  1045→  memberCount,
  1046→  small = false,
  1047→}: {
  1048→  logo?: string | null
  1049→  name: string
  1050→  memberCount: number
  1051→  small?: boolean
  1052→}) {
  1053→  const baseName = name || 'G'
  1054→  const initials = (baseName.replace(/\s+/g, ' ').trim().split(' ').filter(Boolean))
  1055→  const derived: string[] = []
  1056→  for (let i = 0; i < initials.length && derived.length < 3; i++) {
  1057→    derived.push(initials[i][0]?.toUpperCase() || 'U')
  1058→  }
  1059→  while (derived.length < 3) {
  1060→    derived.push(baseName[0]?.toUpperCase() || 'U')
  1061→  }
  1062→  const visible = derived.slice(0, 3)
  1063→  const extra = Math.max(0, memberCount - 3)
  1064→  const sizeClass = small ? 'h-4 w-4' : 'h-5 w-5'
  1065→  const textSize = small ? 'text-[7px]' : 'text-[8px]'
  1066→  const offset = small ? 8 : 10
  1067→  const slotClass = (i: number) => {
  1068→    switch (i % 3) {
  1069→      case 0:
  1070→        return 'bg-emerald-500/15 text-emerald-600'
  1071→      case 1:
  1072→        return 'bg-amber-500/15 text-amber-600'
  1073→      default:
  1074→        return 'bg-sky-500/15 text-sky-600'
  1075→    }
  1076→  }
  1077→  return (
  1078→    <span className="relative inline-flex items-center" aria-hidden>
  1079→      <span className="relative inline-flex" style={{ minWidth: `${offset * 2 + 8}px` }}>
  1080→        {visible.map((letter, i) => {
  1081→          const z = visible.length - i
  1082→          return (
  1083→            <span
  1084→              key={i}
  1085→              className={`absolute flex ${sizeClass} items-center justify-center rounded-full ${textSize} font-bold ring-1 ring-card ${slotClass(i)}`}
  1086→              style={{ left: `${i * offset}px`, zIndex: z }}
  1087→            >
  1088→              {i === 0 && logo ? (
  1089→                <img
  1090→                  src={logo}
  1091→                  alt=""
  1092→                  className="h-full w-full rounded-full object-cover"
  1093→                />
  1094→              ) : (
  1095→                letter
  1096→              )}
  1097→            </span>
  1098→          )
  1099→        })}
  1100→        {extra > 0 && (
  1101→          <span
  1102→            className={`absolute flex ${sizeClass} items-center justify-center rounded-full bg-muted ${textSize} font-bold text-muted-foreground ring-1 ring-card`}
  1103→            style={{ left: `${visible.length * offset}px`, zIndex: visible.length + 1 }}
  1104→          >
  1105→            +{extra > 99 ? '99' : extra}
  1106→          </span>
  1107→        )}
  1108→      </span>
  1109→    </span>
  1110→  )
  1111→}

  1139→
